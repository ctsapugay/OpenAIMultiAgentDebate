// Player state to track quests, clues, inventory, and current NPC/dialogue
const playerState = {
  currentNpcId: null,
  currentDialogueKey: 'start',
  quests: {}, // questId -> status ('not accepted', 'active', 'completed')
  clues: new Set(),
  inventory: {
    ironOre: 0
  }
};

// Cache DOM elements for performance
const npcListEl = document.getElementById('npc-list');
const npcNameEl = document.getElementById('npc-name');
const dialogueTextEl = document.getElementById('dialogue-text');
const choicesEl = document.getElementById('choices');
const questListEl = document.getElementById('quest-list');
const cluesListEl = document.getElementById('clues-list');

/**
 * Initialize the NPC list sidebar with clickable NPCs.
 */
function initNpcList() {
  npcs.forEach(npc => {
    const npcItem = document.createElement('div');
    npcItem.classList.add('npc-item');
    npcItem.textContent = npc.name;
    npcItem.setAttribute('tabindex', '0');
    npcItem.setAttribute('role', 'button');
    npcItem.setAttribute('aria-pressed', 'false');
    npcItem.dataset.npcId = npc.id;

    // Click and keyboard support
    npcItem.addEventListener('click', () => selectNpc(npc.id));
    npcItem.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        selectNpc(npc.id);
      }
    });

    npcListEl.appendChild(npcItem);
  });
}

/**
 * Select an NPC to start or continue conversation.
 * @param {string} npcId 
 */
function selectNpc(npcId) {
  if (playerState.currentNpcId === npcId) return; // Already selected

  playerState.currentNpcId = npcId;

  // Reset dialogue key to 'start' or to a special dialogue if quest active/completed
  const npc = getCurrentNpc();

  // Determine dialogue start point based on quest states
  let dialogueKey = 'start';

  // Check if NPC has quests and if any are active or completed to adjust dialogue
  if (npc.quests.length > 0) {
    for (const quest of npc.quests) {
      const status = playerState.quests[quest.id] || 'not accepted';
      if (status === 'active') {
        dialogueKey = 'after_quest';
        break;
      } else if (status === 'completed') {
        dialogueKey = 'quest_completed';
        break;
      }
    }
  }

  playerState.currentDialogueKey = dialogueKey;

  updateNpcListActive();
  renderDialogue();
}

/**
 * Get the currently selected NPC object.
 * @returns {object|null}
 */
function getCurrentNpc() {
  return npcs.find(npc => npc.id === playerState.currentNpcId) || null;
}

/**
 * Update the NPC list UI to highlight the selected NPC.
 */
function updateNpcListActive() {
  const npcItems = npcListEl.querySelectorAll('.npc-item');
  npcItems.forEach(item => {
    if (item.dataset.npcId === playerState.currentNpcId) {
      item.classList.add('active');
      item.setAttribute('aria-pressed', 'true');
      item.focus();
    } else {
      item.classList.remove('active');
      item.setAttribute('aria-pressed', 'false');
    }
  });
}

/**
 * Render the current dialogue text and choices for the selected NPC.
 */
function renderDialogue() {
  const npc = getCurrentNpc();
  if (!npc) {
    npcNameEl.textContent = '';
    dialogueTextEl.textContent = 'Select an NPC to start conversation.';
    choicesEl.innerHTML = '';
    return;
  }

  npcNameEl.textContent = npc.name;

  // Get dialogue node by currentDialogueKey
  let dialogueNode = npc.dialogues[playerState.currentDialogueKey];

  // If dialogue node not found, fallback to 'start'
  if (!dialogueNode) {
    dialogueNode = npc.dialogues['start'];
    playerState.currentDialogueKey = 'start';
  }

  dialogueTextEl.textContent = dialogueNode.text;

  // Clear previous choices
  choicesEl.innerHTML = '';

  // If no choices, show a disabled button to close dialogue
  if (!dialogueNode.choices || dialogueNode.choices.length === 0) {
    const btn = document.createElement('button');
    btn.className = 'choice-btn';
    btn.textContent = 'Close';
    btn.disabled = true;
    choicesEl.appendChild(btn);
    return;
  }

  // Create buttons for each choice
  dialogueNode.choices.forEach((choice, index) => {
    const btn = document.createElement('button');
    btn.className = 'choice-btn';
    btn.textContent = choice.text;
    btn.setAttribute('role', 'listitem');
    btn.addEventListener('click', () => handleChoice(choice));
    choicesEl.appendChild(btn);
  });
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  // Process any action associated with the choice
  if (choice.action) {
    switch (choice.action) {
      case 'acceptQuest':
        acceptQuest(choice.questId);
        break;
      case 'completeQuest':
        completeQuest(choice.questId);
        break;
      case 'giveClue':
        addClue(choice.clue);
        break;
      case 'forgeWeapon':
        forgeWeapon();
        break;
      default:
        console.warn('Unknown action:', choice.action);
    }
  }

  // Move to the next dialogue node if specified
  if (choice.next) {
    playerState.currentDialogueKey = choice.next;
  } else {
    // If no next, end conversation (reset dialogue)
    playerState.currentDialogueKey = 'end';
  }

  renderDialogue();
  renderPlayerStatus();
}

/**
 * Accept a quest by questId.
 * @param {string} questId 
 */
function acceptQuest(questId) {
  // Mark quest as active in player state
  playerState.quests[questId] = 'active';
}

/**
 * Complete a quest by questId.
 * @param {string} questId 
 */
function completeQuest(questId) {
  // Mark quest as completed in player state
  playerState.quests[questId] = 'completed';
}

/**
 * Add a clue to the player's clue list.
 * @param {string} clue 
 */
function addClue(clue) {
  playerState.clues.add(clue);
}

/**
 * Simulate forging a weapon if player has enough iron ore.
 */
function forgeWeapon() {
  if (playerState.inventory.ironOre >= 5) {
    playerState.inventory.ironOre -= 5;
    alert("Blacksmith Tor has forged a sword for you!");
  } else {
    alert("You don't have enough iron ore. Collect 5 to forge a weapon.");
    // Force dialogue back to start since forging failed
    playerState.currentDialogueKey = 'start';
  }
}

/**
 * Render the player's current quests and clues in the status panel.
 */
function renderPlayerStatus() {
  // Clear quest list
  questListEl.innerHTML = '';

  // Show quests with their status
  const activeQuests = [];
  const completedQuests = [];

  // Collect quests from all NPCs and merge with player state
  npcs.forEach(npc => {
    npc.quests.forEach(quest => {
      const status = playerState.quests[quest.id] || 'not accepted';
      if (status === 'active') activeQuests.push({ ...quest, status });
      else if (status === 'completed') completedQuests.push({ ...quest, status });
    });
  });

  if (activeQuests.length === 0 && completedQuests.length === 0) {
    const noQuest = document.createElement('div');
    noQuest.textContent = 'No active quests.';
    noQuest.style.opacity = '0.7';
    questListEl.appendChild(noQuest);
  } else {
    activeQuests.forEach(quest => {
      const questEl = document.createElement('div');
      questEl.className = 'quest-item';
      questEl.textContent = `🕒 ${quest.title}`;
      questEl.title = quest.description;
      questListEl.appendChild(questEl);
    });
    completedQuests.forEach(quest => {
      const questEl = document.createElement('div');
      questEl.className = 'quest-item completed';
      questEl.textContent = `✔ ${quest.title}`;
      questEl.title = quest.description;
      questListEl.appendChild(questEl);
    });
  }

  // Render clues
  cluesListEl.innerHTML = '';
  if (playerState.clues.size === 0) {
    const noClue = document.createElement('div');
    noClue.textContent = 'No clues obtained.';
    noClue.style.opacity = '0.7';
    cluesListEl.appendChild(noClue);
  } else {
    playerState.clues.forEach(clue => {
      const clueEl = document.createElement('div');
      clueEl.className = 'clue-item';
      clueEl.textContent = `🔍 ${clue}`;
      cluesListEl.appendChild(clueEl);
    });
  }
}

/**
 * Initialize the entire system.
 */
function init() {
  initNpcList();
  renderPlayerStatus();
  // Select first NPC by default
  if (npcs.length > 0) {
    selectNpc(npcs[0].id);
  } else {
    dialogueTextEl.textContent = 'No NPCs available.';
  }
}

// Initialize on page load
window.addEventListener('DOMContentLoaded', init);

/**
 * Select an NPC to start or continue conversation.
 * @param {string} npcId 
 */
function selectNpc(npcId) {
  if (playerState.currentNpcId === npcId) return; // Already selected

  playerState.currentNpcId = npcId;

  // Reset dialogue key to 'start' or to a special dialogue if quest active/completed
  const npc = getCurrentNpc();

  // Determine dialogue start point based on quest states
  let dialogueKey = 'start';

  // Check if NPC has quests and if any are active or completed to adjust dialogue
  if (npc.quests.length > 0) {
    for (const quest of npc.quests) {
      const status = playerState.quests[quest.id] || 'not accepted';
      if (status === 'active') {
        dialogueKey = 'after_quest';
        break;
      } else if (status === 'completed') {
        dialogueKey = 'quest_completed';
        break;
      }
    }
  }

  playerState.currentDialogueKey = dialogueKey;

  updateNpcListActive();
  renderDialogue();
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  // Process any action associated with the choice
  if (choice.action) {
    switch (choice.action) {
      case 'acceptQuest':
        acceptQuest(choice.questId);
        break;
      case 'completeQuest':
        completeQuest(choice.questId);
        break;
      case 'giveClue':
        addClue(choice.clue);
        break;
      case 'forgeWeapon':
        forgeWeapon();
        break;
      default:
        console.warn('Unknown action:', choice.action);
    }
  }

  // Move to the next dialogue node if specified
  if (choice.next) {
    playerState.currentDialogueKey = choice.next;
  } else {
    // If no next, end conversation (reset dialogue)
    playerState.currentDialogueKey = 'end';
  }

  renderDialogue();
  renderPlayerStatus();
}

/**
 * Select an NPC to start or continue conversation.
 * @param {string} npcId 
 */
function selectNpc(npcId) {
  if (!npcId || typeof npcId !== 'string') {
    showFeedback('Invalid NPC selection.');
    return;
  }
  if (playerState.currentNpcId === npcId) return; // Already selected

  const npc = npcs.find(n => n.id === npcId);
  if (!npc) {
    showFeedback('NPC not found.');
    return;
  }

  playerState.currentNpcId = npcId;

  // Reset dialogue key to 'start' or to a special dialogue if quest active/completed
  // Determine dialogue start point based on quest states
  let dialogueKey = 'start';

  try {
    if (npc.quests && npc.quests.length > 0) {
      for (const quest of npc.quests) {
        const status = playerState.quests[quest.id] || 'not accepted';
        if (status === 'active') {
          dialogueKey = 'after_quest';
          break;
        } else if (status === 'completed') {
          dialogueKey = 'quest_completed';
          break;
        }
      }
    }
  } catch (e) {
    console.error('Error determining dialogue start:', e);
    dialogueKey = 'start';
  }

  playerState.currentDialogueKey = dialogueKey;

  updateNpcListActive();
  renderDialogue();
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  if (!choice || typeof choice !== 'object') {
    showFeedback('Invalid choice selected.');
    return;
  }

  // Process any action associated with the choice
  if (choice.action) {
    try {
      switch (choice.action) {
        case 'acceptQuest':
          if (typeof choice.questId === 'string') {
            acceptQuest(choice.questId);
          } else {
            showFeedback('Quest ID missing or invalid.');
          }
          break;
        case 'completeQuest':
          if (typeof choice.questId === 'string') {
            completeQuest(choice.questId);
          } else {
            showFeedback('Quest ID missing or invalid.');
          }
          break;
        case 'giveClue':
          if (typeof choice.clue === 'string') {
            addClue(choice.clue);
          } else {
            showFeedback('Clue data missing or invalid.');
          }
          break;
        case 'forgeWeapon':
          forgeWeapon();
          break;
        default:
          console.warn('Unknown action:', choice.action);
          showFeedback(`Unknown action: ${choice.action}`);
      }
    } catch (e) {
      console.error('Error handling action:', e);
      showFeedback('An error occurred processing your choice.');
    }
  }

  // Move to the next dialogue node if specified
  if (choice.next) {
    playerState.currentDialogueKey = choice.next;
  } else {
    // If no next, end conversation (reset dialogue)
    playerState.currentDialogueKey = 'end';
  }

  renderDialogue();
  renderPlayerStatus();
}

function acceptQuest(questId) {
  if (!questId) {
    showFeedback('Invalid quest ID.');
    return;
  }
  // Mark quest as active in player state
  playerState.quests[questId] = 'active';
  showFeedback('Quest accepted!');
}

/**
 * Complete a quest by questId.
 * @param {string} questId 
 */
function completeQuest(questId) {
  if (!questId) {
    showFeedback('Invalid quest ID.');
    return;
  }
  // Mark quest as completed in player state
  if (playerState.quests[questId] === 'active') {
    playerState.quests[questId] = 'completed';
    showFeedback('Quest completed!');
  } else {
    showFeedback('You have not accepted this quest.');
  }
}

/**
 * Add a clue to the player's clue list.
 * @param {string} clue 
 */
function addClue(clue) {
  if (!clue) {
    showFeedback('Invalid clue.');
    return;
  }
  if (!playerState.clues.has(clue)) {
    playerState.clues.add(clue);
    showFeedback(`New clue obtained: ${clue}`);
  } else {
    showFeedback(`You already have the clue: ${clue}`);
  }
}

/**
 * Simulate forging a weapon if player has enough iron ore.
 */
function forgeWeapon() {
  if (playerState.inventory.ironOre >= 5) {
    playerState.inventory.ironOre -= 5;
    showFeedback("Blacksmith Tor has forged a sword for you!");
  } else {
    showFeedback("You don't have enough iron ore. Collect 5 to forge a weapon.");
    // Force dialogue back to start since forging failed
    playerState.currentDialogueKey = 'start';
  }
}

      if (status === 'active') activeQuests.push({ ...quest, status });
      else if (status === 'completed') completedQuests.push({ ...quest, status });
    });

  if (activeQuests.length === 0 && completedQuests.length === 0) {

  let dialogueKey = 'start';

  try {
    if (npc.quests && npc.quests.length > 0) {
      for (const quest of npc.quests) {
        const status = playerState.quests[quest.id] || 'not accepted';
        if (status === 'active') {
          dialogueKey = 'after_quest';
          break;
        } else if (status === 'completed') {
          dialogueKey = 'quest_completed';
          break;
        }
      }
    }
  } catch (e) {
    console.error('Error determining dialogue start:', e);
    dialogueKey = 'start';
  }

  playerState.currentDialogueKey = dialogueKey;

  updateNpcListActive();
  renderDialogue();
}

/**
 * Get the currently selected NPC object.
 * @returns {object|null}
 */
function getCurrentNpc() {
  if (!playerState.currentNpcId) return null;
  return npcs.find(npc => npc.id === playerState.currentNpcId) || null;
}

/**
 * Update the NPC list UI to highlight the selected NPC.
 */
function updateNpcListActive() {
  const npcItems = npcListEl.querySelectorAll('.npc-item');
  npcItems.forEach(item => {
    if (item.dataset.npcId === playerState.currentNpcId) {
      item.classList.add('active');
      item.setAttribute('aria-pressed', 'true');
      item.focus();
    } else {
      item.classList.remove('active');
      item.setAttribute('aria-pressed', 'false');
    }
  });
}

/**
 * Render the current dialogue text and choices for the selected NPC.
 */
function renderDialogue() {
  const npc = getCurrentNpc();
  if (!npc) {
    npcNameEl.textContent = '';
    dialogueTextEl.textContent = 'Select an NPC to start conversation.';
    choicesEl.innerHTML = '';
    return;
  }

  npcNameEl.textContent = npc.name;

  // Get dialogue node by currentDialogueKey
  let dialogueNode = null;
  try {
    dialogueNode = npc.dialogues[playerState.currentDialogueKey];
  } catch (e) {
    console.error('Error accessing dialogue node:', e);
  }

  // If dialogue node not found, fallback to 'start'
  if (!dialogueNode) {
    dialogueNode = npc.dialogues['start'];
    playerState.currentDialogueKey = 'start';
  }

  dialogueTextEl.textContent = dialogueNode.text || '';

  // Clear previous choices
  choicesEl.innerHTML = '';

  // If no choices, show a disabled button to close dialogue
  if (!dialogueNode.choices || dialogueNode.choices.length === 0) {
    const btn = document.createElement('button');
    btn.className = 'choice-btn';
    btn.textContent = 'Close';
    btn.disabled = true;
    choicesEl.appendChild(btn);
    return;
  }

  // Create buttons for each choice
  dialogueNode.choices.forEach((choice, index) => {
    const btn = document.createElement('button');
    btn.className = 'choice-btn';
    btn.textContent = choice.text || '...';
    btn.setAttribute('role', 'listitem');
    btn.addEventListener('click', () => handleChoice(choice));
    choicesEl.appendChild(btn);
  });
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  if (!choice || typeof choice !== 'object') {
    showFeedback('Invalid choice selected.');
    return;
  }

  // Process any action associated with the choice
  if (choice.action) {
    try {
      switch (choice.action) {
        case 'acceptQuest':
          if (typeof choice.questId === 'string') {
            acceptQuest(choice.questId);
          } else {
            showFeedback('Quest ID missing or invalid.');
          }
          break;
        case 'completeQuest':
          if (typeof choice.questId === 'string') {
            completeQuest(choice.questId);
          } else {
            showFeedback('Quest ID missing or invalid.');
          }
          break;
        case 'giveClue':
          if (typeof choice.clue === 'string') {
            addClue(choice.clue);
          } else {
            showFeedback('Clue data missing or invalid.');
          }
          break;
        case 'forgeWeapon':
          forgeWeapon();
          break;
        default:
          console.warn('Unknown action:', choice.action);
          showFeedback(`Unknown action: ${choice.action}`);
      }
    } catch (e) {
      console.error('Error handling action:', e);
      showFeedback('An error occurred processing your choice.');
    }
  }

  // Move to the next dialogue node if specified
  if (choice.next) {
    playerState.currentDialogueKey = choice.next;
  } else {
    // If no next, end conversation (reset dialogue)
    playerState.currentDialogueKey = 'end';
  }

  renderDialogue();
  renderPlayerStatus();
}

/**
 * Accept a quest by questId.
 * @param {string} questId 
 */
function acceptQuest(questId) {
  if (!questId) {
    showFeedback('Invalid quest ID.');
    return;
  }
  // Mark quest as active in player state
  playerState.quests[questId] = 'active';
  showFeedback('Quest accepted!');
}

/**
 * Complete a quest by questId.
 * @param {string} questId 
 */
function completeQuest(questId) {
  if (!questId) {
    showFeedback('Invalid quest ID.');
    return;
  }
  // Mark quest as completed in player state
  if (playerState.quests[questId] === 'active') {
    playerState.quests[questId] = 'completed';
    showFeedback('Quest completed!');
  } else {
    showFeedback('You have not accepted this quest.');
  }
}

/**
 * Add a clue to the player's clue list.
 * @param {string} clue 
 */
function addClue(clue) {
  if (!clue) {
    showFeedback('Invalid clue.');
    return;
  }
  if (!playerState.clues.has(clue)) {
    playerState.clues.add(clue);
    showFeedback(`New clue obtained: ${clue}`);
  } else {
    showFeedback(`You already have the clue: ${clue}`);
  }
}

/**
 * Simulate forging a weapon if player has enough iron ore.
 */
function forgeWeapon() {
  if (playerState.inventory.ironOre >= 5) {
    playerState.inventory.ironOre -= 5;
    showFeedback("Blacksmith Tor has forged a sword for you!");
  } else {
    showFeedback("You don't have enough iron ore. Collect 5 to forge a weapon.");
    // Force dialogue back to start since forging failed
    playerState.currentDialogueKey = 'start';
  }
}

/**
 * Render the player's current quests and clues in the status panel.
 */
function renderPlayerStatus() {
  // Clear quest list
  questListEl.innerHTML = '';

  // Show quests with their status
  const activeQuests = [];
  const completedQuests = [];

  // Collect quests from all NPCs and merge with player state
  npcs.forEach(npc => {
    if (!npc.quests) return;
    npc.quests.forEach(quest => {
      const status = playerState.quests[quest.id] || 'not accepted';
      if (status === 'active') activeQuests.push({ ...quest, status });
      else if (status === 'completed') completedQuests.push({ ...quest, status });
    });
  });

  if (activeQuests.length === 0 && completedQuests.length === 0) {
    const noQuest = document.createElement('div');
    noQuest.textContent = 'No active quests.';
    noQuest.style.opacity = '0.7';
    questListEl.appendChild(noQuest);
  } else {
    activeQuests.forEach(quest => {
      const questEl = document.createElement('div');
      questEl.className = 'quest-item';
      questEl.textContent = `🕒 ${quest.title}`;
      questEl.title = quest.description || '';
      questListEl.appendChild(questEl);
    });
    completedQuests.forEach(quest => {
      const questEl = document.createElement('div');
      questEl.className = 'quest-item completed';
      questEl.textContent = `✔ ${quest.title}`;
      questEl.title = quest.description || '';
      questListEl.appendChild(questEl);
    });
  }

  // Render clues
  cluesListEl.innerHTML = '';
  if (playerState.clues.size === 0) {
    const noClue = document.createElement('div');
    noClue.textContent = 'No clues obtained.';
    noClue.style.opacity = '0.7';
    cluesListEl.appendChild(noClue);
  } else {
    playerState.clues.forEach(clue => {
      const clueEl = document.createElement('div');
      clueEl.className = 'clue-item';
      clueEl.textContent = `🔍 ${clue}`;
      cluesListEl.appendChild(clueEl);
    });
  }
}

/**
 * Initialize the entire system.
 */
function init() {
  try {
    initNpcList();
    renderPlayerStatus();
    // Select first NPC by default
    if (npcs.length > 0) {
      selectNpc(npcs[0].id);
    } else {
      dialogueTextEl.textContent = 'No NPCs available.';
    }
  } catch (e) {
    console.error('Initialization error:', e);
    dialogueTextEl.textContent = 'An error occurred initializing the system.';
  }
}

// Initialize on page load
window.addEventListener('DOMContentLoaded', init);
