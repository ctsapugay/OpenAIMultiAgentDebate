    if (!blockA || !blockB) return false;
    const typeA = BLOCK_TYPES[blockA.type];
    const typeB = BLOCK_TYPES[blockB.type];
    if (!typeA || !typeB) return false;
    const rotA = blockA.rotation;
    const rotB = blockB.rotation;

    // Calculate actual connection directions for blockA

    // Calculate actual connection directions for blockB

    // blockA must have connection in dir
    if (!connsA.includes(dir)) return false;
    // blockB must have connection in opposite direction
    if (!connsB.includes(oppositeDir(dir))) return false;

    return true;
  }

  function checkPath() {
    const { start, end } = findStartEnd();
    if (!start || !end) return false;

    // BFS from start to find end through connected pipes
    const visited = new Set();
    const queue = [];
    queue.push(start);
    visited.add(`${start.r},${start.c}`);

    while (queue.length > 0) {
      const current = queue.shift();
      if (current === end) return true;

      // Check neighbors in all directions
      for (let dir = 0; dir < 4; dir++) {
        const nr = current.r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
        const nc = current.c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
        if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) continue;
        const neighbor = blocks[nr][nc];
        if (!neighbor) continue;
        if (neighbor.type === 'empty') continue;
        if (visited.has(`${nr},${nc}`)) continue;
        if (blocksConnect(current, dir, neighbor)) {
          visited.add(`${nr},${nc}`);
          queue.push(neighbor);
        }
      }
    }
    return false;
  }

    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (!blockGroup) return;
    blockGroup.style.transition = 'none';
    blockGroup.style.position = 'relative';
    blockGroup.style.zIndex = 1000;
    blockGroup.style.left = dx + 'px';
    blockGroup.style.top = dy + 'px';
  }

  function onPointerUp(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (blockGroup) {
      blockGroup.style.transition = '';
      blockGroup.style.position = '';
      blockGroup.style.zIndex = '';
      blockGroup.style.left = '';
      blockGroup.style.top = '';
    }
    if (dragData.dragging) {
      // Calculate new cell based on pointer position relative to board
      const boardRect = boardEl.getBoundingClientRect();
      const x = e.clientX - boardRect.left;
      const y = e.clientY - boardRect.top;
      const newC = Math.floor(x / CELL_SIZE);
      const newR = Math.floor(y / CELL_SIZE);
      if (
        newR >= 0 && newR < GRID_SIZE &&
        newC >= 0 && newC < GRID_SIZE &&
        blocks[newR][newC].type === 'empty'
      ) {
        // Move block to new position
        moveBlock(selectedBlock.r, selectedBlock.c, newR, newC);
        announce(`Moved block to row ${newR + 1}, column ${newC + 1}.`);
      } else {
        announce('Invalid move. Block returned to original position.');
      }
    } else {
      // No drag, treat as click: rotate block
      rotateBlock(selectedBlock);
      announce(`Rotated block at row ${selectedBlock.r + 1}, column ${selectedBlock.c + 1}.`);
    }
    dragData = null;
    selectedBlock.el.classList.remove('dragging');
    selectedBlock = null;
    checkWinCondition();
  }

  // Move block from (r1,c1) to (r2,c2)
  function moveBlock(r1, c1, r2, c2) {
    if (r1 === r2 && c1 === c2) return;
    const block = blocks[r1][c1];
    if (!block) return;
    if (blocks[r2][c2].type !== 'empty') return;

    // Swap blocks in data
    blocks[r2][c2] = {
      ...block,
      r: r2,
      c: c2,
    };
    blocks[r1][c1] = {
      type: 'empty',
      rotation: 0,
      r: r1,
      c: c1,
      el: null,
    };

    // Update DOM elements
    // Remove old cell groups
    svgEl.removeChild(blocks[r2][c2].el);
    svgEl.removeChild(blocks[r1][c1].el);

    // Create new cells for both positions
    const newCell1 = createCell(r1, c1, 'empty', 0);
    const newCell2 = createCell(r2, c2, block.type, block.rotation);
    svgEl.appendChild(newCell1);
    svgEl.appendChild(newCell2);

    blocks[r1][c1].el = newCell1;
    blocks[r2][c2].el = newCell2;

    // Update block reference for moved block
    blocks[r2][c2].el = newCell2;
    blocks[r2][c2].r = r2;
    blocks[r2][c2].c = c2;

    // Re-attach event listeners for new cells

    attachCellListeners(newCell2);
  }

  // Update block SVG rotation transform

  // Find start and end positions
  function findStartEnd() {
    let start = null;
    let end = null;
    for (let r = 0; r < GRID_SIZE; r++) {
      for (let c = 0; c < GRID_SIZE; c++) {
        if (blocks[r][c].type === 'start') start = blocks[r][c];
        if (blocks[r][c].type === 'end') end = blocks[r][c];
      }

    return { start, end };

  // Check if two blocks connect in a given direction
  function blocksConnect(blockA, dir, blockB) {
    if (!blockA || !blockB) return false;
    const typeA = BLOCK_TYPES[blockA.type];
    const typeB = BLOCK_TYPES[blockB.type];
    if (!typeA || !typeB) return false;
    const rotA = blockA.rotation;
    const rotB = blockB.rotation;

    // Calculate actual connection directions for blockA
    const connsA = typeA.connections.map(d => (d + rotA) % 4);
    // Calculate actual connection directions for blockB
    const connsB = typeB.connections.map(d => (d + rotB) % 4);

    // blockA must have connection in dir
    if (!connsA.includes(dir)) return false;
    // blockB must have connection in opposite direction
    if (!connsB.includes(oppositeDir(dir))) return false;

    return true;
  }

  // Check if the path from start to end is connected
  function checkPath() {
    const { start, end } = findStartEnd();
    if (!start || !end) return false;

    // BFS from start to find end through connected pipes
    const visited = new Set();
    const queue = [];
    queue.push(start);
    visited.add(`${start.r},${start.c}`);

    while (queue.length > 0) {
      const current = queue.shift();
      if (current === end) return true;

      // Check neighbors in all directions
      for (let dir = 0; dir < 4; dir++) {
        const nr = current.r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
        const nc = current.c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
        if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) continue;
        const neighbor = blocks[nr][nc];
        if (!neighbor) continue;
        if (neighbor.type === 'empty') continue;
        if (visited.has(`${nr},${nc}`)) continue;
        if (blocksConnect(current, dir, neighbor)) {
          visited.add(`${nr},${nc}`);
          queue.push(neighbor);
        }
      }
    }
    return false;
  }

  // Drag and drop handlers
  function onPointerDown(e) {
    if (e.pointerType !== 'mouse' && e.pointerType !== 'touch') return;
    const target = e.target;
    if (!target) return;
    const cellGroup = target.closest('g.cell');
    if (!cellGroup) return;
    const r = parseInt(cellGroup.getAttribute('data-row'), 10);
    const c = parseInt(cellGroup.getAttribute('data-col'), 10);
    if (isNaN(r) || isNaN(c)) return;
    const block = blocks[r][c];
    if (!block) return;
    if (block.type === 'start' || block.type === 'end' || block.type === 'empty') return; // no drag for these

    e.preventDefault();
    selectedBlock = block;
    dragData = {
      startX: e.clientX,
      startY: e.clientY,
      origR: r,
      origC: c,
      dragging: false,
      pointerId: e.pointerId,
    };
    // Set pointer capture for smooth dragging
    e.target.setPointerCapture(e.pointerId);
  }

  function onPointerMove(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const dx = e.clientX - dragData.startX;
    const dy = e.clientY - dragData.startY;
    if (!dragData.dragging) {
      if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
        dragData.dragging = true;
        selectedBlock.el.classList.add('dragging');
      } else {
        return;
      }
    }
    // Move the block SVG group visually with pointer
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (!blockGroup) return;
    blockGroup.style.transition = 'none';
    blockGroup.style.position = 'relative';
    blockGroup.style.zIndex = 1000;
    blockGroup.style.left = dx + 'px';
    blockGroup.style.top = dy + 'px';
  }

  function onPointerUp(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (blockGroup) {
      blockGroup.style.transition = '';
      blockGroup.style.position = '';
      blockGroup.style.zIndex = '';
      blockGroup.style.left = '';
      blockGroup.style.top = '';
    }
    if (dragData.dragging) {
      // Calculate new cell based on pointer position relative to board
      const boardRect = boardEl.getBoundingClientRect();
      const x = e.clientX - boardRect.left;
      const y = e.clientY - boardRect.top;
      const newC = Math.floor(x / CELL_SIZE);
      const newR = Math.floor(y / CELL_SIZE);
      if (
        newR >= 0 && newR < GRID_SIZE &&
        newC >= 0 && newC < GRID_SIZE &&
        blocks[newR][newC].type === 'empty'
      ) {
        // Move block to new position
        moveBlock(selectedBlock.r, selectedBlock.c, newR, newC);
        announce(`Moved block to row ${newR + 1}, column ${newC + 1}.`);
      } else {
        announce('Invalid move. Block returned to original position.');
      }
    } else {
      // No drag, treat as click: rotate block
      rotateBlock(selectedBlock);
      announce(`Rotated block at row ${selectedBlock.r + 1}, column ${selectedBlock.c + 1}.`);
    }
    dragData = null;
    selectedBlock.el.classList.remove('dragging');
    selectedBlock = null;
    checkWinCondition();
  }

  // Move block from (r1,c1) to (r2,c2)
  function moveBlock(r1, c1, r2, c2) {
    if (r1 === r2 && c1 === c2) return;
    const block = blocks[r1][c1];
    if (!block) return;
    if (blocks[r2][c2].type !== 'empty') return;

    // Swap blocks in data
    blocks[r2][c2] = {
      ...block,
      r: r2,
      c: c2,
    };
    blocks[r1][c1] = {
      type: 'empty',
      rotation: 0,
      r: r1,
      c: c1,
      el: null,
    };

    // Update DOM elements
    // Remove old cell groups
    svgEl.removeChild(blocks[r2][c2].el);
    svgEl.removeChild(blocks[r1][c1].el);

    // Create new cells for both positions
    const newCell1 = createCell(r1, c1, 'empty', 0);
    const newCell2 = createCell(r2, c2, block.type, block.rotation);
    svgEl.appendChild(newCell1);
    svgEl.appendChild(newCell2);

    blocks[r1][c1].el = newCell1;
    blocks[r2][c2].el = newCell2;

    // Update block reference for moved block
    blocks[r2][c2].el = newCell2;
    blocks[r2][c2].r = r2;
    blocks[r2][c2].c = c2;

    // Re-attach event listeners for new cells
    attachCellListeners(newCell1);
    attachCellListeners(newCell2);
  }

  // Attach pointer event listeners to a cell group
  function attachCellListeners(cellGroup) {
    const blockGroup = cellGroup.querySelector('g.block-svg');
    if (!blockGroup) return;
    blockGroup.style.touchAction = 'none';
    blockGroup.addEventListener('pointerdown', onPointerDown);
    blockGroup.addEventListener('pointermove', onPointerMove);
    blockGroup.addEventListener('pointerup', onPointerUp);
    blockGroup.addEventListener('pointercancel', onPointerUp);
    blockGroup.addEventListener('lostpointercapture', onPointerUp);
  }

  // Attach listeners to all blocks
  function attachAllListeners() {
    for (let r = 0; r < GRID_SIZE; r++) {
      for (let c = 0; c < GRID_SIZE; c++) {
        const block = blocks[r][c];
        if (!block || !block.el) continue;
        attachCellListeners(block.el);
      }
    }
  }

  // Check win condition and announce
  function checkWinCondition() {
    if (checkPath()) {

    } else {
      announce('Keep trying to connect Cub to the Star.');
    }
  }

  }

  function blocksConnect(blockA, dir, blockB) {
    if (!blockA || !blockB) return false;
    const typeA = BLOCK_TYPES[blockA.type];
    const typeB = BLOCK_TYPES[blockB.type];
    if (!typeA || !typeB) return false;
    const rotA = blockA.rotation;
    const rotB = blockB.rotation;

    // Calculate actual connection directions for blockA
    const connsA = typeA.connections.map(d => (d + rotA) % 4);
    // Calculate actual connection directions for blockB
    const connsB = typeB.connections.map(d => (d + rotB) % 4);

    // blockA must have connection in dir
    if (!connsA.includes(dir)) return false;
    // blockB must have connection in opposite direction
    if (!connsB.includes(oppositeDir(dir))) return false;

    return true;

    const { start, end } = findStartEnd();
    if (!start || !end) return false;

    // BFS from start to find end through connected pipes
    const visited = new Set();
    const queue = [];
    queue.push(start);
    visited.add(`${start.r},${start.c}`);

    while (queue.length > 0) {
      const current = queue.shift();
      if (current === end) return true;

      // Check neighbors in all directions
      for (let dir = 0; dir < 4; dir++) {
        const nr = current.r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
        const nc = current.c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
        if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) continue;
        const neighbor = blocks[nr][nc];
        if (!neighbor) continue;
        if (neighbor.type === 'empty') continue;
        if (visited.has(`${nr},${nc}`)) continue;
        if (blocksConnect(current, dir, neighbor)) {
          visited.add(`${nr},${nc}`);
          queue.push(neighbor);
        }
      }
    }
    return false;
  }

    if (e.pointerType !== 'mouse' && e.pointerType !== 'touch') return;
    const target = e.target;
    if (!target) return;
    const cellGroup = target.closest('g.cell');
    if (!cellGroup) return;
    const r = parseInt(cellGroup.getAttribute('data-row'), 10);
    const c = parseInt(cellGroup.getAttribute('data-col'), 10);
    if (isNaN(r) || isNaN(c)) return;
    const block = blocks[r][c];
    if (!block) return;
    if (block.type === 'start' || block.type === 'end' || block.type === 'empty') return; // no drag for these

    e.preventDefault();
    selectedBlock = block;
    dragData = {
      startX: e.clientX,
      startY: e.clientY,
      origR: r,
      origC: c,
      dragging: false,
      pointerId: e.pointerId,
    };
    // Set pointer capture for smooth dragging
    e.target.setPointerCapture(e.pointerId);
  }

  function onPointerMove(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const dx = e.clientX - dragData.startX;
    const dy = e.clientY - dragData.startY;
    if (!dragData.dragging) {
      if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
        dragData.dragging = true;
        selectedBlock.el.classList.add('dragging');
      } else {
        return;
      }
    }
    // Move the block SVG group visually with pointer
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (!blockGroup) return;
    blockGroup.style.transition = 'none';
    blockGroup.style.position = 'relative';
    blockGroup.style.zIndex = 1000;
    blockGroup.style.left = dx + 'px';
    blockGroup.style.top = dy + 'px';
  }

  function onPointerUp(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (blockGroup) {
      blockGroup.style.transition = '';
      blockGroup.style.position = '';
      blockGroup.style.zIndex = '';
      blockGroup.style.left = '';
      blockGroup.style.top = '';
    }
    if (dragData.dragging) {
      // Calculate new cell based on pointer position relative to board
      const boardRect = boardEl.getBoundingClientRect();
      const x = e.clientX - boardRect.left;
      const y = e.clientY - boardRect.top;
      const newC = Math.floor(x / CELL_SIZE);
      const newR = Math.floor(y / CELL_SIZE);
      if (
        newR >= 0 && newR < GRID_SIZE &&
        newC >= 0 && newC < GRID_SIZE &&
        blocks[newR][newC].type === 'empty'
      ) {
        // Move block to new position
        pushUndo();
        moveBlock(selectedBlock.r, selectedBlock.c, newR, newC);
        announce(`Moved block to row ${newR + 1}, column ${newC + 1}.`);
        redoStack = [];
        updateUndoRedoButtons();
      } else {
        announce('Invalid move. Block returned to original position.');
      }
    } else {
      // No drag, treat as click: rotate block
      pushUndo();
      rotateBlock(selectedBlock, false);
      announce(`Rotated block at row ${selectedBlock.r + 1}, column ${selectedBlock.c + 1}.`);
      redoStack = [];
      updateUndoRedoButtons();
    }
    dragData = null;
    selectedBlock.el.classList.remove('dragging');
    selectedBlock = null;
    checkWinCondition();
  }

  // Move block from (r1,c1) to (r2,c2)
  function moveBlock(r1, c1, r2, c2) {
    if (r1 === r2 && c1 === c2) return;
    const block = blocks[r1][c1];
    if (!block) return;
    if (blocks[r2][c2].type !== 'empty') return;

    // Swap blocks in data
    blocks[r2][c2] = {
      ...block,
      r: r2,
      c: c2,
    };
    blocks[r1][c1] = {
      type: 'empty',
      rotation: 0,
      r: r1,
      c: c1,
      el: null,
    };

    // Update DOM elements
    // Remove old cell groups
    if (blocks[r2][c2].el) svgEl.removeChild(blocks[r2][c2].el);
    if (blocks[r1][c1].el) svgEl.removeChild(blocks[r1][c1].el);

    // Create new cells for both positions
    const newCell1 = createCell(r1, c1, 'empty', 0);
    const newCell2 = createCell(r2, c2, block.type, block.rotation);
    svgEl.appendChild(newCell1);
    svgEl.appendChild(newCell2);

    blocks[r1][c1].el = newCell1;
    blocks[r2][c2].el = newCell2;

    // Update block reference for moved block
    blocks[r2][c2].r = r2;
    blocks[r2][c2].c = c2;

    // Re-attach event listeners for new cells
    attachCellListeners(newCell1);
    attachCellListeners(newCell2);

  // Undo/Redo functionality

    const snapshot = blocks.map(row => row.map(b => ({
      type: b.type,
      rotation: b.rotation,
      r: b.r,
      c: b.c,

    undoStack.push(snapshot);

  }

    const snapshot = blocks.map(row => row.map(b => ({
      type: b.type,
      rotation: b.rotation,
      r: b.r,
      c: b.c,
    })));
    redoStack.push(snapshot);
    if (redoStack.length > 50) redoStack.shift();

    svgEl.innerHTML = '';

      blocks[r] = [];
      for (let c = 0; c < GRID_SIZE; c++) {

        blocks[r][c] = {

          r: r,

  // Hint system: find shortest path and highlight next block to rotate or move
  function showHint() {
    const { start, end } = findStartEnd();
    if (!start || !end) {
      announce('No start or end block found.');
      return;
    }

    // BFS to find shortest path with current connections
    const visited = new Set();
    const queue = [];
    const parentMap = new Map();
    queue.push(start);
    visited.add(`${start.r},${start.c}`);

    while (queue.length > 0) {
      const current = queue.shift();
      if (current === end) break;

      for (let dir = 0; dir < 4; dir++) {
        const nr = current.r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
        const nc = current.c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
        if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) continue;
        const neighbor = blocks[nr][nc];
        if (!neighbor) continue;
        if (neighbor.type === 'empty') continue;
        if (visited.has(`${nr},${nc}`)) continue;
        if (blocksConnect(current, dir, neighbor)) {
          visited.add(`${nr},${nc}`);
          queue.push(neighbor);
          parentMap.set(neighbor, current);
        }
      }
    }

    if (!visited.has(`${end.r},${end.c}`)) {
      // No path found, suggest a block to rotate or move
      // Find blocks adjacent to path or start that can be rotated or moved to connect
      // For simplicity, highlight first rotatable block not connected properly

      // Clear previous preview
      clearPathPreview();

      // Find first rotatable block that is not connected properly
      for (let r = 0; r < GRID_SIZE; r++) {
        for (let c = 0; c < GRID_SIZE; c++) {
          const block = blocks[r][c];
          if (!block || !BLOCK_TYPES[block.type].rotatable) continue;
          // Check if block has any connection that does not connect to neighbor
          const conns = BLOCK_TYPES[block.type].connections.map(d => (d + block.rotation) % 4);
          let needsRotation = false;
          for (const dir of conns) {
            const nr = r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
            const nc = c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
            if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) {
              needsRotation = true;
              break;
            }
            const neighbor = blocks[nr][nc];
            if (!neighbor || neighbor.type === 'empty') {
              needsRotation = true;
              break;
            }
            if (!blocksConnect(block, dir, neighbor)) {
              needsRotation = true;
              break;
            }
          }
          if (needsRotation) {
            highlightBlock(block);
            announce(`Hint: Try rotating the block at row ${r + 1}, column ${c + 1}.`);
            return;
          }
        }
      }
      announce('No hints available. Try different moves.');
      return;
    }

    // Path found, highlight the path
    const path = [];
    let current = end;
    while (current !== start) {
      path.push(current);
      current = parentMap.get(current);
      if (!current) break;
    }
    path.push(start);
    path.reverse();

    clearPathPreview();
    for (const block of path) {
      highlightBlock(block);
    }
    announce('Hint: Follow the highlighted path.');
  }

  // Initialize the board for a level
  function initLevel(levelIndex) {
    currentLevelIndex = levelIndex;
    const level = LEVELS[levelIndex];
    if (!level) return;

    boardEl.innerHTML = '';

    svgEl.setAttribute('height', BOARD_SIZE);
    svgEl.setAttribute('role', 'presentation');
    svgEl.style.touchAction = 'none';
    boardEl.appendChild(svgEl);

    const rotations = expandRotations(level);

      blocks[r] = [];
      for (let c = 0; c < GRID_SIZE; c++) {

        const rot = rotations[r][c] || 0;
        blocks[r][c] = {
          type,
          rotation: rot % 4,
          r,
          c,
          el: null,
        };

        svgEl.appendChild(cellGroup);
        blocks[r][c].el = cellGroup;
      }

    undoStack = [];
    redoStack = [];

  // Rotate a block clockwise by 90 degrees
  function rotateBlock(block, record = true) {
    if (!block || !BLOCK_TYPES[block.type].rotatable) return;
    if (record) pushUndo();
    block.rotation = (block.rotation + 1) % 4;
    updateBlockVisual(block);
  }

  // Update block SVG rotation transform
  function updateBlockVisual(block) {
    if (!block.el) return;
    const blockGroup = block.el.querySelector('g.block-svg');
    if (!blockGroup) return;
    const x = block.c * CELL_SIZE;
    const y = block.r * CELL_SIZE;
    blockGroup.setAttribute('transform', `translate(${x},${y}) rotate(${block.rotation * ROTATION_ANGLE} ${CELL_SIZE/2} ${CELL_SIZE/2})`);
  }

  // Find start and end positions
  function findStartEnd() {
    let start = null;
    let end = null;
    for (let r = 0; r < GRID_SIZE; r++) {
      for (let c = 0; c < GRID_SIZE; c++) {
        if (blocks[r][c].type === 'start') start = blocks[r][c];
        if (blocks[r][c].type === 'end') end = blocks[r][c];
      }
    }
    return { start, end };
  }

  // Check if two blocks connect in a given direction
  function blocksConnect(blockA, dir, blockB) {
    if (!blockA || !blockB) return false;
    const typeA = BLOCK_TYPES[blockA.type];
    const typeB = BLOCK_TYPES[blockB.type];
    if (!typeA || !typeB) return false;
    const rotA = blockA.rotation;
    const rotB = blockB.rotation;

    // Calculate actual connection directions for blockA
    const connsA = typeA.connections.map(d => (d + rotA) % 4);
    // Calculate actual connection directions for blockB
    const connsB = typeB.connections.map(d => (d + rotB) % 4);

    // blockA must have connection in dir
    if (!connsA.includes(dir)) return false;
    // blockB must have connection in opposite direction
    if (!connsB.includes(oppositeDir(dir))) return false;

    return true;
  }

  // Check if the path from start to end is connected
  function checkPath() {
    const { start, end } = findStartEnd();
    if (!start || !end) return false;

    // BFS from start to find end through connected pipes
    const visited = new Set();
    const queue = [];
    queue.push(start);
    visited.add(`${start.r},${start.c}`);

    while (queue.length > 0) {
      const current = queue.shift();
      if (current === end) return true;

      // Check neighbors in all directions
      for (let dir = 0; dir < 4; dir++) {
        const nr = current.r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
        const nc = current.c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
        if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) continue;
        const neighbor = blocks[nr][nc];
        if (!neighbor) continue;
        if (neighbor.type === 'empty') continue;
        if (visited.has(`${nr},${nc}`)) continue;
        if (blocksConnect(current, dir, neighbor)) {
          visited.add(`${nr},${nc}`);
          queue.push(neighbor);
        }
      }
    }
    return false;
  }

  // Drag and drop handlers
  function onPointerDown(e) {
    if (e.pointerType !== 'mouse' && e.pointerType !== 'touch') return;
    const target = e.target;
    if (!target) return;
    const cellGroup = target.closest('g.cell');
    if (!cellGroup) return;
    const r = parseInt(cellGroup.getAttribute('data-row'), 10);
    const c = parseInt(cellGroup.getAttribute('data-col'), 10);
    if (isNaN(r) || isNaN(c)) return;
    const block = blocks[r][c];
    if (!block) return;
    if (block.type === 'start' || block.type === 'end' || block.type === 'empty') return; // no drag for these

    e.preventDefault();
    selectedBlock = block;
    dragData = {
      startX: e.clientX,
      startY: e.clientY,
      origR: r,
      origC: c,
      dragging: false,
      pointerId: e.pointerId,
    };
    // Set pointer capture for smooth dragging
    e.target.setPointerCapture(e.pointerId);
  }

  function onPointerMove(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const dx = e.clientX - dragData.startX;
    const dy = e.clientY - dragData.startY;
    if (!dragData.dragging) {
      if (Math.abs(dx) > 5 || Math.abs(dy) > 5) {
        dragData.dragging = true;
        selectedBlock.el.classList.add('dragging');
      } else {
        return;
      }
    }
    // Move the block SVG group visually with pointer
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (!blockGroup) return;
    blockGroup.style.transition = 'none';
    blockGroup.style.position = 'relative';
    blockGroup.style.zIndex = 1000;
    blockGroup.style.left = dx + 'px';
    blockGroup.style.top = dy + 'px';
  }

  function onPointerUp(e) {
    if (!dragData || e.pointerId !== dragData.pointerId) return;
    e.preventDefault();
    const blockGroup = selectedBlock.el.querySelector('g.block-svg');
    if (blockGroup) {
      blockGroup.style.transition = '';
      blockGroup.style.position = '';
      blockGroup.style.zIndex = '';
      blockGroup.style.left = '';
      blockGroup.style.top = '';
    }
    if (dragData.dragging) {
      // Calculate new cell based on pointer position relative to board
      const boardRect = boardEl.getBoundingClientRect();
      const x = e.clientX - boardRect.left;
      const y = e.clientY - boardRect.top;
      const newC = Math.floor(x / CELL_SIZE);
      const newR = Math.floor(y / CELL_SIZE);
      if (
        newR >= 0 && newR < GRID_SIZE &&
        newC >= 0 && newC < GRID_SIZE &&
        blocks[newR][newC].type === 'empty'
      ) {
        // Move block to new position
        pushUndo();
        moveBlock(selectedBlock.r, selectedBlock.c, newR, newC);
        announce(`Moved block to row ${newR + 1}, column ${newC + 1}.`);
        redoStack = [];
        updateUndoRedoButtons();
      } else {
        announce('Invalid move. Block returned to original position.');
      }
    } else {
      // No drag, treat as click: rotate block
      pushUndo();
      rotateBlock(selectedBlock, false);
      announce(`Rotated block at row ${selectedBlock.r + 1}, column ${selectedBlock.c + 1}.`);
      redoStack = [];
      updateUndoRedoButtons();
    }
    dragData = null;
    selectedBlock.el.classList.remove('dragging');
    selectedBlock = null;
    checkWinCondition();
  }

  // Move block from (r1,c1) to (r2,c2)
  function moveBlock(r1, c1, r2, c2) {
    if (r1 === r2 && c1 === c2) return;
    const block = blocks[r1][c1];
    if (!block) return;
    if (blocks[r2][c2].type !== 'empty') return;

    // Swap blocks in data
    blocks[r2][c2] = {
      ...block,
      r: r2,
      c: c2,
    };
    blocks[r1][c1] = {
      type: 'empty',
      rotation: 0,
      r: r1,
      c: c1,
      el: null,
    };

    // Update DOM elements
    // Remove old cell groups
    if (blocks[r2][c2].el) svgEl.removeChild(blocks[r2][c2].el);
    if (blocks[r1][c1].el) svgEl.removeChild(blocks[r1][c1].el);

    // Create new cells for both positions
    const newCell1 = createCell(r1, c1, 'empty', 0);
    const newCell2 = createCell(r2, c2, block.type, block.rotation);
    svgEl.appendChild(newCell1);
    svgEl.appendChild(newCell2);

    blocks[r1][c1].el = newCell1;
    blocks[r2][c2].el = newCell2;

    // Update block reference for moved block
    blocks[r2][c2].r = r2;
    blocks[r2][c2].c = c2;

    // Re-attach event listeners for new cells
    attachCellListeners(newCell1);
    attachCellListeners(newCell2);
  }

  // Attach pointer event listeners to a cell group
  function attachCellListeners(cellGroup) {
    const blockGroup = cellGroup.querySelector('g.block-svg');
    if (!blockGroup) return;
    blockGroup.style.touchAction = 'none';
    blockGroup.addEventListener('pointerdown', onPointerDown);
    blockGroup.addEventListener('pointermove', onPointerMove);
    blockGroup.addEventListener('pointerup', onPointerUp);
    blockGroup.addEventListener('pointercancel', onPointerUp);
    blockGroup.addEventListener('lostpointercapture', onPointerUp);
    // Add hover listeners for path preview
    cellGroup.addEventListener('mouseenter', onCellMouseEnter);
    cellGroup.addEventListener('mouseleave', onCellMouseLeave);
  }

  // Attach listeners to all blocks
  function attachAllListeners() {
    for (let r = 0; r < GRID_SIZE; r++) {
      for (let c = 0; c < GRID_SIZE; c++) {
        const block = blocks[r][c];
        if (!block || !block.el) continue;
        attachCellListeners(block.el);
      }
    }
  }

  // Check win condition and announce
  function checkWinCondition() {
    if (checkPath()) {
      announce('Congratulations! Cub reached the Star! 🎉');
    } else {
      announce('Keep trying to connect Cub to the Star.');
    }
  }

  // Rotate selected block button handler
  function onRotateButton() {
    if (!selectedBlock) {
      announce('Select a block by clicking or dragging it first.');
      return;
    }
    pushUndo();
    rotateBlock(selectedBlock, false);
    announce(`Rotated block at row ${selectedBlock.r + 1}, column ${selectedBlock.c + 1}.`);
    redoStack = [];
    updateUndoRedoButtons();
    checkWinCondition();
  }

  // Select block by click on cell
  function onCellClick(e) {
    if (dragData) return; // ignore if dragging
    const cellGroup = e.currentTarget;
    const r = parseInt(cellGroup.getAttribute('data-row'), 10);
    const c = parseInt(cellGroup.getAttribute('data-col'), 10);
    if (isNaN(r) || isNaN(c)) return;
    const block = blocks[r][c];
    if (!block) return;
    if (block.type === 'start' || block.type === 'end' || block.type === 'empty') {
      announce('This block cannot be rotated or moved.');
      return;
    }
    selectedBlock = block;
    announce(`Selected block at row ${r + 1}, column ${c + 1}. Use rotate button or drag to move.`);
  }

  // Undo/Redo functionality
  function pushUndo() {
    const snapshot = blocks.map(row => row.map(b => ({
      type: b.type,
      rotation: b.rotation,
      r: b.r,
      c: b.c,
    })));
    undoStack.push(snapshot);
    if (undoStack.length > 50) undoStack.shift(); // limit stack size
  }

  function pushRedo() {
    const snapshot = blocks.map(row => row.map(b => ({
      type: b.type,
      rotation: b.rotation,
      r: b.r,
      c: b.c,
    })));
    redoStack.push(snapshot);
    if (redoStack.length > 50) redoStack.shift();
  }

  function restoreSnapshot(snapshot) {
    if (!snapshot) return;
    blocks = [];
    svgEl.innerHTML = '';
    for (let r = 0; r < GRID_SIZE; r++) {
      blocks[r] = [];
      for (let c = 0; c < GRID_SIZE; c++) {
        const b = snapshot[r][c];
        blocks[r][c] = {
          type: b.type,
          rotation: b.rotation,
          r: r,
          c: c,
          el: null,
        };
        const cellGroup = createCell(r, c, b.type, b.rotation);
        svgEl.appendChild(cellGroup);
        blocks[r][c].el = cellGroup;
      }
    }
    attachAllListeners();
  }

  function undo() {
    if (undoStack.length === 0) {
      announce('Nothing to undo.');
      return;
    }
    pushRedo();
    const snapshot = undoStack.pop();
    restoreSnapshot(snapshot);
    announce('Undo performed.');
    updateUndoRedoButtons();
    checkWinCondition();
  }

  function redo() {
    if (redoStack.length === 0) {
      announce('Nothing to redo.');
      return;
    }
    pushUndo();
    const snapshot = redoStack.pop();
    restoreSnapshot(snapshot);
    announce('Redo performed.');
    updateUndoRedoButtons();
    checkWinCondition();
  }

  function updateUndoRedoButtons() {
    const btnUndo = document.getElementById('btn-undo');
    const btnRedo = document.getElementById('btn-redo');
    btnUndo.disabled = undoStack.length === 0;
    btnRedo.disabled = redoStack.length === 0;
  }

  // Hint system: find shortest path and highlight next block to rotate or move
  function showHint() {
    const { start, end } = findStartEnd();
    if (!start || !end) {
      announce('No start or end block found.');
      return;
    }

    // BFS to find shortest path with current connections
    const visited = new Set();
    const queue = [];
    const parentMap = new Map();
    queue.push(start);
    visited.add(`${start.r},${start.c}`);

    while (queue.length > 0) {
      const current = queue.shift();
      if (current === end) break;

      for (let dir = 0; dir < 4; dir++) {
        const nr = current.r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
        const nc = current.c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
        if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) continue;
        const neighbor = blocks[nr][nc];
        if (!neighbor) continue;
        if (neighbor.type === 'empty') continue;
        if (visited.has(`${nr},${nc}`)) continue;
        if (blocksConnect(current, dir, neighbor)) {
          visited.add(`${nr},${nc}`);
          queue.push(neighbor);
          parentMap.set(neighbor, current);
        }
      }
    }

    if (!visited.has(`${end.r},${end.c}`)) {
      // No path found, suggest a block to rotate or move
      // Find blocks adjacent to path or start that can be rotated or moved to connect
      // For simplicity, highlight first rotatable block not connected properly

      // Clear previous preview
      clearPathPreview();

      // Find first rotatable block that is not connected properly
      for (let r = 0; r < GRID_SIZE; r++) {
        for (let c = 0; c < GRID_SIZE; c++) {
          const block = blocks[r][c];
          if (!block || !BLOCK_TYPES[block.type].rotatable) continue;
          // Check if block has any connection that does not connect to neighbor
          const conns = BLOCK_TYPES[block.type].connections.map(d => (d + block.rotation) % 4);
          let needsRotation = false;
          for (const dir of conns) {
            const nr = r + (dir === DIRS.DOWN ? 1 : dir === DIRS.UP ? -1 : 0);
            const nc = c + (dir === DIRS.RIGHT ? 1 : dir === DIRS.LEFT ? -1 : 0);
            if (nr < 0 || nr >= GRID_SIZE || nc < 0 || nc >= GRID_SIZE) {
              needsRotation = true;
              break;
            }
            const neighbor = blocks[nr][nc];
            if (!neighbor || neighbor.type === 'empty') {
              needsRotation = true;
              break;
            }
            if (!blocksConnect(block, dir, neighbor)) {
              needsRotation = true;
              break;
            }
          }
          if (needsRotation) {
            highlightBlock(block);
            announce(`Hint: Try rotating the block at row ${r + 1}, column ${c + 1}.`);
            return;
          }
        }
      }
      announce('No hints available. Try different moves.');
      return;
    }

    // Path found, highlight the path
    const path = [];
    let current = end;
    while (current !== start) {
      path.push(current);
      current = parentMap.get(current);
      if (!current) break;
    }
    path.push(start);
    path.reverse();

    clearPathPreview();
    for (const block of path) {
      highlightBlock(block);
    }
    announce('Hint: Follow the highlighted path.');
  }

  // Highlight a block for hint or preview
  function highlightBlock(block) {
    if (!block || !block.el) return;
    block.el.classList.add('path-preview');
  }

  // Clear all path previews
  function clearPathPreview() {
    for (let r = 0; r < GRID_SIZE; r++) {
      for (let c = 0; c < GRID_SIZE; c++) {
        const block = blocks[r][c];
        if (block && block.el) {
          block.el.classList.remove('path-preview');
        }
      }
    }
  }

  // Path preview on hover: show path if block is rotated or moved
  function onCellMouseEnter(e) {
    if (dragData) return; // don't show preview while dragging
    const cellGroup = e.currentTarget;
    const r = parseInt(cellGroup.getAttribute('data-row'), 10);
    const c = parseInt(cellGroup.getAttribute('data-col'), 10);
    if (isNaN(r) || isNaN(c)) return;
    const block = blocks[r][c];
    if (!block) return;
    if (block.type === 'empty' || block.type === 'start' || block.type === 'end') return;

    // Temporarily simulate rotation and show path preview
    clearPathPreview();

    // Save current rotation
    const originalRotation = block.rotation;
    // Simulate rotation
    block.rotation = (block.rotation + 1) % 4;
    updateBlockVisual(block);

    // Check path with simulated rotation
    const pathExists = checkPath();

    if (pathExists) {
      // Highlight path
      showHint();
    } else {
      // No path, just highlight this block
      highlightBlock(block);
    }

    // Revert rotation
    block.rotation = originalRotation;
    updateBlockVisual(block);
  }

  function onCellMouseLeave(e) {
    clearPathPreview();
  }

  // Setup controls buttons
  function setupControls() {
    const btnPrev = document.getElementById('btn-prev');
    const btnNext = document.getElementById('btn-next');
    const btnRotate = document.getElementById('btn-rotate');
    const btnReset = document.getElementById('btn-reset');
    const btnUndo = document.getElementById('btn-undo');
    const btnRedo = document.getElementById('btn-redo');
    const btnHint = document.getElementById('btn-hint');

    btnPrev.addEventListener('click', () => {
      currentLevelIndex = (currentLevelIndex - 1 + LEVELS.length) % LEVELS.length;
      initLevel(currentLevelIndex);
      attachAllListeners();
    });
    btnNext.addEventListener('click', () => {
      currentLevelIndex = (currentLevelIndex + 1) % LEVELS.length;
      initLevel(currentLevelIndex);
      attachAllListeners();
    });
    btnRotate.addEventListener('click', () => {
      if (!selectedBlock) {
        announce('Select a block first by clicking or dragging it.');
        return;
      }
      pushUndo();
      rotateBlock(selectedBlock, false);
      announce(`Rotated block at row ${selectedBlock.r + 1}, column ${selectedBlock.c + 1}.`);
      redoStack = [];
      updateUndoRedoButtons();
      checkWinCondition();
    });
    btnReset.addEventListener('click', () => {
      initLevel(currentLevelIndex);
      attachAllListeners();
    });
    btnUndo.addEventListener('click', () => {
      undo();
    });
    btnRedo.addEventListener('click', () => {
      redo();
    });
    btnHint.addEventListener('click', () => {
      clearPathPreview();
      showHint();
    });
  }

  // Tooltip for rotate button
  function setupTooltip() {
    tooltipEl = document.getElementById('tooltip');
    const btnRotate = document.getElementById('btn-rotate');
    btnRotate.addEventListener('mouseenter', (e) => {
      tooltipEl.textContent = 'Rotate the selected block by 90° clockwise';
      const rect = btnRotate.getBoundingClientRect();
      tooltipEl.style.left = rect.left + rect.width / 2 + 'px';
      tooltipEl.style.top = rect.top - 30 + 'px';
      tooltipEl.style.opacity = '1';
      tooltipEl.setAttribute('aria-hidden', 'false');
    });
    btnRotate.addEventListener('mouseleave', () => {
      tooltipEl.style.opacity = '0';
      tooltipEl.setAttribute('aria-hidden', 'true');
    });
  }

  // Handle window resize to clear any drag or preview states
  function onWindowResize() {
    if (dragData && selectedBlock) {
      const blockGroup = selectedBlock.el.querySelector('g.block-svg');
      if (blockGroup) {
        blockGroup.style.transition = '';
        blockGroup.style.position = '';
        blockGroup.style.zIndex = '';
        blockGroup.style.left = '';
        blockGroup.style.top = '';
      }
      dragData = null;
      if (selectedBlock) selectedBlock.el.classList.remove('dragging');
      selectedBlock = null;
      announce('Window resized. Drag operation cancelled.');
    }
    clearPathPreview();
  }

  // Initialize game
  function init() {
    boardEl = document.getElementById('board');
    messageEl = document.getElementById('message');
    levelInfoEl = document.getElementById('level-info');
    setupControls();
    setupTooltip();
    initLevel(currentLevelIndex);
    attachAllListeners();
    window.addEventListener('resize', onWindowResize);
  }

  // Start game on DOM ready
  window.addEventListener('DOMContentLoaded', init);
})();
