    block.rotation = (block.rotation + 1) % 4;
    renderBlocks();
    checkWinCondition();
  }

  // Convert position object to key string
  function posKey(pos) {
    return `${pos.x},${pos.y}`;
  }

  // Check if a position is inside the grid
  function isInsideGrid(pos) {
    return pos.x >= 0 && pos.x < gridSize && pos.y >= 0 && pos.y < gridSize;
  }

  // Find block at position
  function findBlockAt(pos) {
    const key = posKey(pos);
    const idx = positionMap.get(key);
    if (idx === undefined) return null;
    return blocks[idx];
  }

  // Update position map from blocks array
  function updatePositionMap() {
    positionMap.clear();
    blocks.forEach((b, i) => {
      positionMap.set(posKey(b.position), i);
    });
  }

  // Drag and drop handlers
  function onPointerDown(e) {
    if (e.button !== 0) return; // left click only
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    draggingBlock = block;
    el.classList.add('dragging');
    el.setAttribute('aria-grabbed', 'true');

    const rect = el.getBoundingClientRect();
    dragOffset.x = e.clientX - rect.left;
    dragOffset.y = e.clientY - rect.top;

    window.addEventListener('pointermove', onPointerMove);
    window.addEventListener('pointerup', onPointerUp);
    window.addEventListener('pointercancel', onPointerUp);

    e.preventDefault();
  }

  function onPointerMove(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    // Calculate new position relative to grid container
    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40; // center offset
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    // Snap to grid cells
    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    // Clamp inside grid
    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, do not move there, but still show dragging
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
      return;
    }

    // Move block visually to pointer position (free drag)
    el.style.left = `${e.clientX - gridRect.left - dragOffset.x}px`;
    el.style.top = `${e.clientY - gridRect.top - dragOffset.y}px`;
  }

  function onPointerUp(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40;
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {

    window.removeEventListener('pointerup', onPointerUp);
    window.removeEventListener('pointercancel', onPointerUp);

    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    switch (e.key) {

      case ' ':

        break;
      case 'ArrowUp':
      case 'ArrowDown':
      case 'ArrowLeft':
      case 'ArrowRight':
        e.preventDefault();
        moveBlockWithKeyboard(block, e.key);
        break;
    }

  // Move block with keyboard arrow keys if possible
  function moveBlockWithKeyboard(block, arrowKey) {
    const dirMap = {
      ArrowUp: { x: 0, y: -1 },
      ArrowDown: { x: 0, y: 1 },
      ArrowLeft: { x: -1, y: 0 },
      ArrowRight: { x: 1, y: 0 },
    };
    const delta = dirMap[arrowKey];
    if (!delta) return;

    const newPos = { x: block.position.x + delta.x, y: block.position.y + delta.y };
    if (!isInsideGrid(newPos)) return;

    const key = posKey(newPos);
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined) return; // occupied

    block.position = newPos;
    updatePositionMap();
    renderBlocks();
    checkWinCondition();

    // Focus moved block
    setTimeout(() => {
      const newEl = gridElement.querySelector(`.block[data-id="${block.id}"]`);
      if (newEl) newEl.focus();
    }, 0);
  }

  // Check if the path from start to end is connected properly
  // We simulate the path traversal from start block's exit direction
  // and follow connected blocks until we reach the end block or dead end
  function checkWinCondition() {
    const level = LEVELS[currentLevelIndex];
    const startBlock = blocks.find(b => b.type === BLOCK_TYPES.START);
    const endBlock = blocks.find(b => b.type === BLOCK_TYPES.END);
    if (!startBlock || !endBlock) return;

    // Get start block connections with rotation applied
    const startConns = rotateDirections(BLOCK_CONNECTIONS[startBlock.type], startBlock.rotation);
    if (startConns.length !== 1) return; // start block must have exactly one connection
    const startDir = startConns[0];

    // Traverse path
    let currentPos = { x: startBlock.position.x, y: startBlock.position.y };
    let currentDir = startDir;
    let visited = new Set();
    visited.add(posKey(currentPos));

    while (true) {
      // Move to next cell in currentDir
      const vec = DIR_VECTORS[currentDir];
      const nextPos = { x: currentPos.x + vec.x, y: currentPos.y + vec.y };
      if (!isInsideGrid(nextPos)) break;

      const nextBlock = findBlockAt(nextPos);
      if (!nextBlock) break;

      // Get next block connections with rotation
      const nextConns = rotateDirections(BLOCK_CONNECTIONS[nextBlock.type], nextBlock.rotation);

      // Check if next block has connection opposite to currentDir
      const oppositeDir = (currentDir + 2) % 4;
      if (!nextConns.includes(oppositeDir)) break;

      // If next block is end block, check if it has connection oppositeDir only
      if (nextBlock.type === BLOCK_TYPES.END) {
        if (nextConns.length === 1 && nextConns[0] === oppositeDir) {
          // Win condition met
          messageElement.textContent = '🎉 Cub reached the Star! Level Complete!';
          return true;
        } else {
          break;
        }
      }

      // Find the other connection direction in next block (not oppositeDir)
      const otherDirs = nextConns.filter(d => d !== oppositeDir);
      if (otherDirs.length !== 1) break; // must have exactly one other connection to continue

      currentPos = nextPos;
      currentDir = otherDirs[0];

      const key = posKey(currentPos);
      if (visited.has(key)) break; // loop detected, break
      visited.add(key);
    }

    // If we reach here, no win
    messageElement.textContent = '';
    return false;
  }

  // Initialize game on DOM ready
  window.addEventListener('load', init);
})();

  // Convert position object to key string

  // Check if a position is inside the grid

    return pos.x >= 0 && pos.x < gridSize && pos.y >= 0 && pos.y < gridSize;
  }

  // Find block at position

    const key = posKey(pos);
    const idx = positionMap.get(key);
    if (idx === undefined) return null;
    return blocks[idx];
  }

  // Update position map from blocks array

    positionMap.clear();
    blocks.forEach((b, i) => {
      positionMap.set(posKey(b.position), i);

  // Drag and drop handlers
  function onPointerDown(e) {
    if (e.button !== 0) return; // left click only
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    draggingBlock = block;
    el.classList.add('dragging');
    el.setAttribute('aria-grabbed', 'true');

    const rect = el.getBoundingClientRect();
    dragOffset.x = e.clientX - rect.left;
    dragOffset.y = e.clientY - rect.top;

    window.addEventListener('pointermove', onPointerMove);
    window.addEventListener('pointerup', onPointerUp);
    window.addEventListener('pointercancel', onPointerUp);

    e.preventDefault();
  }

  function onPointerMove(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    // Calculate new position relative to grid container
    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40; // center offset
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    // Snap to grid cells
    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    // Clamp inside grid
    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, do not move there, but still show dragging
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
      return;
    }

    // Move block visually to pointer position (free drag)
    el.style.left = `${e.clientX - gridRect.left - dragOffset.x}px`;
    el.style.top = `${e.clientY - gridRect.top - dragOffset.y}px`;
  }

  function onPointerUp(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40;
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, revert to original position
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
    } else {
      // Update block position
      draggingBlock.position.x = gridX;
      draggingBlock.position.y = gridY;
      updatePositionMap();
      el.style.left = `${gridX * 86}px`;
      el.style.top = `${gridY * 86}px`;
      checkWinCondition();
    }

    el.classList.remove('dragging');
    el.setAttribute('aria-grabbed', 'false');
    draggingBlock = null;

    window.removeEventListener('pointermove', onPointerMove);
    window.removeEventListener('pointerup', onPointerUp);
    window.removeEventListener('pointercancel', onPointerUp);
  }

  // Keyboard accessibility: rotate block on Enter or Space, move with arrow keys
  function onBlockKeyDown(e) {
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    switch (e.key) {
      case 'Enter':
      case ' ':
        e.preventDefault();
        rotateBlock(id);
        break;
      case 'ArrowUp':
      case 'ArrowDown':
      case 'ArrowLeft':
      case 'ArrowRight':
        e.preventDefault();
        moveBlockWithKeyboard(block, e.key);
        break;
    }
  }

  // Move block with keyboard arrow keys if possible
  function moveBlockWithKeyboard(block, arrowKey) {
    const dirMap = {
      ArrowUp: { x: 0, y: -1 },
      ArrowDown: { x: 0, y: 1 },
      ArrowLeft: { x: -1, y: 0 },
      ArrowRight: { x: 1, y: 0 },
    };
    const delta = dirMap[arrowKey];
    if (!delta) return;

    const newPos = { x: block.position.x + delta.x, y: block.position.y + delta.y };
    if (!isInsideGrid(newPos)) return;

    const key = posKey(newPos);
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined) return; // occupied

    block.position = newPos;
    updatePositionMap();
    renderBlocks();
    checkWinCondition();

    // Focus moved block
    setTimeout(() => {
      const newEl = gridElement.querySelector(`.block[data-id="${block.id}"]`);
      if (newEl) newEl.focus();
    }, 0);
  }

  // Check if the path from start to end is connected properly
  // We simulate the path traversal from start block's exit direction
  // and follow connected blocks until we reach the end block or dead end
  function checkWinCondition() {
    const level = LEVELS[currentLevelIndex];
    const startBlock = blocks.find(b => b.type === BLOCK_TYPES.START);
    const endBlock = blocks.find(b => b.type === BLOCK_TYPES.END);
    if (!startBlock || !endBlock) return;

    // Get start block connections with rotation applied
    const startConns = rotateDirections(BLOCK_CONNECTIONS[startBlock.type], startBlock.rotation);
    if (startConns.length !== 1) return; // start block must have exactly one connection
    const startDir = startConns[0];

    // Traverse path
    let currentPos = { x: startBlock.position.x, y: startBlock.position.y };
    let currentDir = startDir;
    let visited = new Set();
    visited.add(posKey(currentPos));

    while (true) {
      // Move to next cell in currentDir
      const vec = DIR_VECTORS[currentDir];
      const nextPos = { x: currentPos.x + vec.x, y: currentPos.y + vec.y };
      if (!isInsideGrid(nextPos)) break;

      const nextBlock = findBlockAt(nextPos);
      if (!nextBlock) break;

      // Get next block connections with rotation
      const nextConns = rotateDirections(BLOCK_CONNECTIONS[nextBlock.type], nextBlock.rotation);

      // Check if next block has connection opposite to currentDir
      const oppositeDir = (currentDir + 2) % 4;
      if (!nextConns.includes(oppositeDir)) break;

      // If next block is end block, check if it has connection oppositeDir only
      if (nextBlock.type === BLOCK_TYPES.END) {
        if (nextConns.length === 1 && nextConns[0] === oppositeDir) {
          // Win condition met
          messageElement.textContent = '🎉 Cub reached the Star! Level Complete!';
          return true;
        } else {
          break;
        }
      }

      // Find the other connection direction in next block (not oppositeDir)
      const otherDirs = nextConns.filter(d => d !== oppositeDir);
      if (otherDirs.length !== 1) break; // must have exactly one other connection to continue

      currentPos = nextPos;
      currentDir = otherDirs[0];

      const key = posKey(currentPos);
      if (visited.has(key)) break; // loop detected, break
      visited.add(key);
    }

    // If we reach here, no win
    messageElement.textContent = '';
    return false;
  }

  // Initialize game on DOM ready
  window.addEventListener('load', init);
})();

    btnUndo = document.getElementById('btn-undo');
    btnRedo = document.getElementById('btn-redo');
    btnHint = document.getElementById('btn-hint');

    btnPrev.addEventListener('click', () => {
      if (currentLevelIndex > 0) {
        currentLevelIndex--;
        loadLevel(currentLevelIndex);
      }
    });
    btnNext.addEventListener('click', () => {
      if (currentLevelIndex < LEVELS.length - 1) {
        currentLevelIndex++;
        loadLevel(currentLevelIndex);
      }
    });
    btnReset.addEventListener('click', () => {
      loadLevel(currentLevelIndex);
    });
    btnUndo.addEventListener('click', () => {
      undo();
    });
    btnRedo.addEventListener('click', () => {
      redo();
    });
    btnHint.addEventListener('click', () => {
      showHint();
    });

    loadLevel(currentLevelIndex);
  }

  // Load a level by index
  function loadLevel(index) {
    const level = LEVELS[index];
    gridSize = level.gridSize;
    blocks = level.blocks.map(b => ({
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: !!b.fixed,
      id: generateId(),
    }));
    positionMap.clear();
    blocks.forEach((b, i) => {
      positionMap.set(posKey(b.position), i);
    });
    undoStack = [];
    redoStack = [];
    updateUndoRedoButtons();
    renderGrid();
    renderBlocks();
    updateLevelInfo(level);
    updateControls();
    messageElement.textContent = '';
    clearPathPreview();
  }

  // Generate unique ID for blocks
  function generateId() {
    return 'b' + Math.random().toString(36).slice(2, 10);
  }

  // Update level info text
  function updateLevelInfo(level) {
    levelInfoElement.textContent = `Level ${currentLevelIndex + 1} / ${LEVELS.length} - ${level.description}`;
  }

  // Update controls button states
  function updateControls() {
    btnPrev.disabled = currentLevelIndex === 0;
    btnNext.disabled = currentLevelIndex === LEVELS.length - 1;
  }

  // Render the grid cells
  function renderGrid() {
    gridElement.innerHTML = '';
    gridElement.style.gridTemplateColumns = `repeat(${gridSize}, 80px)`;
    gridElement.style.gridTemplateRows = `repeat(${gridSize}, 80px)`;
    for (let y = 0; y < gridSize; y++) {
      for (let x = 0; x < gridSize; x++) {
        const cell = document.createElement('div');
        cell.className = 'cell';
        cell.dataset.x = x;
        cell.dataset.y = y;
        cell.setAttribute('role', 'gridcell');
        cell.setAttribute('aria-label', `Grid cell ${x + 1},${y + 1}`);
        gridElement.appendChild(cell);
      }
    }
  }

  // Render all blocks on the grid
  function renderBlocks() {
    // Remove existing blocks
    const existingBlocks = gridElement.querySelectorAll('.block');
    existingBlocks.forEach(b => b.remove());

    blocks.forEach(block => {
      const blockEl = createBlockElement(block);
      gridElement.appendChild(blockEl);
    });
  }

  // Create a block DOM element with SVG and rotate button
  function createBlockElement(block) {
    const el = document.createElement('div');
    el.className = 'block';
    el.style.width = '80px';
    el.style.height = '80px';
    el.style.left = `${block.position.x * 86}px`;
    el.style.top = `${block.position.y * 86}px`;
    el.style.position = 'absolute';
    el.style.touchAction = 'none';
    el.dataset.id = block.id;
    el.dataset.type = block.type;
    el.dataset.fixed = block.fixed ? 'true' : 'false';
    el.setAttribute('aria-grabbed', 'false');
    el.setAttribute('role', 'button');
    el.setAttribute('tabindex', block.fixed ? '-1' : '0');
    el.setAttribute('aria-label', `Block type ${block.type}, rotation ${block.rotation * 90} degrees${block.fixed ? ', fixed' : ''}`);

    // Create SVG container
    const svg = createSVGElement('svg', { viewBox: '0 0 80 80', 'aria-hidden': 'true', focusable: 'false' });
    const group = createSVGElement('g');
    group.setAttribute('transform', `rotate(${block.rotation * 90} 40 40)`);

    // Add pipe or icon paths depending on block type
    switch (block.type) {
      case BLOCK_TYPES.EMPTY:
        // no paths
        break;
      case BLOCK_TYPES.STRAIGHT:
        group.appendChild(createStraightPath());
        break;
      case BLOCK_TYPES.CURVE:
        group.appendChild(createCurvePath());
        break;
      case BLOCK_TYPES.T:
        group.appendChild(createTPath());
        break;
      case BLOCK_TYPES.CROSS:
        group.appendChild(createCrossPath());
        break;
      case BLOCK_TYPES.START:
        group.appendChild(createStartSVG());
        break;
      case BLOCK_TYPES.END:
        group.appendChild(createEndSVG());
        break;
    }
    svg.appendChild(group);
    el.appendChild(svg);

    // Add rotate button if not fixed and not empty and not start/end
    if (!block.fixed && block.type !== BLOCK_TYPES.EMPTY && block.type !== BLOCK_TYPES.START && block.type !== BLOCK_TYPES.END) {
      const rotateBtn = document.createElement('div');
      rotateBtn.className = 'rotate-btn';
      rotateBtn.title = 'Rotate block';
      rotateBtn.appendChild(createRotateButtonSVG());
      rotateBtn.addEventListener('click', e => {
        e.stopPropagation();
        saveStateForUndo();
        rotateBlock(block.id);
      });
      el.appendChild(rotateBtn);
    }

    // Drag handlers
    if (!block.fixed) {
      el.addEventListener('pointerdown', onPointerDown);
      el.addEventListener('keydown', onBlockKeyDown);
      el.addEventListener('mouseenter', onBlockMouseEnter);
      el.addEventListener('mouseleave', onBlockMouseLeave);
    }

    return el;
  }

  // Rotate a block by 90 degrees clockwise
  function rotateBlock(id) {
    const block = blocks.find(b => b.id === id);
    if (!block) return;
    block.rotation = (block.rotation + 1) % 4;
    renderBlocks();
    checkWinCondition();
  }

  // Convert position object to key string
  function posKey(pos) {
    return `${pos.x},${pos.y}`;
  }

  // Check if a position is inside the grid
  function isInsideGrid(pos) {
    return pos.x >= 0 && pos.x < gridSize && pos.y >= 0 && pos.y < gridSize;
  }

  // Find block at position
  function findBlockAt(pos) {
    const key = posKey(pos);
    const idx = positionMap.get(key);
    if (idx === undefined) return null;
    return blocks[idx];
  }

  // Update position map from blocks array
  function updatePositionMap() {
    positionMap.clear();
    blocks.forEach((b, i) => {
      positionMap.set(posKey(b.position), i);
    });
  }

  // Drag and drop handlers
  function onPointerDown(e) {
    if (e.button !== 0) return; // left click only
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    saveStateForUndo();

    draggingBlock = block;
    el.classList.add('dragging');
    el.setAttribute('aria-grabbed', 'true');

    const rect = el.getBoundingClientRect();
    dragOffset.x = e.clientX - rect.left;
    dragOffset.y = e.clientY - rect.top;

    window.addEventListener('pointermove', onPointerMove);
    window.addEventListener('pointerup', onPointerUp);
    window.addEventListener('pointercancel', onPointerUp);

    e.preventDefault();
  }

  function onPointerMove(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    // Calculate new position relative to grid container
    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40; // center offset
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    // Snap to grid cells
    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    // Clamp inside grid
    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, do not move there, but still show dragging
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
      return;
    }

    // Move block visually to pointer position (free drag)
    el.style.left = `${e.clientX - gridRect.left - dragOffset.x}px`;
    el.style.top = `${e.clientY - gridRect.top - dragOffset.y}px`;
  }

  function onPointerUp(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40;
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, revert to original position
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
      // Undo the saved state since no move happened
      undoStack.pop();
      updateUndoRedoButtons();
    } else {
      // Update block position
      draggingBlock.position.x = gridX;
      draggingBlock.position.y = gridY;
      updatePositionMap();
      el.style.left = `${gridX * 86}px`;
      el.style.top = `${gridY * 86}px`;
      checkWinCondition();
    }

    el.classList.remove('dragging');
    el.setAttribute('aria-grabbed', 'false');
    draggingBlock = null;

    window.removeEventListener('pointermove', onPointerMove);
    window.removeEventListener('pointerup', onPointerUp);
    window.removeEventListener('pointercancel', onPointerUp);
  }

  // Keyboard accessibility: rotate block on Enter or Space, move with arrow keys
  function onBlockKeyDown(e) {
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    switch (e.key) {
      case 'Enter':
      case ' ':
        e.preventDefault();
        saveStateForUndo();
        rotateBlock(id);
        break;
      case 'ArrowUp':
      case 'ArrowDown':
      case 'ArrowLeft':
      case 'ArrowRight':
        e.preventDefault();
        saveStateForUndo();
        moveBlockWithKeyboard(block, e.key);
        break;
    }
  }

  // Move block with keyboard arrow keys if possible
  function moveBlockWithKeyboard(block, arrowKey) {
    const dirMap = {
      ArrowUp: { x: 0, y: -1 },
      ArrowDown: { x: 0, y: 1 },
      ArrowLeft: { x: -1, y: 0 },
      ArrowRight: { x: 1, y: 0 },
    };
    const delta = dirMap[arrowKey];
    if (!delta) return;

    const newPos = { x: block.position.x + delta.x, y: block.position.y + delta.y };
    if (!isInsideGrid(newPos)) {
      undoStack.pop();
      updateUndoRedoButtons();
      return;
    }

    const key = posKey(newPos);
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined) {
      undoStack.pop();
      updateUndoRedoButtons();
      return; // occupied
    }

    block.position = newPos;
    updatePositionMap();
    renderBlocks();
    checkWinCondition();

    // Focus moved block
    setTimeout(() => {
      const newEl = gridElement.querySelector(`.block[data-id="${block.id}"]`);
      if (newEl) newEl.focus();
    }, 0);
  }

  // Check if the path from start to end is connected properly
  // We simulate the path traversal from start block's exit direction
  // and follow connected blocks until we reach the end block or dead end
  function checkWinCondition() {
    const level = LEVELS[currentLevelIndex];
    const startBlock = blocks.find(b => b.type === BLOCK_TYPES.START);
    const endBlock = blocks.find(b => b.type === BLOCK_TYPES.END);
    if (!startBlock || !endBlock) return;

    // Get start block connections with rotation applied
    const startConns = rotateDirections(BLOCK_CONNECTIONS[startBlock.type], startBlock.rotation);
    if (startConns.length !== 1) return; // start block must have exactly one connection
    const startDir = startConns[0];

    // Traverse path
    let currentPos = { x: startBlock.position.x, y: startBlock.position.y };
    let currentDir = startDir;
    let visited = new Set();
    visited.add(posKey(currentPos));

    while (true) {
      // Move to next cell in currentDir
      const vec = DIR_VECTORS[currentDir];
      const nextPos = { x: currentPos.x + vec.x, y: currentPos.y + vec.y };
      if (!isInsideGrid(nextPos)) break;

      const nextBlock = findBlockAt(nextPos);
      if (!nextBlock) break;

      // Get next block connections with rotation
      const nextConns = rotateDirections(BLOCK_CONNECTIONS[nextBlock.type], nextBlock.rotation);

      // Check if next block has connection opposite to currentDir
      const oppositeDir = (currentDir + 2) % 4;
      if (!nextConns.includes(oppositeDir)) break;

      // If next block is end block, check if it has connection oppositeDir only
      if (nextBlock.type === BLOCK_TYPES.END) {
        if (nextConns.length === 1 && nextConns[0] === oppositeDir) {
          // Win condition met
          messageElement.textContent = '🎉 Cub reached the Star! Level Complete!';
          return true;
        } else {
          break;
        }
      }

      // Find the other connection direction in next block (not oppositeDir)
      const otherDirs = nextConns.filter(d => d !== oppositeDir);
      if (otherDirs.length !== 1) break; // must have exactly one other connection to continue

      currentPos = nextPos;
      currentDir = otherDirs[0];

      const key = posKey(currentPos);
      if (visited.has(key)) break; // loop detected, break
      visited.add(key);
    }

    // If we reach here, no win
    messageElement.textContent = '';
    return false;
  }

  // Undo/Redo functionality
  function saveStateForUndo() {
    // Save deep copy of blocks state
    const state = blocks.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    undoStack.push(state);
    // Clear redo stack on new action
    redoStack = [];
    updateUndoRedoButtons();
  }

  function undo() {
    if (undoStack.length === 0) return;
    const currentState = blocks.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    redoStack.push(currentState);

    const prevState = undoStack.pop();
    blocks = prevState.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    updatePositionMap();
    renderBlocks();
    checkWinCondition();
    updateUndoRedoButtons();
  }

  function redo() {
    if (redoStack.length === 0) return;
    const currentState = blocks.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    undoStack.push(currentState);

    const nextState = redoStack.pop();
    blocks = nextState.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    updatePositionMap();
    renderBlocks();
    checkWinCondition();
    updateUndoRedoButtons();
  }

  function updateUndoRedoButtons() {
    btnUndo.disabled = undoStack.length === 0;
    btnRedo.disabled = redoStack.length === 0;
  }

  // Show path preview starting from hovered block if possible
  function showPathPreview(startBlock) {
    clearPathPreview();

    // We will highlight the path starting from the hovered block's connections
    // For start block, start from its exit direction
    // For others, try each connection direction to find a path

    // Find all possible start directions for preview
    const conns = rotateDirections(BLOCK_CONNECTIONS[startBlock.type], startBlock.rotation);
    if (conns.length === 0) return;

    // For start block, only one direction
    // For others, try all directions to find a path
    let paths = [];

    if (startBlock.type === BLOCK_TYPES.START) {
      paths.push(traversePathFrom(startBlock.position, conns[0]));
    } else {
      for (const dir of conns) {
        const path = traversePathFrom(startBlock.position, dir);
        if (path.length > 1) {
          paths.push(path);
        }
      }
    }

    if (paths.length === 0) return;

    // Choose the longest path for preview
    let longestPath = paths.reduce((a, b) => (a.length > b.length ? a : b));

    // Create path preview elements
    pathPreviewEl = document.createElement('div');
    pathPreviewEl.className = 'path-preview';
    gridElement.appendChild(pathPreviewEl);

    // For each position in path, create an SVG overlay with highlight
    longestPath.forEach(({ pos, block, rotation }) => {
      const previewBlock = createPreviewBlockSVG(block.type, rotation);
      previewBlock.style.position = 'absolute';
      previewBlock.style.left = `${pos.x * 86}px`;
      previewBlock.style.top = `${pos.y * 86}px`;
      previewBlock.style.width = '80px';
      previewBlock.style.height = '80px';
      pathPreviewEl.appendChild(previewBlock);
    });
  }

  // Clear path preview
  function clearPathPreview() {
    if (pathPreviewEl) {
      pathPreviewEl.remove();
      pathPreviewEl = null;
    }
  }

  // Traverse path from position and direction, return array of {pos, block, rotation}
  function traversePathFrom(startPos, startDir) {
    let path = [];
    let currentPos = { x: startPos.x, y: startPos.y };
    let currentDir = startDir;
    let visited = new Set();
    visited.add(posKey(currentPos));

    while (true) {
      const block = findBlockAt(currentPos);
      if (!block) break;
      path.push({ pos: { x: currentPos.x, y: currentPos.y }, block, rotation: block.rotation });

      // Move to next cell in currentDir
      const vec = DIR_VECTORS[currentDir];
      const nextPos = { x: currentPos.x + vec.x, y: currentPos.y + vec.y };
      if (!isInsideGrid(nextPos)) break;

      const nextBlock = findBlockAt(nextPos);
      if (!nextBlock) break;

      // Get next block connections with rotation
      const nextConns = rotateDirections(BLOCK_CONNECTIONS[nextBlock.type], nextBlock.rotation);

      // Check if next block has connection opposite to currentDir
      const oppositeDir = (currentDir + 2) % 4;
      if (!nextConns.includes(oppositeDir)) break;

      // If next block is end block, add and stop
      if (nextBlock.type === BLOCK_TYPES.END) {
        path.push({ pos: { x: nextPos.x, y: nextPos.y }, block: nextBlock, rotation: nextBlock.rotation });
        break;
      }

      // Find the other connection direction in next block (not oppositeDir)
      const otherDirs = nextConns.filter(d => d !== oppositeDir);
      if (otherDirs.length !== 1) break; // must have exactly one other connection to continue

      currentPos = nextPos;
      currentDir = otherDirs[0];

      const key = posKey(currentPos);
      if (visited.has(key)) break; // loop detected, break
      visited.add(key);
    }
    return path;
  }

    // Find first block that differs from solution in position or rotation (excluding fixed blocks)
    for (let i = 0; i < blocks.length; i++) {
      const currentBlock = blocks[i];
      if (currentBlock.fixed) continue;
      const solutionBlock = level.blocks.find(b => b.type === currentBlock.type && b.position.x === currentBlock.position.x && b.position.y === currentBlock.position.y);
      if (!solutionBlock) {
        // Block is misplaced, highlight it
        highlightBlock(currentBlock.id, 'Block is misplaced. Try moving it to the correct position.');
        return;
      }
      if (currentBlock.rotation !== solutionBlock.rotation) {
        highlightBlock(currentBlock.id, 'Block rotation is incorrect. Try rotating it.');
        return;
      }
    }

    // If all blocks match positions and rotations, but no win, suggest to check connections
    messageElement.textContent = 'No hints available. Try adjusting block positions or rotations.';
  }

    const blockEl = gridElement.querySelector(`.block[data-id="${blockId}"]`);

      messageElement.textContent = '';

  // Initialize the game

    gridElement = document.getElementById('grid');

      }
    });

      loadLevel(currentLevelIndex);
    });
    btnUndo.addEventListener('click', () => {
      undo();
    });

  // Load a level by index
  function loadLevel(index) {
    const level = LEVELS[index];
    gridSize = level.gridSize;
    blocks = level.blocks.map(b => ({
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: !!b.fixed,
      id: generateId(),
    }));
    positionMap.clear();
    blocks.forEach((b, i) => {
      positionMap.set(posKey(b.position), i);
    });
    undoStack = [];
    redoStack = [];
    updateUndoRedoButtons();
    renderGrid();
    renderBlocks();
    updateLevelInfo(level);
    updateControls();
    messageElement.textContent = '';
    clearPathPreview();
  }

  // Generate unique ID for blocks
  function generateId() {
    return 'b' + Math.random().toString(36).slice(2, 10);
  }

  // Update level info text
  function updateLevelInfo(level) {
    levelInfoElement.textContent = `Level ${currentLevelIndex + 1} / ${LEVELS.length} - ${level.description}`;
  }

  // Update controls button states
  function updateControls() {
    btnPrev.disabled = currentLevelIndex === 0;
    btnNext.disabled = currentLevelIndex === LEVELS.length - 1;
  }

  // Render the grid cells
  function renderGrid() {
    gridElement.innerHTML = '';
    gridElement.style.gridTemplateColumns = `repeat(${gridSize}, 80px)`;
    gridElement.style.gridTemplateRows = `repeat(${gridSize}, 80px)`;
    for (let y = 0; y < gridSize; y++) {
      for (let x = 0; x < gridSize; x++) {
        const cell = document.createElement('div');
        cell.className = 'cell';
        cell.dataset.x = x;
        cell.dataset.y = y;
        cell.setAttribute('role', 'gridcell');
        cell.setAttribute('aria-label', `Grid cell ${x + 1},${y + 1}`);
        gridElement.appendChild(cell);
      }
    }
  }

  // Render all blocks on the grid
  function renderBlocks() {
    // Remove existing blocks
    const existingBlocks = gridElement.querySelectorAll('.block');
    existingBlocks.forEach(b => b.remove());

    blocks.forEach(block => {
      const blockEl = createBlockElement(block);
      gridElement.appendChild(blockEl);
    });
  }

  // Create a block DOM element with SVG and rotate button
  function createBlockElement(block) {

    el.className = 'block';
    el.style.width = '80px';
    el.style.height = '80px';
    el.style.left = `${block.position.x * 86}px`;
    el.style.top = `${block.position.y * 86}px`;
    el.style.position = 'absolute';
    el.style.touchAction = 'none';
    el.dataset.id = block.id;
    el.dataset.type = block.type;
    el.dataset.fixed = block.fixed ? 'true' : 'false';
    el.setAttribute('aria-grabbed', 'false');

    el.setAttribute('tabindex', block.fixed ? '-1' : '0');
    el.setAttribute('aria-label', `Block type ${block.type}, rotation ${block.rotation * 90} degrees${block.fixed ? ', fixed' : ''}`);

    // Create SVG container
    const svg = createSVGElement('svg', { viewBox: '0 0 80 80', 'aria-hidden': 'true', focusable: 'false' });
    const group = createSVGElement('g');
    group.setAttribute('transform', `rotate(${block.rotation * 90} 40 40)`);

    // Add pipe or icon paths depending on block type
    switch (block.type) {
      case BLOCK_TYPES.EMPTY:

        group.appendChild(createStraightPath());

        group.appendChild(createStartSVG());
        break;

        group.appendChild(createEndSVG());
        break;

    svg.appendChild(group);
    el.appendChild(svg);

    // Add rotate button if not fixed and not empty and not start/end

      const rotateBtn = document.createElement('div');
      rotateBtn.className = 'rotate-btn';
      rotateBtn.title = 'Rotate block';
      rotateBtn.appendChild(createRotateButtonSVG());
      rotateBtn.addEventListener('click', e => {
        e.stopPropagation();
        saveStateForUndo();
        rotateBlock(block.id);
      });
      el.appendChild(rotateBtn);
    }

    // Drag handlers
    if (!block.fixed) {
      el.addEventListener('pointerdown', onPointerDown);
      el.addEventListener('keydown', onBlockKeyDown);
      el.addEventListener('mouseenter', onBlockMouseEnter);
      el.addEventListener('mouseleave', onBlockMouseLeave);
    }

    return el;
  }

  // Rotate a block by 90 degrees clockwise
  function rotateBlock(id) {
    const block = blocks.find(b => b.id === id);
    if (!block) return;
    block.rotation = (block.rotation + 1) % 4;
    renderBlocks();
    checkWinCondition();
  }

  // Convert position object to key string
  function posKey(pos) {
    return `${pos.x},${pos.y}`;
  }

  // Check if a position is inside the grid
  function isInsideGrid(pos) {
    return pos.x >= 0 && pos.x < gridSize && pos.y >= 0 && pos.y < gridSize;
  }

  // Find block at position
  function findBlockAt(pos) {
    const key = posKey(pos);
    const idx = positionMap.get(key);
    if (idx === undefined) return null;
    return blocks[idx];
  }

  // Update position map from blocks array
  function updatePositionMap() {
    positionMap.clear();
    blocks.forEach((b, i) => {
      positionMap.set(posKey(b.position), i);
    });
  }

  // Drag and drop handlers
  function onPointerDown(e) {
    if (e.button !== 0) return; // left click only
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    saveStateForUndo();

    draggingBlock = block;
    el.classList.add('dragging');
    el.setAttribute('aria-grabbed', 'true');

    const rect = el.getBoundingClientRect();
    dragOffset.x = e.clientX - rect.left;
    dragOffset.y = e.clientY - rect.top;

    window.addEventListener('pointermove', onPointerMove);
    window.addEventListener('pointerup', onPointerUp);
    window.addEventListener('pointercancel', onPointerUp);

    e.preventDefault();
  }

  function onPointerMove(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    // Calculate new position relative to grid container
    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40; // center offset
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    // Snap to grid cells
    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    // Clamp inside grid
    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, do not move there, but still show dragging
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
      return;
    }

    // Move block visually to pointer position (free drag)
    el.style.left = `${e.clientX - gridRect.left - dragOffset.x}px`;
    el.style.top = `${e.clientY - gridRect.top - dragOffset.y}px`;
  }

  function onPointerUp(e) {
    if (!draggingBlock) return;
    const el = gridElement.querySelector(`.block[data-id="${draggingBlock.id}"]`);
    if (!el) return;

    const gridRect = gridElement.getBoundingClientRect();
    let x = e.clientX - gridRect.left - dragOffset.x + 40;
    let y = e.clientY - gridRect.top - dragOffset.y + 40;

    let gridX = Math.floor(x / 86);
    let gridY = Math.floor(y / 86);

    gridX = Math.min(Math.max(gridX, 0), gridSize - 1);
    gridY = Math.min(Math.max(gridY, 0), gridSize - 1);

    // Check if target cell is occupied by another block (except self)
    const key = posKey({ x: gridX, y: gridY });
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined && blocks[occupiedIdx].id !== draggingBlock.id) {
      // Occupied, revert to original position
      el.style.left = `${draggingBlock.position.x * 86}px`;
      el.style.top = `${draggingBlock.position.y * 86}px`;
      // Undo the saved state since no move happened
      undoStack.pop();
      updateUndoRedoButtons();
    } else {
      // Update block position
      draggingBlock.position.x = gridX;
      draggingBlock.position.y = gridY;
      updatePositionMap();
      el.style.left = `${gridX * 86}px`;
      el.style.top = `${gridY * 86}px`;
      checkWinCondition();
    }

    el.classList.remove('dragging');
    el.setAttribute('aria-grabbed', 'false');
    draggingBlock = null;

    window.removeEventListener('pointermove', onPointerMove);
    window.removeEventListener('pointerup', onPointerUp);
    window.removeEventListener('pointercancel', onPointerUp);
  }

  // Keyboard accessibility: rotate block on Enter or Space, move with arrow keys
  function onBlockKeyDown(e) {
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block || block.fixed) return;

    switch (e.key) {
      case 'Enter':
      case ' ':
        e.preventDefault();
        saveStateForUndo();
        rotateBlock(id);
        break;
      case 'ArrowUp':
      case 'ArrowDown':
      case 'ArrowLeft':
      case 'ArrowRight':
        e.preventDefault();
        saveStateForUndo();
        moveBlockWithKeyboard(block, e.key);
        break;
    }
  }

  // Move block with keyboard arrow keys if possible
  function moveBlockWithKeyboard(block, arrowKey) {
    const dirMap = {
      ArrowUp: { x: 0, y: -1 },
      ArrowDown: { x: 0, y: 1 },
      ArrowLeft: { x: -1, y: 0 },
      ArrowRight: { x: 1, y: 0 },
    };
    const delta = dirMap[arrowKey];
    if (!delta) return;

    const newPos = { x: block.position.x + delta.x, y: block.position.y + delta.y };
    if (!isInsideGrid(newPos)) {
      undoStack.pop();
      updateUndoRedoButtons();
      return;
    }

    const key = posKey(newPos);
    const occupiedIdx = positionMap.get(key);
    if (occupiedIdx !== undefined) {
      undoStack.pop();
      updateUndoRedoButtons();
      return; // occupied
    }

    block.position = newPos;
    updatePositionMap();
    renderBlocks();
    checkWinCondition();

    // Focus moved block
    setTimeout(() => {
      const newEl = gridElement.querySelector(`.block[data-id="${block.id}"]`);
      if (newEl) newEl.focus();
    }, 0);
  }

  // Check if the path from start to end is connected properly
  // We simulate the path traversal from start block's exit direction
  // and follow connected blocks until we reach the end block or dead end
  function checkWinCondition() {
    const level = LEVELS[currentLevelIndex];
    const startBlock = blocks.find(b => b.type === BLOCK_TYPES.START);
    const endBlock = blocks.find(b => b.type === BLOCK_TYPES.END);
    if (!startBlock || !endBlock) return;

    // Get start block connections with rotation applied
    const startConns = rotateDirections(BLOCK_CONNECTIONS[startBlock.type], startBlock.rotation);
    if (startConns.length !== 1) return; // start block must have exactly one connection
    const startDir = startConns[0];

    // Traverse path
    let currentPos = { x: startBlock.position.x, y: startBlock.position.y };
    let currentDir = startDir;
    let visited = new Set();
    visited.add(posKey(currentPos));

    while (true) {
      // Move to next cell in currentDir
      const vec = DIR_VECTORS[currentDir];
      const nextPos = { x: currentPos.x + vec.x, y: currentPos.y + vec.y };
      if (!isInsideGrid(nextPos)) break;

      const nextBlock = findBlockAt(nextPos);
      if (!nextBlock) break;

      // Get next block connections with rotation
      const nextConns = rotateDirections(BLOCK_CONNECTIONS[nextBlock.type], nextBlock.rotation);

      // Check if next block has connection opposite to currentDir
      const oppositeDir = (currentDir + 2) % 4;
      if (!nextConns.includes(oppositeDir)) break;

      // If next block is end block, check if it has connection oppositeDir only
      if (nextBlock.type === BLOCK_TYPES.END) {
        if (nextConns.length === 1 && nextConns[0] === oppositeDir) {
          // Win condition met
          messageElement.textContent = '🎉 Cub reached the Star! Level Complete!';
          return true;
        } else {
          break;
        }
      }

      // Find the other connection direction in next block (not oppositeDir)
      const otherDirs = nextConns.filter(d => d !== oppositeDir);
      if (otherDirs.length !== 1) break; // must have exactly one other connection to continue

      currentPos = nextPos;
      currentDir = otherDirs[0];

      const key = posKey(currentPos);
      if (visited.has(key)) break; // loop detected, break
      visited.add(key);
    }

    // If we reach here, no win
    messageElement.textContent = '';
    return false;
  }

  // Undo/Redo functionality
  function saveStateForUndo() {
    // Save deep copy of blocks state
    const state = blocks.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    undoStack.push(state);
    // Clear redo stack on new action
    redoStack = [];
    updateUndoRedoButtons();
  }

  function undo() {
    if (undoStack.length === 0) return;
    const currentState = blocks.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    redoStack.push(currentState);

    const prevState = undoStack.pop();
    blocks = prevState.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    updatePositionMap();
    renderBlocks();
    checkWinCondition();
    updateUndoRedoButtons();
  }

  function redo() {
    if (redoStack.length === 0) return;
    const currentState = blocks.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    undoStack.push(currentState);

    const nextState = redoStack.pop();
    blocks = nextState.map(b => ({
      id: b.id,
      type: b.type,
      rotation: b.rotation,
      position: { x: b.position.x, y: b.position.y },
      fixed: b.fixed,
    }));
    updatePositionMap();
    renderBlocks();
    checkWinCondition();
    updateUndoRedoButtons();
  }

  function updateUndoRedoButtons() {
    btnUndo.disabled = undoStack.length === 0;
    btnRedo.disabled = redoStack.length === 0;
  }

  // Path preview on hover
  function onBlockMouseEnter(e) {
    const el = e.currentTarget;
    const id = el.dataset.id;
    const block = blocks.find(b => b.id === id);
    if (!block) return;
    showPathPreview(block);
  }

  function onBlockMouseLeave(e) {
    clearPathPreview();
  }

  // Show path preview starting from hovered block if possible
  function showPathPreview(startBlock) {
    clearPathPreview();

    // We will highlight the path starting from the hovered block's connections
    // For start block, start from its exit direction
    // For others, try each connection direction to find a path

    // Find all possible start directions for preview
    const conns = rotateDirections(BLOCK_CONNECTIONS[startBlock.type], startBlock.rotation);
    if (conns.length === 0) return;

    // For start block, only one direction
    // For others, try all directions to find a path
    let paths = [];

    if (startBlock.type === BLOCK_TYPES.START) {
      paths.push(traversePathFrom(startBlock.position, conns[0]));
    } else {
      for (const dir of conns) {
        const path = traversePathFrom(startBlock.position, dir);
        if (path.length > 1) {
          paths.push(path);
        }
      }
    }

    if (paths.length === 0) return;

    // Choose the longest path for preview
    let longestPath = paths.reduce((a, b) => (a.length > b.length ? a : b));

    // Create path preview elements
    pathPreviewEl = document.createElement('div');
    pathPreviewEl.className = 'path-preview';
    gridElement.appendChild(pathPreviewEl);

    // For each position in path, create an SVG overlay with highlight
    longestPath.forEach(({ pos, block, rotation }) => {
      const previewBlock = createPreviewBlockSVG(block.type, rotation);
      previewBlock.style.position = 'absolute';
      previewBlock.style.left = `${pos.x * 86}px`;
      previewBlock.style.top = `${pos.y * 86}px`;
      previewBlock.style.width = '80px';
      previewBlock.style.height = '80px';
      pathPreviewEl.appendChild(previewBlock);
    });
  }

  // Clear path preview
  function clearPathPreview() {
    if (pathPreviewEl) {
      pathPreviewEl.remove();
      pathPreviewEl = null;
    }
  }

  // Traverse path from position and direction, return array of {pos, block, rotation}
  function traversePathFrom(startPos, startDir) {
    let path = [];
    let currentPos = { x: startPos.x, y: startPos.y };
    let currentDir = startDir;
    let visited = new Set();
    visited.add(posKey(currentPos));

    while (true) {
      const block = findBlockAt(currentPos);
      if (!block) break;
      path.push({ pos: { x: currentPos.x, y: currentPos.y }, block, rotation: block.rotation });

      // Move to next cell in currentDir
      const vec = DIR_VECTORS[currentDir];
      const nextPos = { x: currentPos.x + vec.x, y: currentPos.y + vec.y };
      if (!isInsideGrid(nextPos)) break;

      const nextBlock = findBlockAt(nextPos);
      if (!nextBlock) break;

      // Get next block connections with rotation
      const nextConns = rotateDirections(BLOCK_CONNECTIONS[nextBlock.type], nextBlock.rotation);

      // Check if next block has connection opposite to currentDir
      const oppositeDir = (currentDir + 2) % 4;
      if (!nextConns.includes(oppositeDir)) break;

      // If next block is end block, add and stop
      if (nextBlock.type === BLOCK_TYPES.END) {
        path.push({ pos: { x: nextPos.x, y: nextPos.y }, block: nextBlock, rotation: nextBlock.rotation });
        break;
      }

      // Find the other connection direction in next block (not oppositeDir)
      const otherDirs = nextConns.filter(d => d !== oppositeDir);
      if (otherDirs.length !== 1) break; // must have exactly one other connection to continue

      currentPos = nextPos;
      currentDir = otherDirs[0];

      const key = posKey(currentPos);
      if (visited.has(key)) break; // loop detected, break
      visited.add(key);
    }
    return path;
  }

      'stroke-width': 16,
      'stroke-linecap': 'round',

  // Hint system: show next correct move by comparing current state to solution
  function showHint() {
    clearPathPreview();
    messageElement.textContent = '';

    const level = LEVELS[currentLevelIndex];
    if (!level) return;

    // Find first block that differs from solution in position or rotation (excluding fixed blocks)
    for (let i = 0; i < blocks.length; i++) {
      const currentBlock = blocks[i];
      if (currentBlock.fixed) continue;
      const solutionBlock = level.blocks.find(b => b.type === currentBlock.type && b.position.x === currentBlock.position.x && b.position.y === currentBlock.position.y);
      if (!solutionBlock) {
        // Block is misplaced, highlight it
        highlightBlock(currentBlock.id, 'Block is misplaced. Try moving it to the correct position.');
        return;
      }
      if (currentBlock.rotation !== solutionBlock.rotation) {
        highlightBlock(currentBlock.id, 'Block rotation is incorrect. Try rotating it.');
        return;
      }
    }

    // If all blocks match positions and rotations, but no win, suggest to check connections
    messageElement.textContent = 'No hints available. Try adjusting block positions or rotations.';
  }

  // Highlight a block with a temporary message
  function highlightBlock(blockId, hintMessage) {
    const blockEl = gridElement.querySelector(`.block[data-id="${blockId}"]`);
    if (!blockEl) return;
    blockEl.classList.add('highlight');
    messageElement.textContent = hintMessage;
    setTimeout(() => {
      blockEl.classList.remove('highlight');
      messageElement.textContent = '';
    }, 4000);
  }

  window.addEventListener('load', init);
