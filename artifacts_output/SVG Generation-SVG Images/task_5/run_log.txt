  const liquid = document.getElementById('liquid');
  const wipeStrokes = [
    document.getElementById('wipe1'),

    document.getElementById('wipe3'),
  ];

  // Animate mop swiping and liquid disappearing
  // Timeline:
  // 0-4s: mop swipes over liquid repeatedly, liquid fades out gradually
  // 4-6s: mop wipes remaining liquid with wipe strokes fading out
  // Loop the entire animation

  function animateCleaning() {
    // Reset states
    liquid.style.opacity = 0.8;
    liquid.classList.remove('wiped');
    wipeStrokes.forEach(w => {
      w.style.opacity = 0.6;
      w.classList.remove('wiped');
    });
    mopHandle.classList.add('moving');

    // Animate liquid fade out over 4 seconds
    let start = null;
    function fadeLiquid(timestamp) {
      if (!start) start = timestamp;
      const elapsed = timestamp - start;
      const duration = 4000;
      const progress = Math.min(elapsed / duration, 1);
      liquid.style.opacity = 0.8 * (1 - progress);
      if (progress < 1) {
        requestAnimationFrame(fadeLiquid);
      } else {
        liquid.style.opacity = 0;
        liquid.classList.add('wiped');
        mopHandle.classList.remove('moving');
        wipeRemaining();
      }
    }
    requestAnimationFrame(fadeLiquid);
  }

  function wipeRemaining() {
    // Show wipe strokes and fade them out over 2 seconds
    let start = null;
    function fadeWipes(timestamp) {
      if (!start) start = timestamp;
      const elapsed = timestamp - start;
      const duration = 2000;
      const progress = Math.min(elapsed / duration, 1);
      wipeStrokes.forEach(w => {
        w.style.opacity = 0.6 * (1 - progress);
      });
      if (progress < 1) {
        requestAnimationFrame(fadeWipes);
      } else {
        wipeStrokes.forEach(w => {
          w.style.opacity = 0;
          w.classList.add('wiped');
        });
        setTimeout(() => {
          animateCleaning();
        }, 1000);
      }
    }
    requestAnimationFrame(fadeWipes);
  }

  animateCleaning();
