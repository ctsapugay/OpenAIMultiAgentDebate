      this.damage = damage;

      // Calculate normalized direction vector
      const dx = targetX - x;
      const dy = targetY - y;
      const dist = Math.hypot(dx, dy);
      this.vx = (dx / dist) * this.speed;
      this.vy = (dy / dist) * this.speed;

      this.isDead = false;
    }

      this.x += this.vx * delta;
      this.y += this.vy * delta;

      // Remove projectile if out of bounds

        this.y < -this.radius || this.y > GAME_HEIGHT + this.radius

      ctx.save();

      ctx.fillStyle = this.color;

      ctx.shadowBlur = 10;

      ctx.fill();
      ctx.restore();
    }

      const dy = player.movingTo.y - player.y;

        player.x = player.movingTo.x;
        player.y = player.movingTo.y;
        player.movingTo = null;
      } else {

      player.y = clamp(player.y, player.radius, EARTH_Y - player.radius - 10);

      endGame();

  }

  }

  function gameLoop(timestamp = 0) {

    deltaTime = timestamp - lastTime;

      this.type = type;
      this.x = x;
      this.y = y;
      this.radius = type.radius;
      this.color = type.color;
      this.speed = speed;
      this.health = health;
      this.maxHealth = health;
      this.scoreValue = scoreValue;
      this.isDead = false;
    }

    update(delta) {
      // Move enemy downward toward Earth
      this.y += this.speed * delta;

      // Check if reached Earth (bottom)
      if (this.y + this.radius >= EARTH_Y) {
        this.isDead = true;
        // Damage player shield directly
        player.shield -= this.health * 0.5; // enemy health as damage factor
        if (player.shield < 0) player.shield = 0;
      }
    }

    draw(ctx) {
      // Draw enemy body
      ctx.save();
      ctx.beginPath();
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 8;
      ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
      ctx.fill();

      // Draw health bar above enemy
      const barWidth = this.radius * 2;
      const barHeight = 5;
      const healthRatio = this.health / this.maxHealth;
      ctx.fillStyle = '#222';
      ctx.fillRect(this.x - this.radius, this.y - this.radius - 12, barWidth, barHeight);
      ctx.fillStyle = '#ff4444';
      ctx.fillRect(this.x - this.radius, this.y - this.radius - 12, barWidth * healthRatio, barHeight);
      ctx.restore();
    }
  }

  // Projectile class
  class Projectile {
    constructor(x, y, targetX, targetY, damage) {
      this.x = x;
      this.y = y;
      this.radius = PROJECTILE_RADIUS;
      this.color = '#7fffd4';
      this.speed = PROJECTILE_SPEED;
      this.damage = damage;

      // Calculate normalized direction vector
      const dx = targetX - x;
      const dy = targetY - y;
      const dist = Math.hypot(dx, dy);
      this.vx = (dx / dist) * this.speed;
      this.vy = (dy / dist) * this.speed;

      this.isDead = false;
    }

    update(delta) {
      this.x += this.vx * delta;
      this.y += this.vy * delta;

      // Remove projectile if out of bounds
      if (
        this.x < -this.radius || this.x > GAME_WIDTH + this.radius ||
        this.y < -this.radius || this.y > GAME_HEIGHT + this.radius
      ) {
        this.isDead = true;
      }
    }

    draw(ctx) {
      ctx.save();
      ctx.beginPath();
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 10;
      ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  }

  // Draw the player target with weapon and shield
  function drawPlayer(ctx) {
    ctx.save();
    ctx.translate(player.x, player.y);

    // Draw shield ring
    const shieldRatio = player.shield / player.maxShield;
    const shieldColor = `rgba(127, 255, 212, ${0.3 + 0.7 * shieldRatio})`;
    ctx.beginPath();
    ctx.strokeStyle = shieldColor;
    ctx.lineWidth = 6;
    ctx.shadowColor = shieldColor;
    ctx.shadowBlur = 15;
    ctx.arc(0, 0, player.radius + 8, 0, Math.PI * 2);
    ctx.stroke();

    // Draw target body
    ctx.beginPath();
    ctx.fillStyle = '#00ffff';
    ctx.shadowColor = '#00ffff';
    ctx.shadowBlur = 20;
    ctx.arc(0, 0, player.radius, 0, Math.PI * 2);
    ctx.fill();

    // Draw weapon barrel (pointing upward)
    ctx.beginPath();
    ctx.fillStyle = '#00cccc';
    ctx.shadowColor = '#00cccc';
    ctx.shadowBlur = 10;
    ctx.rect(-6, -player.radius - 14, 12, 14);
    ctx.fill();

    ctx.restore();
  }

  // Draw Earth at bottom center
  function drawEarth(ctx) {
    const earthRadius = 60;
    const earthX = GAME_WIDTH / 2;
    const earthY = EARTH_Y + earthRadius - 10;

    // Earth glow
    const gradient = ctx.createRadialGradient(earthX, earthY, earthRadius * 0.5, earthX, earthY, earthRadius);
    gradient.addColorStop(0, '#004400');
    gradient.addColorStop(1, '#001100');

    ctx.save();
    ctx.beginPath();
    ctx.fillStyle = gradient;
    ctx.shadowColor = '#00ff00';
    ctx.shadowBlur = 20;
    ctx.arc(earthX, earthY, earthRadius, 0, Math.PI * 2);
    ctx.fill();

    // Earth surface details (simple)
    ctx.fillStyle = '#003300';
    for (let i = 0; i < 10; i++) {
      const angle = (i / 10) * Math.PI * 2;
      const x = earthX + Math.cos(angle) * earthRadius * 0.7;
      const y = earthY + Math.sin(angle) * earthRadius * 0.7;
      ctx.beginPath();
      ctx.arc(x, y, 6, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();
  }

  // Spawn a new enemy at random top position
  function spawnEnemy() {
    // Randomly select enemy type weighted by level
    const typeIndex = Math.min(level - 1, ENEMY_TYPES.length - 1);
    const type = ENEMY_TYPES[typeIndex];

    // Random x position within canvas width, with margin
    const margin = 30;
    const x = Math.random() * (GAME_WIDTH - margin * 2) + margin;
    const y = -type.radius - 10;

    // Speed and health scale with level and type
    const speed = enemySpeed * type.speedMultiplier;
    const health = enemyHealth * type.healthMultiplier;

    const enemy = new Enemy(type, x, y, speed, health, type.score);
    enemies.push(enemy);
  }

  // Handle player shooting
  function playerShoot(targetX, targetY, time) {
    if (time - player.lastShotTime < player.fireRate) return;
    if (player.energy < 10) return; // Require energy to shoot

    player.lastShotTime = time;
    player.energy -= 10;

    // Create projectile from player position toward target
    const proj = new Projectile(player.x, player.y - player.radius - 10, targetX, targetY, player.damage);
    projectiles.push(proj);
  }

  // Update all game objects
  function update(delta, time) {
    if (!gameRunning) return;

    // Regenerate energy
    player.energy += (ENERGY_REGEN_RATE * delta) / 1000;
    if (player.energy > player.maxEnergy) player.energy = player.maxEnergy;

    // Move player toward movingTo if set (smooth movement)
    if (player.movingTo) {
      const dx = player.movingTo.x - player.x;
      const dy = player.movingTo.y - player.y;
      const dist = Math.hypot(dx, dy);
      if (dist < player.speed) {
        player.x = player.movingTo.x;
        player.y = player.movingTo.y;
        player.movingTo = null;
      } else {
        player.x += (dx / dist) * player.speed;
        player.y += (dy / dist) * player.speed;
      }
      // Clamp player position inside canvas and above Earth
      player.x = clamp(player.x, player.radius, GAME_WIDTH - player.radius);
      player.y = clamp(player.y, player.radius, EARTH_Y - player.radius - 10);
    }

    // Update enemies
    enemies.forEach(enemy => enemy.update(delta));

    // Update projectiles
    projectiles.forEach(proj => proj.update(delta));

    // Collision detection: projectiles vs enemies
    projectiles.forEach(proj => {
      if (proj.isDead) return;
      enemies.forEach(enemy => {
        if (enemy.isDead) return;
        const dist = distance(proj.x, proj.y, enemy.x, enemy.y);
        if (dist < proj.radius + enemy.radius) {
          // Hit detected
          enemy.health -= proj.damage;
          proj.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += enemy.scoreValue;
            resources += Math.floor(enemy.scoreValue / 2);
          }
        }
      });
    });

    // Remove dead enemies and projectiles
    enemies = enemies.filter(e => !e.isDead);
    projectiles = projectiles.filter(p => !p.isDead);

    // Check player shield health
    if (player.shield <= 0) {
      player.shield = 0;
      endGame();
    }

    // Spawn enemies based on timer
    enemySpawnTimer += delta;
    if (enemySpawnTimer >= enemySpawnInterval) {
      enemySpawnTimer = 0;
      spawnEnemy();
    }

    // Level progression: increase difficulty every 30 seconds
    if (time > level * 30000) {
      level++;
      levelEl.textContent = level;
      enemySpawnInterval = Math.max(500, ENEMY_SPAWN_INTERVAL_BASE - (level - 1) * 150);
      enemySpeed = ENEMY_SPEED_BASE + (level - 1) * 0.15;
      enemyHealth = ENEMY_HEALTH_BASE + (level - 1) * 5;
      tipEl.textContent = getRandomTip();
    }

    // Update UI
    scoreEl.textContent = score;
    healthEl.textContent = Math.floor(player.shield);
    energyEl.textContent = Math.floor(player.energy);

    // Update upgrade costs display
    costDamageEl.textContent = upgrades.damage.cost;
    costFireRateEl.textContent = upgrades.fireRate.cost;
    costSpeedEl.textContent = upgrades.speed.cost;
    costShieldEl.textContent = upgrades.shield.cost;

    // Enable/disable upgrade buttons based on resources
    upgradeDamageBtn.disabled = resources < upgrades.damage.cost;
    upgradeFireRateBtn.disabled = resources < upgrades.fireRate.cost;
    upgradeSpeedBtn.disabled = resources < upgrades.speed.cost;
    upgradeShieldBtn.disabled = resources < upgrades.shield.cost;
  }

  // Draw all game objects
  function draw() {
    // Clear canvas
    ctx.clearRect(0, 0, GAME_WIDTH, GAME_HEIGHT);

    // Draw Earth
    drawEarth(ctx);

    // Draw enemies
    enemies.forEach(enemy => enemy.draw(ctx));

    // Draw projectiles
    projectiles.forEach(proj => proj.draw(ctx));

    // Draw player
    drawPlayer(ctx);
  }

  // Game loop
  function gameLoop(timestamp = 0) {
    if (!lastTime) lastTime = timestamp;
    deltaTime = timestamp - lastTime;
    lastTime = timestamp;

    update(deltaTime, timestamp);
    draw();

    if (gameRunning) {
      requestAnimationFrame(gameLoop);
    }
  }

  // Handle game start
  function startGame() {
    // Reset game state
    gameRunning = true;
    lastTime = 0;
    deltaTime = 0;
    score = 0;
    level = 1;
    enemies = [];
    projectiles = [];
    enemySpawnTimer = 0;
    enemySpawnInterval = ENEMY_SPAWN_INTERVAL_BASE;
    enemySpeed = ENEMY_SPEED_BASE;
    enemyHealth = ENEMY_HEALTH_BASE;
    resources = 0;

    // Reset player stats
    player.x = GAME_WIDTH / 2;
    player.y = EARTH_Y - TARGET_RADIUS - 10;
    player.speed = TARGET_BASE_SPEED + upgrades.speed.level * upgrades.speed.increment;
    player.fireRate = Math.max(100, TARGET_BASE_FIRE_RATE - upgrades.fireRate.level * upgrades.fireRate.increment);
    player.damage = TARGET_BASE_DAMAGE + upgrades.damage.level * upgrades.damage.increment;
    player.maxShield = TARGET_BASE_SHIELD + upgrades.shield.level * upgrades.shield.increment;
    player.shield = player.maxShield;
    player.energy = player.maxEnergy;
    player.lastShotTime = 0;
    player.movingTo = null;

    // Update UI
    scoreEl.textContent = score;
    levelEl.textContent = level;
    healthEl.textContent = player.shield;
    energyEl.textContent = player.energy;

    tipEl.textContent = getRandomTip();

    // Hide game over screen
    gameOverScreen.hidden = true;
    gameOverScreen.setAttribute('aria-hidden', 'true');

    // Disable start button
    startButton.disabled = true;

    // Start game loop
    requestAnimationFrame(gameLoop);
  }

  // Handle game over
  function endGame() {
    gameRunning = false;
    finalScoreEl.textContent = score;
    gameOverScreen.hidden = false;
    gameOverScreen.setAttribute('aria-hidden', 'false');
    startButton.disabled = false;
  }

  // Get random gameplay tip

  // Input handling for player movement and shooting
  function setupInput() {
    const container = document.getElementById('game-container');

    // Mouse and touch dragging for movement
    function pointerDown(e) {
      e.preventDefault();
      isDragging = true;
      container.classList.remove('grab');
      container.classList.add('grabbing');

      const rect = canvas.getBoundingClientRect();
      const pointerX = e.clientX ?? (e.touches ? e.touches[0].clientX : 0);
      const pointerY = e.clientY ?? (e.touches ? e.touches[0].clientY : 0);

      dragOffset.x = player.x - (pointerX - rect.left);
      dragOffset.y = player.y - (pointerY - rect.top);
    }
    function pointerMove(e) {
      if (!isDragging) return;
      e.preventDefault();

      const rect = canvas.getBoundingClientRect();
      const pointerX = e.clientX ?? (e.touches ? e.touches[0].clientX : 0);
      const pointerY = e.clientY ?? (e.touches ? e.touches[0].clientY : 0);

      let newX = pointerX - rect.left + dragOffset.x;
      let newY = pointerY - rect.top + dragOffset.y;

      // Clamp inside canvas and above Earth
      newX = clamp(newX, player.radius, GAME_WIDTH - player.radius);
      newY = clamp(newY, player.radius, EARTH_Y - player.radius - 10);

      player.x = newX;
      player.y = newY;
      player.movingTo = null; // cancel smooth movement if dragging
    }
    function pointerUp(e) {
      e.preventDefault();
      isDragging = false;
      container.classList.remove('grabbing');
      container.classList.add('grab');
    }

    canvas.addEventListener('mousedown', pointerDown);
    canvas.addEventListener('touchstart', pointerDown, { passive: false });

    window.addEventListener('mousemove', pointerMove);
    window.addEventListener('touchmove', pointerMove, { passive: false });

    window.addEventListener('mouseup', pointerUp);
    window.addEventListener('touchend', pointerUp);
    window.addEventListener('touchcancel', pointerUp);

    // Shooting on click or tap (if not dragging)
    canvas.addEventListener('click', e => {
      if (isDragging) return;
      const rect = canvas.getBoundingClientRect();
      const clickX = e.clientX - rect.left;
      const clickY = e.clientY - rect.top;
      playerShoot(clickX, clickY, performance.now());
    });

    // Keyboard controls: spacebar to shoot, arrow keys or WASD to move
    window.addEventListener('keydown', e => {
      if (!gameRunning) return;
      if (e.code === 'Space') {
        e.preventDefault();
        // Shoot straight upward from player center
        playerShoot(player.x, player.y - 100, performance.now());
      }
      // Movement keys move player smoothly
      const moveAmount = player.speed;
      switch (e.code) {
        case 'ArrowLeft':
        case 'KeyA':
          player.movingTo = { x: clamp(player.x - moveAmount * 5, player.radius, GAME_WIDTH - player.radius), y: player.y };
          break;
        case 'ArrowRight':
        case 'KeyD':
          player.movingTo = { x: clamp(player.x + moveAmount * 5, player.radius, GAME_WIDTH - player.radius), y: player.y };
          break;
        case 'ArrowUp':
        case 'KeyW':
          player.movingTo = { x: player.x, y: clamp(player.y - moveAmount * 5, player.radius, EARTH_Y - player.radius - 10) };
          break;
        case 'ArrowDown':
        case 'KeyS':
          player.movingTo = { x: player.x, y: clamp(player.y + moveAmount * 5, player.radius, EARTH_Y - player.radius - 10) };
          break;
      }
    });
  }

  }

    const type = ENEMY_TYPES[typeIndex];

    // Random x position within canvas width, with margin
    const margin = 30;
    const x = Math.random() * (GAME_WIDTH - margin * 2) + margin;
    const y = -type.radius - 10;

    // Speed and health scale with level and type
    const speed = enemySpeed * type.speedMultiplier;
    const health = enemyHealth * type.healthMultiplier;

    const enemy = new Enemy(type, x, y, speed, health, type.score);
    enemies.push(enemy);
  }

  // Handle player shooting
  function playerShoot(targetX, targetY, time) {
    if (time - player.lastShotTime < player.fireRate) return;
    if (player.energy < 10) return; // Require energy to shoot

    player.lastShotTime = time;
    player.energy -= 10;

    // Create projectile from player position toward target
    const proj = new Projectile(player.x, player.y - player.radius - 10, targetX, targetY, player.damage);
    projectiles.push(proj);
  }

  // Update all game objects
  function update(delta, time) {
    if (!gameRunning) return;

    // Regenerate energy
    player.energy += (ENERGY_REGEN_RATE * delta) / 1000;
    if (player.energy > player.maxEnergy) player.energy = player.maxEnergy;

    // Move player toward movingTo if set (smooth movement)
    if (player.movingTo) {
      const dx = player.movingTo.x - player.x;
      const dy = player.movingTo.y - player.y;
      const dist = Math.hypot(dx, dy);
      if (dist < player.speed) {
        player.x = player.movingTo.x;
        player.y = player.movingTo.y;
        player.movingTo = null;
      } else {
        player.x += (dx / dist) * player.speed;
        player.y += (dy / dist) * player.speed;
      }
      // Clamp player position inside canvas and above Earth
      player.x = clamp(player.x, player.radius, GAME_WIDTH - player.radius);
      player.y = clamp(player.y, player.radius, EARTH_Y - player.radius - 10);
    }

    // Update enemies
    enemies.forEach(enemy => enemy.update(delta));

    // Update projectiles
    projectiles.forEach(proj => proj.update(delta));

    // Collision detection: projectiles vs enemies
    projectiles.forEach(proj => {
      if (proj.isDead) return;
      enemies.forEach(enemy => {
        if (enemy.isDead) return;
        const dist = distance(proj.x, proj.y, enemy.x, enemy.y);
        if (dist < proj.radius + enemy.radius) {
          // Hit detected
          enemy.health -= proj.damage;
          proj.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += enemy.scoreValue;
            resources += Math.floor(enemy.scoreValue / 2);
          }
        }
      });
    });

    // Remove dead enemies and projectiles
    enemies = enemies.filter(e => !e.isDead);
    projectiles = projectiles.filter(p => !p.isDead);

    // Check player shield health
    if (player.shield <= 0) {
      player.shield = 0;
      endGame();
    }

    // Spawn enemies based on timer
    enemySpawnTimer += delta;
    if (enemySpawnTimer >= enemySpawnInterval) {
      enemySpawnTimer = 0;
      spawnEnemy();
    }

    // Level progression: increase difficulty every 30 seconds
    if (time > level * 30000) {
      level++;
      levelEl.textContent = level;
      enemySpawnInterval = Math.max(500, ENEMY_SPAWN_INTERVAL_BASE - (level - 1) * 150);
      enemySpeed = ENEMY_SPEED_BASE + (level - 1) * 0.15;
      enemyHealth = ENEMY_HEALTH_BASE + (level - 1) * 5;
      tipEl.textContent = getRandomTip();
      // Change weather effect on level up
      changeWeatherEffect(level);
    }

    // Update UI
    scoreEl.textContent = score;
    healthEl.textContent = Math.floor(player.shield);
    energyEl.textContent = Math.floor(player.energy);

    // Update upgrade costs display
    costDamageEl.textContent = upgrades.damage.cost;
    costFireRateEl.textContent = upgrades.fireRate.cost;
    costSpeedEl.textContent = upgrades.speed.cost;
    costShieldEl.textContent = upgrades.shield.cost;

    // Enable/disable upgrade buttons based on resources
    upgradeDamageBtn.disabled = resources < upgrades.damage.cost;
    upgradeFireRateBtn.disabled = resources < upgrades.fireRate.cost;
    upgradeSpeedBtn.disabled = resources < upgrades.speed.cost;
    upgradeShieldBtn.disabled = resources < upgrades.shield.cost;
  }

    lastTime = timestamp;

  // Input handling for player movement and shooting
  function setupInput() {
    const container = document.getElementById('game-container');

    // Mouse and touch dragging for movement
    function pointerDown(e) {
      e.preventDefault();
      isDragging = true;
      container.classList.remove('grab');
      container.classList.add('grabbing');

      const rect = canvas.getBoundingClientRect();
      const pointerX = e.clientX ?? (e.touches ? e.touches[0].clientX : 0);
      const pointerY = e.clientY ?? (e.touches ? e.touches[0].clientY : 0);

      dragOffset.x = player.x - (pointerX - rect.left);
      dragOffset.y = player.y - (pointerY - rect.top);
    }
    function pointerMove(e) {
      if (!isDragging) return;
      e.preventDefault();

      const rect = canvas.getBoundingClientRect();
      const pointerX = e.clientX ?? (e.touches ? e.touches[0].clientX : 0);
      const pointerY = e.clientY ?? (e.touches ? e.touches[0].clientY : 0);

      let newX = pointerX - rect.left + dragOffset.x;
      let newY = pointerY - rect.top + dragOffset.y;

      // Clamp inside canvas and above Earth
      newX = clamp(newX, player.radius, GAME_WIDTH - player.radius);
      newY = clamp(newY, player.radius, EARTH_Y - player.radius - 10);

      player.x = newX;
      player.y = newY;
      player.movingTo = null; // cancel smooth movement if dragging
    }
    function pointerUp(e) {
      e.preventDefault();
      isDragging = false;
      container.classList.remove('grabbing');
      container.classList.add('grab');
    }

    canvas.addEventListener('mousedown', pointerDown);
    canvas.addEventListener('touchstart', pointerDown, { passive: false });

    window.addEventListener('mousemove', pointerMove);
    window.addEventListener('touchmove', pointerMove, { passive: false });

    window.addEventListener('mouseup', pointerUp);
    window.addEventListener('touchend', pointerUp);
    window.addEventListener('touchcancel', pointerUp);

    // Shooting on click or tap (if not dragging)
    canvas.addEventListener('click', e => {
      if (isDragging) return;
      const rect = canvas.getBoundingClientRect();
      const clickX = e.clientX - rect.left;
      const clickY = e.clientY - rect.top;
      playerShoot(clickX, clickY, performance.now());
    });

    // Keyboard controls: spacebar to shoot, arrow keys or WASD to move
    window.addEventListener('keydown', e => {
      if (!gameRunning) return;
      if (e.code === 'Space') {
        e.preventDefault();
        // Shoot straight upward from player center
        playerShoot(player.x, player.y - 100, performance.now());
      }
      // Movement keys move player smoothly
      const moveAmount = player.speed;
      switch (e.code) {
        case 'ArrowLeft':
        case 'KeyA':
          player.movingTo = { x: clamp(player.x - moveAmount * 5, player.radius, GAME_WIDTH - player.radius), y: player.y };
          break;
        case 'ArrowRight':
        case 'KeyD':
          player.movingTo = { x: clamp(player.x + moveAmount * 5, player.radius, GAME_WIDTH - player.radius), y: player.y };
          break;
        case 'ArrowUp':
        case 'KeyW':
          player.movingTo = { x: player.x, y: clamp(player.y - moveAmount * 5, player.radius, EARTH_Y - player.radius - 10) };
          break;
        case 'ArrowDown':
        case 'KeyS':
          player.movingTo = { x: player.x, y: clamp(player.y + moveAmount * 5, player.radius, EARTH_Y - player.radius - 10) };
          break;
      }
    });

  // Particle class for weather effects
  class WeatherParticle {
    constructor(x, y, vx, vy, size, color, life, type) {
      this.x = x;
      this.y = y;
      this.vx = vx;
      this.vy = vy;
      this.size = size;
      this.color = color;
      this.life = life;
      this.type = type; // 'meteor', 'flare', 'asteroid'
      this.age = 0;
    }
    update(delta) {
      this.x += this.vx * delta;
      this.y += this.vy * delta;
      this.age += delta;
      if (this.age > this.life) {
        this.dead = true;
      }
    }
    draw(ctx) {
      ctx.save();
      ctx.beginPath();
      if (this.type === 'meteor') {
        // Draw meteor with tail
        const tailLength = this.size * 4;
        const angle = Math.atan2(this.vy, this.vx);
        ctx.strokeStyle = this.color;
        ctx.lineWidth = this.size;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 15;
        ctx.moveTo(this.x, this.y);
        ctx.lineTo(this.x - Math.cos(angle) * tailLength, this.y - Math.sin(angle) * tailLength);
        ctx.stroke();
        ctx.beginPath();
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 20;
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
      } else if (this.type === 'flare') {
        // Solar flare glow
        const gradient = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.size * 3);
        gradient.addColorStop(0, this.color);
        gradient.addColorStop(1, 'transparent');
        ctx.fillStyle = gradient;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 30;
        ctx.arc(this.x, this.y, this.size * 3, 0, Math.PI * 2);
        ctx.fill();
      } else if (this.type === 'asteroid') {
        // Asteroid rock shape
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.moveTo(this.x + this.size * 0.5, this.y);
        ctx.lineTo(this.x, this.y + this.size * 0.7);
        ctx.lineTo(this.x - this.size * 0.6, this.y + this.size * 0.3);
        ctx.lineTo(this.x - this.size * 0.3, this.y - this.size * 0.5);
        ctx.lineTo(this.x + this.size * 0.4, this.y - this.size * 0.6);
        ctx.closePath();
        ctx.fill();
      }
      ctx.restore();
    }
  }

      if (weatherTimer > 300) {
        weatherTimer = 0;
        spawnAsteroid();

    weatherParticles = weatherParticles.filter(p => !p.dead);

  // Voice control system
  let recognition = null;
  let voiceActive = false;
  let voiceCommandCooldown = 0;

  function startVoiceRecognition() {
    if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
      voiceStatusText.textContent = 'Not supported';
      return;
    }
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();
    recognition.continuous = true;
    recognition.interimResults = false;
    recognition.lang = 'en-US';

    recognition.onstart = () => {
      voiceActive = true;
      voiceStatusText.textContent = 'Active';
    };
    recognition.onend = () => {
      voiceActive = false;
      voiceStatusText.textContent = 'Inactive';
      // Auto-restart recognition for continuous control
      if (gameRunning) {
        recognition.start();
      }
    };
    recognition.onerror = (event) => {
      // Ignore no-speech errors to keep recognition alive
      if (event.error !== 'no-speech') {
        voiceStatusText.textContent = 'Error: ' + event.error;
      }
    };
    recognition.onresult = (event) => {
      if (!gameRunning) return;
      const transcript = event.results[event.results.length - 1][0].transcript.trim().toLowerCase();
      processVoiceCommand(transcript);
    };

    recognition.start();
  }

  function stopVoiceRecognition() {
    if (recognition) {
      recognition.onresult = null;
      recognition.onend = null;
      recognition.onerror = null;
      recognition.stop();
      recognition = null;
      voiceActive = false;
      voiceStatusText.textContent = 'Inactive';
    }
  }

  // Process voice commands for movement and shooting
  function processVoiceCommand(command) {
    if (voiceCommandCooldown > performance.now()) return; // throttle commands

    // Movement commands
    if (command.includes('left')) {
      player.movingTo = { x: clamp(player.x - player.speed * 10, player.radius, GAME_WIDTH - player.radius), y: player.y };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('right')) {
      player.movingTo = { x: clamp(player.x + player.speed * 10, player.radius, GAME_WIDTH - player.radius), y: player.y };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('up')) {
      player.movingTo = { x: player.x, y: clamp(player.y - player.speed * 10, player.radius, EARTH_Y - player.radius - 10) };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('down')) {
      player.movingTo = { x: player.x, y: clamp(player.y + player.speed * 10, player.radius, EARTH_Y - player.radius - 10) };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('shoot') || command.includes('fire')) {
      // Shoot straight upward
      playerShoot(player.x, player.y - 100, performance.now());
      voiceCommandCooldown = performance.now() + 500;
    }
  }

      const y = earthY + Math.sin(angle) * earthRadius * 0.7;
      ctx.beginPath();
      ctx.arc(x, y, 6, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();
  }

  // Spawn a new enemy at random top position
  function spawnEnemy() {
    // Randomly select enemy type weighted by level
    const typeIndex = Math.min(level - 1, ENEMY_TYPES.length - 1);
    const type = ENEMY_TYPES[typeIndex];

    // Random x position within canvas width, with margin
    const margin = 30;
    const x = Math.random() * (GAME_WIDTH - margin * 2) + margin;
    const y = -type.radius - 10;

    // Speed and health scale with level and type
    const speed = enemySpeed * type.speedMultiplier;
    const health = enemyHealth * type.healthMultiplier;

    const enemy = new Enemy(type, x, y, speed, health, type.score);
    enemies.push(enemy);
  }

  // Handle player shooting
  function playerShoot(targetX, targetY, time) {
    if (time - player.lastShotTime < player.fireRate) return;
    if (player.energy < 10) return; // Require energy to shoot

    player.lastShotTime = time;
    player.energy -= 10;

    // Create projectile from player position toward target
    const proj = new Projectile(player.x, player.y - player.radius - 10, targetX, targetY, player.damage);
    projectiles.push(proj);
  }

  // Update all game objects
  function update(delta, time) {
    if (!gameRunning) return;

    // Regenerate energy
    player.energy += (ENERGY_REGEN_RATE * delta) / 1000;
    if (player.energy > player.maxEnergy) player.energy = player.maxEnergy;

    // Move player toward movingTo if set (smooth movement)
    if (player.movingTo) {
      const dx = player.movingTo.x - player.x;
      const dy = player.movingTo.y - player.y;
      const dist = Math.hypot(dx, dy);
      if (dist < player.speed) {
        player.x = player.movingTo.x;
        player.y = player.movingTo.y;
        player.movingTo = null;
      } else {
        player.x += (dx / dist) * player.speed;
        player.y += (dy / dist) * player.speed;
      }
      // Clamp player position inside canvas and above Earth
      player.x = clamp(player.x, player.radius, GAME_WIDTH - player.radius);
      player.y = clamp(player.y, player.radius, EARTH_Y - player.radius - 10);
    }

    // Update enemies
    enemies.forEach(enemy => enemy.update(delta));

    // Update projectiles
    projectiles.forEach(proj => proj.update(delta));

    // Collision detection: projectiles vs enemies
    projectiles.forEach(proj => {
      if (proj.isDead) return;
      enemies.forEach(enemy => {
        if (enemy.isDead) return;
        const dist = distance(proj.x, proj.y, enemy.x, enemy.y);
        if (dist < proj.radius + enemy.radius) {
          // Hit detected
          enemy.health -= proj.damage;
          proj.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += enemy.scoreValue;
            resources += Math.floor(enemy.scoreValue / 2);
          }
        }
      });
    });

    // Remove dead enemies and projectiles
    enemies = enemies.filter(e => !e.isDead);
    projectiles = projectiles.filter(p => !p.isDead);

    // Check player shield health
    if (player.shield <= 0) {
      player.shield = 0;
      endGame();
    }

    // Spawn enemies based on timer
    enemySpawnTimer += delta;
    if (enemySpawnTimer >= enemySpawnInterval) {
      enemySpawnTimer = 0;
      spawnEnemy();
    }

    // Level progression: increase difficulty every 30 seconds
    if (time > level * 30000) {
      level++;
      levelEl.textContent = level;
      enemySpawnInterval = Math.max(500, ENEMY_SPAWN_INTERVAL_BASE - (level - 1) * 150);
      enemySpeed = ENEMY_SPEED_BASE + (level - 1) * 0.15;
      enemyHealth = ENEMY_HEALTH_BASE + (level - 1) * 5;
      tipEl.textContent = getRandomTip();
      // Change weather effect on level up
      changeWeatherEffect(level);
    }

    // Update UI
    scoreEl.textContent = score;
    healthEl.textContent = Math.floor(player.shield);
    energyEl.textContent = Math.floor(player.energy);

    // Update upgrade costs display
    costDamageEl.textContent = upgrades.damage.cost;
    costFireRateEl.textContent = upgrades.fireRate.cost;
    costSpeedEl.textContent = upgrades.speed.cost;
    costShieldEl.textContent = upgrades.shield.cost;

    // Enable/disable upgrade buttons based on resources
    upgradeDamageBtn.disabled = resources < upgrades.damage.cost;
    upgradeFireRateBtn.disabled = resources < upgrades.fireRate.cost;
    upgradeSpeedBtn.disabled = resources < upgrades.speed.cost;
    upgradeShieldBtn.disabled = resources < upgrades.shield.cost;
  }

  // Input handling for player movement and shooting
  function setupInput() {
    const container = document.getElementById('game-container');

    // Mouse and touch dragging for movement
    function pointerDown(e) {
      e.preventDefault();
      isDragging = true;
      container.classList.remove('grab');
      container.classList.add('grabbing');

      const rect = canvas.getBoundingClientRect();
      const pointerX = e.clientX ?? (e.touches ? e.touches[0].clientX : 0);
      const pointerY = e.clientY ?? (e.touches ? e.touches[0].clientY : 0);

      dragOffset.x = player.x - (pointerX - rect.left);
      dragOffset.y = player.y - (pointerY - rect.top);
    }
    function pointerMove(e) {
      if (!isDragging) return;
      e.preventDefault();

      const rect = canvas.getBoundingClientRect();
      const pointerX = e.clientX ?? (e.touches ? e.touches[0].clientX : 0);
      const pointerY = e.clientY ?? (e.touches ? e.touches[0].clientY : 0);

      let newX = pointerX - rect.left + dragOffset.x;
      let newY = pointerY - rect.top + dragOffset.y;

      // Clamp inside canvas and above Earth
      newX = clamp(newX, player.radius, GAME_WIDTH - player.radius);
      newY = clamp(newY, player.radius, EARTH_Y - player.radius - 10);

      player.x = newX;
      player.y = newY;
      player.movingTo = null; // cancel smooth movement if dragging
    }
    function pointerUp(e) {
      e.preventDefault();
      isDragging = false;
      container.classList.remove('grabbing');
      container.classList.add('grab');
    }

    canvas.addEventListener('mousedown', pointerDown);
    canvas.addEventListener('touchstart', pointerDown, { passive: false });

    window.addEventListener('mousemove', pointerMove);
    window.addEventListener('touchmove', pointerMove, { passive: false });

    window.addEventListener('mouseup', pointerUp);
    window.addEventListener('touchend', pointerUp);
    window.addEventListener('touchcancel', pointerUp);

    // Shooting on click or tap (if not dragging)
    canvas.addEventListener('click', e => {
      if (isDragging) return;
      const rect = canvas.getBoundingClientRect();
      const clickX = e.clientX - rect.left;
      const clickY = e.clientY - rect.top;
      playerShoot(clickX, clickY, performance.now());
    });

    // Keyboard controls: spacebar to shoot, arrow keys or WASD to move
    window.addEventListener('keydown', e => {
      if (!gameRunning) return;
      if (e.code === 'Space') {
        e.preventDefault();
        // Shoot straight upward from player center
        playerShoot(player.x, player.y - 100, performance.now());
      }
      // Movement keys move player smoothly
      const moveAmount = player.speed;
      switch (e.code) {
        case 'ArrowLeft':
        case 'KeyA':
          player.movingTo = { x: clamp(player.x - moveAmount * 5, player.radius, GAME_WIDTH - player.radius), y: player.y };
          break;
        case 'ArrowRight':
        case 'KeyD':
          player.movingTo = { x: clamp(player.x + moveAmount * 5, player.radius, GAME_WIDTH - player.radius), y: player.y };
          break;
        case 'ArrowUp':
        case 'KeyW':
          player.movingTo = { x: player.x, y: clamp(player.y - moveAmount * 5, player.radius, EARTH_Y - player.radius - 10) };
          break;
        case 'ArrowDown':
        case 'KeyS':
          player.movingTo = { x: player.x, y: clamp(player.y + moveAmount * 5, player.radius, EARTH_Y - player.radius - 10) };

  // Particle class for weather effects
  class WeatherParticle {
    constructor(x, y, vx, vy, size, color, life, type) {
      this.x = x;
      this.y = y;
      this.vx = vx;
      this.vy = vy;
      this.size = size;
      this.color = color;
      this.life = life;
      this.type = type; // 'meteor', 'flare', 'asteroid'
      this.age = 0;
    }
    update(delta) {
      this.x += this.vx * delta;
      this.y += this.vy * delta;
      this.age += delta;
      if (this.age > this.life) {
        this.dead = true;
      }
    }
    draw(ctx) {
      ctx.save();
      ctx.beginPath();
      if (this.type === 'meteor') {
        // Draw meteor with tail
        const tailLength = this.size * 4;
        const angle = Math.atan2(this.vy, this.vx);
        ctx.strokeStyle = this.color;
        ctx.lineWidth = this.size;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 15;
        ctx.moveTo(this.x, this.y);
        ctx.lineTo(this.x - Math.cos(angle) * tailLength, this.y - Math.sin(angle) * tailLength);
        ctx.stroke();
        ctx.beginPath();
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 20;
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
      } else if (this.type === 'flare') {
        // Solar flare glow
        const gradient = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.size * 3);
        gradient.addColorStop(0, this.color);
        gradient.addColorStop(1, 'transparent');
        ctx.fillStyle = gradient;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 30;
        ctx.arc(this.x, this.y, this.size * 3, 0, Math.PI * 2);
        ctx.fill();
      } else if (this.type === 'asteroid') {
        // Asteroid rock shape
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 10;
        ctx.beginPath();
        ctx.moveTo(this.x + this.size * 0.5, this.y);
        ctx.lineTo(this.x, this.y + this.size * 0.7);
        ctx.lineTo(this.x - this.size * 0.6, this.y + this.size * 0.3);
        ctx.lineTo(this.x - this.size * 0.3, this.y - this.size * 0.5);
        ctx.lineTo(this.x + this.size * 0.4, this.y - this.size * 0.6);
        ctx.closePath();
        ctx.fill();
      }
      ctx.restore();
    }
  }

  function startVoiceRecognition() {
    if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
      voiceStatusText.textContent = 'Not supported';
      return;
    }
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();
    recognition.continuous = true;
    recognition.interimResults = false;
    recognition.lang = 'en-US';

    recognition.onstart = () => {
      voiceActive = true;
      voiceStatusText.textContent = 'Active';
    };
    recognition.onend = () => {
      voiceActive = false;
      voiceStatusText.textContent = 'Inactive';
      // Auto-restart recognition for continuous control
      if (gameRunning) {
        recognition.start();
      }
    };
    recognition.onerror = (event) => {
      // Ignore no-speech errors to keep recognition alive
      if (event.error !== 'no-speech') {
        voiceStatusText.textContent = 'Error: ' + event.error;
      }
    };
    recognition.onresult = (event) => {
      if (!gameRunning) return;
      const transcript = event.results[event.results.length - 1][0].transcript.trim().toLowerCase();
      processVoiceCommand(transcript);
    };

    recognition.start();
  }

  function stopVoiceRecognition() {
    if (recognition) {
      recognition.onresult = null;
      recognition.onend = null;
      recognition.onerror = null;
      recognition.stop();
      recognition = null;
      voiceActive = false;
      voiceStatusText.textContent = 'Inactive';
    }
  }

  // Process voice commands for movement and shooting
  function processVoiceCommand(command) {
    if (voiceCommandCooldown > performance.now()) return; // throttle commands

    // Movement commands
    if (command.includes('left')) {
      player.movingTo = { x: clamp(player.x - player.speed * 10, player.radius, GAME_WIDTH - player.radius), y: player.y };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('right')) {
      player.movingTo = { x: clamp(player.x + player.speed * 10, player.radius, GAME_WIDTH - player.radius), y: player.y };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('up')) {
      player.movingTo = { x: player.x, y: clamp(player.y - player.speed * 10, player.radius, EARTH_Y - player.radius - 10) };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('down')) {
      player.movingTo = { x: player.x, y: clamp(player.y + player.speed * 10, player.radius, EARTH_Y - player.radius - 10) };
      voiceCommandCooldown = performance.now() + 500;
    } else if (command.includes('shoot') || command.includes('fire')) {
      // Shoot straight upward
      playerShoot(player.x, player.y - 100, performance.now());
      voiceCommandCooldown = performance.now() + 500;
    }
  }
