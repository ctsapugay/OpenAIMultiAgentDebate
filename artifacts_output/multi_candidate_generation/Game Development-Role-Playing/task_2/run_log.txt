/**
 * Calculate total attribute bonuses from skills
 * Returns an object with attribute names and total bonuses
 */
function calculateSkillBonuses() {
  const bonuses = {
    Strength: 0,
    Agility: 0,
    Intelligence: 0,
    Vitality: 0,
  };
  for (const skill of skills) {
    if (skill.currentLevel > 0) {
      const effect = skill.applyEffect(skill.currentLevel);
      for (const attr in effect) {
        if (bonuses.hasOwnProperty(attr)) {
          bonuses[attr] += effect[attr];
        }
      }
    }
  }
  return bonuses;
}

/**
 * Update the character attributes display
 * Shows base attributes + bonuses from skills
 */
function updateAttributesDisplay() {
  const bonuses = calculateSkillBonuses();
  attributesContainer.innerHTML = '';
  for (const attr in character.attributes) {
    const base = character.attributes[attr];
    const bonus = bonuses[attr] || 0;
    const total = base + bonus;
    const attrDiv = document.createElement('div');
    attrDiv.className = 'attribute';
    attrDiv.innerHTML = `<span>${attr}:</span> ${total} <small style="color:#6f6;">(Base: ${base} + Bonus: ${bonus})</small>`;
    attributesContainer.appendChild(attrDiv);
  }
}

/**
 * Update the experience bar and text
 */
function updateExpDisplay() {
  const percent = Math.min(100, (character.currentExp / character.expToNextLevel) * 100);
  expBar.style.width = percent + '%';
  expText.textContent = `EXP: ${character.currentExp} / ${character.expToNextLevel}`;
  expBar.parentElement.setAttribute('aria-valuenow', Math.floor(percent));
}

/**
 * Level up the character if enough EXP
 * Grants skill points on level up
 */
function tryLevelUp() {
  while (character.currentExp >= character.expToNextLevel) {
    character.currentExp -= character.expToNextLevel;
    character.level++;
    character.skillPoints += 1; // 1 skill point per level
    // Increase exp needed for next level (e.g. 1.3x per level)
    character.expToNextLevel = Math.floor(character.expToNextLevel * 1.3);
    alert(`🎉 You leveled up! You are now level ${character.level} and gained 1 skill point.`);
  }
  updateExpDisplay();
  updateSkillPointsDisplay();
}

/**
 * Add EXP to the character
 * @param {number} amount 
 */
function gainExp(amount) {
  character.currentExp += amount;
  tryLevelUp();
}

/**
 * Check if a skill is unlocked (available to be leveled)
 * Conditions:
 * - All dependencies must be met (dependency skill at max level)
 * - Skill is not at max level
 */
function isSkillUnlocked(skill) {
  if (skill.currentLevel >= skill.maxLevel) return false;
  for (const depId of skill.dependencies) {
    const depSkill = skills.find(s => s.id === depId);
    if (!depSkill || depSkill.currentLevel < depSkill.maxLevel) {
      return false;
    }
  }
  return true;
}

/**
 * Render the skill tree UI
 */
function renderSkillTree() {
  skillTreeContainer.innerHTML = '';
  for (const skill of skills) {
    const skillDiv = document.createElement('div');
    skillDiv.className = 'skill-node';
    skillDiv.setAttribute('role', 'listitem');
    skillDiv.setAttribute('tabindex', '0');
    skillDiv.dataset.skillId = skill.id;

    // Determine skill state
    if (skill.currentLevel === 0 && !isSkillUnlocked(skill)) {
      skillDiv.classList.add('locked');
    } else if (skill.currentLevel === skill.maxLevel) {
      skillDiv.classList.add('enhanced');
    } else if (skill.currentLevel > 0) {
      skillDiv.classList.add('unlocked');
    } else {
      skillDiv.classList.add('unlocked');
    }

    // Skill title and level
    const title = document.createElement('h3');
    title.textContent = skill.name;
    skillDiv.appendChild(title);

    // Skill description
    const desc = document.createElement('p');
    desc.textContent = skill.description;
    skillDiv.appendChild(desc);

    // Skill level display
    const levelSpan = document.createElement('div');
    levelSpan.className = 'level';
    levelSpan.textContent = `Level: ${skill.currentLevel} / ${skill.maxLevel}`;
    skillDiv.appendChild(levelSpan);

    // Skill cost display
    const costSpan = document.createElement('div');
    costSpan.className = 'cost';
    costSpan.textContent = `Cost: ${skill.cost} SP`;
    skillDiv.appendChild(costSpan);

    // Tooltip for dependencies
    if (skill.dependencies.length > 0) {
      const tooltip = document.createElement('div');
      tooltip.className = 'tooltip';
      const depNames = skill.dependencies.map(depId => {
        const depSkill = skills.find(s => s.id === depId);
        return depSkill ? depSkill.name : depId;
      });
      tooltip.textContent = `Requires: ${depNames.join(', ')}`;
      skillDiv.appendChild(tooltip);
    }

    // Click handler to unlock or enhance skill
    skillDiv.addEventListener('click', () => {
      handleSkillClick(skill.id);
    });
    skillDiv.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        handleSkillClick(skill.id);
      }
    });

    skillTreeContainer.appendChild(skillDiv);
  }
}

/**
 * Handle skill node click to unlock or enhance skill
 * @param {string} skillId 
 */
function handleSkillClick(skillId) {
  const skill = skills.find(s => s.id === skillId);
  if (!skill) return;

  // Check if skill can be leveled up
  if (!isSkillUnlocked(skill)) {
    alert('This skill is locked. Fulfill dependencies first.');
    return;
  }
  if (skill.currentLevel >= skill.maxLevel) {
    alert('Skill is already at max level.');
    return;
  }
  if (character.skillPoints < skill.cost) {
    alert('Not enough skill points.');
    return;
  }

  // Deduct skill points and level up skill
  character.skillPoints -= skill.cost;
  skill.currentLevel++;
  updateSkillPointsDisplay();
  updateAttributesDisplay();
  renderSkillTree();
}

/**
 * Update skill points display in the character panel
 */
function updateSkillPointsDisplay() {
  // Show skill points below attributes
  let spDisplay = document.getElementById('skill-points-display');
  if (!spDisplay) {
    spDisplay = document.createElement('div');
    spDisplay.id = 'skill-points-display';
    spDisplay.style.marginBottom = '20px';
    spDisplay.style.fontWeight = '700';
    spDisplay.style.fontSize = '1.1rem';
    attributesContainer.parentElement.insertBefore(spDisplay, expBar.parentElement);
  }
  spDisplay.textContent = `Skill Points: ${character.skillPoints}`;
}

/**
 * Initialize the UI and event listeners
 */
function init() {
  updateAttributesDisplay();
  updateExpDisplay();
  updateSkillPointsDisplay();
  renderSkillTree();

  gainExpBtn.addEventListener('click', () => {
    gainExp(25);
  });
}

function calculateSkillBonuses() {
  const bonuses = {

    Agility: 0,

    Vitality: 0,

  for (const skill of skills) {

      const effect = skill.applyEffect(skill.currentLevel);
      for (const attr in effect) {
        if (bonuses.hasOwnProperty(attr)) {
          bonuses[attr] += effect[attr];
        }
      }
    }
  }
  return bonuses;
}

/**
 * Check if a skill is unlocked (available to be leveled)
 * Conditions:
 * - All dependencies must be met (dependency skill at max level)
 * - Skill is not at max level
 */
function isSkillUnlocked(skill) {
  if (skill.currentLevel >= skill.maxLevel) return false;
  for (const depId of skill.dependencies) {
    const depSkill = skills.find(s => s.id === depId);
    if (!depSkill || depSkill.currentLevel < depSkill.maxLevel) {
      return false;
    }
  }
  return true;
}

    return;

/**

 * Returns an object with attribute names and total bonuses

function calculateSkillBonuses() {
  const bonuses = {
    Strength: 0,
    Agility: 0,
    Intelligence: 0,
    Vitality: 0,
  };
  // Add skill effects
  for (const skill of skills) {
    if (skill.currentLevel > 0) {
      const effect = skill.applyEffect(skill.currentLevel);
      for (const attr in effect) {
        if (bonuses.hasOwnProperty(attr)) {
          bonuses[attr] += effect[attr];
        }
      }
    }
  }
  // Add combo effects if combos are active
  for (const combo of skillCombos) {
    if (isComboActive(combo)) {
      const effect = combo.applyEffect();
      for (const attr in effect) {

        }
      }
    }
  }
  return bonuses;
}

/**

/**

  let leveledUp = false;

    character.currentExp -= character.expToNextLevel;
    character.level++;
    character.skillPoints += 1; // 1 skill point per level

/**

 * Conditions:
 * - All dependencies must be met (dependency skill at max level)
 * - Skill is not at max level
 */
function isSkillUnlocked(skill) {
  if (skill.currentLevel >= skill.maxLevel) return false;
  for (const depId of skill.dependencies) {
    const depSkill = skills.find(s => s.id === depId);
    if (!depSkill || depSkill.currentLevel < depSkill.maxLevel) {
      return false;
    }
  }
  return true;
}

/**
 * Get skill preview text for next level effect
 * @param {object} skill 
 * @returns {string}
 */
function getSkillNextLevelPreview(skill) {
  const nextLevel = skill.currentLevel + 1;
  if (nextLevel > skill.maxLevel) return 'Max level reached';
  const currentEffect = skill.applyEffect(skill.currentLevel);
  const nextEffect = skill.applyEffect(nextLevel);
  const diffs = [];
  for (const attr in nextEffect) {
    const diff = nextEffect[attr] - (currentEffect[attr] || 0);
    if (diff > 0) {
      diffs.push(`+${diff} ${attr}`);
    }
  }
  return `Next Level: ${diffs.join(', ')}`;
}

/**

 * @param {string} skillId

function handleSkillClick(skillId) {
  const skill = skills.find(s => s.id === skillId);
  if (!skill) return;

  // Check if skill can be leveled up
  if (!isSkillUnlocked(skill)) {
    alert('This skill is locked. Fulfill dependencies first.');
    return;

    alert('Skill is already at max level.');
    return;

  if (character.skillPoints < skill.cost) {
    alert('Not enough skill points.');
    return;
  }

  // Deduct skill points and level up skill
  character.skillPoints -= skill.cost;
  skill.currentLevel++;
  updateSkillPointsDisplay();
  updateAttributesDisplay();
  renderSkillTree();
}

  respecBtn.disabled = character.skillPoints < 2 || !hasAnySkillPointsSpent();
}

/**
 * Check if any skill points have been spent (any skill currentLevel > 0)
 * @returns {boolean}
 */
function hasAnySkillPointsSpent() {
  return skills.some(skill => skill.currentLevel > 0);
}

/**
 * Check if a skill is part of any active combo
 * @param {string} skillId 
 * @returns {boolean}
 */
function isSkillInActiveCombo(skillId) {
  return skillCombos.some(combo => {
    return combo.skills.includes(skillId) && isComboActive(combo);
  });
}

/**
 * Respec all skills: reset all skill levels and refund skill points minus cost
 */
function respecSkills() {
  if (character.skillPoints < 2) {
    alert('Not enough skill points to respec (cost: 2 SP).');
    return;
  }
  if (!confirm('Are you sure you want to reset all skills? This will cost 2 skill points.')) {
    return;
  }
  // Deduct respec cost
  character.skillPoints -= 2;
  // Refund all spent skill points from skills
  let refundedPoints = 0;
  for (const skill of skills) {
    refundedPoints += skill.currentLevel * skill.cost;
    skill.currentLevel = 0;
  }
  character.skillPoints += refundedPoints;
  updateSkillPointsDisplay();
  updateAttributesDisplay();
  renderSkillTree();
  showLevelUpNotification('🔄 Skills have been reset. Spend your points wisely!');
}

/**
 * Initialize the UI and event listeners
 */
function init() {
  updateAttributesDisplay();
  updateExpDisplay();
  updateSkillPointsDisplay();
  renderSkillTree();

  gainExpBtn.addEventListener('click', () => {
    gainExp(25);
  });

  respecBtn.addEventListener('click', () => {
    respecSkills();
  });
}

init();
