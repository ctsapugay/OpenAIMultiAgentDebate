let currentNpc = null;

const npcNameEl = document.getElementById('npc-name');

/**
 * Initialize the NPC list UI.
 */
function initNpcList() {
  npcListEl.innerHTML = '';
  npcs.forEach((npc, index) => {
    const li = document.createElement('li');
    li.textContent = npc.name;
    li.setAttribute('role', 'button');

    li.addEventListener('keydown', e => {

      }
    });
    npcListEl.appendChild(li);
  });

/**
 * Select an NPC by id and start dialogue.
 * @param {string} npcId 
 */
function selectNpc(npcId) {
  if (currentNpc && currentNpc.id === npcId) return; // Already selected

  currentNpc = npcs.find(n => n.id === npcId);
  currentNodeKey = 'start';

  // Highlight selected NPC in list
  Array.from(npcListEl.children).forEach(li => {
    li.classList.toggle('active', li.dataset.npcId === npcId);
  });

  renderDialogue();
}

/**
 * Render the current dialogue node for the selected NPC.
 */
function renderDialogue() {
  if (!currentNpc || !currentNodeKey) {
    npcNameEl.textContent = '';
    dialogueTextEl.textContent = 'Select an NPC to start conversation.';
    choicesEl.innerHTML = '';
    return;
  }

  const node = currentNpc.dialogueTree[currentNodeKey];
  if (!node) {
    dialogueTextEl.textContent = '...';
    choicesEl.innerHTML = '';
    return;
  }

  // Update NPC name
  npcNameEl.textContent = currentNpc.name;

  // Special handling for quest completion dialogue
  if (currentNodeKey === 'start' && currentNpc.quests) {
    // Check if any quest is active and completed by player
    for (const questId in currentNpc.quests) {
      const quest = currentNpc.quests[questId];
      if (playerState.quests[questId]?.status === 'active' && quest.completed) {
        // Switch dialogue to questComplete if defined
        if (currentNpc.dialogueTree['questComplete']) {
          currentNodeKey = 'questComplete';
          renderDialogue();
          return;
        }
      }
    }
  }

  // Show dialogue text
  dialogueTextEl.textContent = node.text;

  // Clear previous choices
  choicesEl.innerHTML = '';

  // Create buttons for choices
  node.choices.forEach((choice, idx) => {
    const btn = document.createElement('button');
    btn.textContent = choice.text;
    btn.type = 'button';
    btn.setAttribute('aria-label', `Choice: ${choice.text}`);
    btn.addEventListener('click', () => handleChoice(choice));
    choicesEl.appendChild(btn);
  });

  // Focus first choice button for accessibility
  if (choicesEl.firstChild) {
    choicesEl.firstChild.focus();
  }
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  // Handle special actions
  if (choice.action) {
    switch (choice.action) {
      case 'acceptQuest':
        acceptQuest(choice.questId);
        break;
      case 'completeQuest':
        completeQuest(choice.questId);
        break;
      case 'addClue':
        addClue(choice.clue);
        break;
      default:
        break;
    }
  }

  // Move to next dialogue node or end conversation
  if (choice.next) {
    currentNodeKey = choice.next;
    renderDialogue();
  } else {
    // End conversation: clear dialogue and choices
    npcNameEl.textContent = currentNpc.name;
    dialogueTextEl.textContent = "Conversation ended. Select another NPC or start again.";
    choicesEl.innerHTML = '';
  }
}

/**
 * Accept a quest and add it to player state.
 * @param {string} questId 
 */
function acceptQuest(questId) {
  if (!currentNpc.quests[questId]) return;
  if (playerState.quests[questId]) return; // Already accepted

  playerState.quests[questId] = {
    title: currentNpc.quests[questId].title,
    status: 'active',
  };
  updatePlayerStatus();
}

/**
 * Complete a quest and update player state.
 * @param {string} questId 
 */
function completeQuest(questId) {
  if (!playerState.quests[questId]) return;
  if (playerState.quests[questId].status === 'completed') return;

  playerState.quests[questId].status = 'completed';

  // Mark quest as completed in NPC data (simulate game progression)
  if (currentNpc.quests[questId]) {
    currentNpc.quests[questId].completed = true;
  }

  updatePlayerStatus();
}

/**
 * Add a clue to player state.
 * @param {string} clue 
 */
function addClue(clue) {
  if (!clue) return;
  playerState.clues.add(clue);
  updatePlayerStatus();
}

/**
 * Update the player status panel UI.
 */
function updatePlayerStatus() {
  // Update quests list
  questListEl.innerHTML = '';
  const quests = Object.entries(playerState.quests);
  if (quests.length === 0) {
    const li = document.createElement('li');
    li.textContent = 'No active quests.';
    questListEl.appendChild(li);
  } else {
    quests.forEach(([id, quest]) => {
      const li = document.createElement('li');
      li.textContent = `${quest.title} [${quest.status}]`;
      questListEl.appendChild(li);
    });
  }

  // Update clues list
  clueListEl.innerHTML = '';
  if (playerState.clues.size === 0) {
    const li = document.createElement('li');
    li.textContent = 'No clues obtained.';
    clueListEl.appendChild(li);
  } else {
    playerState.clues.forEach(clue => {
      const li = document.createElement('li');
      li.textContent = clue;
      clueListEl.appendChild(li);
    });
  }
}

    currentNpc.quests['hunt_beast'].completed = true;
    renderDialogue();
    updatePlayerStatus();
  }
}

function selectNpc(npcId) {

/**

function renderDialogue() {
  if (!currentNpc || !currentNodeKey) {
    npcNameEl.textContent = '';
    dialogueTextEl.textContent = 'Select an NPC to start conversation.';
    choicesEl.innerHTML = '';
    return;
  }

  const node = currentNpc.dialogueTree[currentNodeKey];
  if (!node) {
    dialogueTextEl.textContent = '...';
    choicesEl.innerHTML = '';
    return;
  }

  // Update NPC name
  npcNameEl.textContent = currentNpc.name;

  // Special handling for quest completion dialogue
  if (currentNodeKey === 'start' && currentNpc.quests) {
    // Check if any quest is active and completed by player
    for (const questId in currentNpc.quests) {
      const quest = currentNpc.quests[questId];
      if (playerState.quests[questId]?.status === 'active' && quest.completed) {
        // Switch dialogue to questComplete if defined
        if (currentNpc.dialogueTree['questComplete']) {
          currentNodeKey = 'questComplete';
          renderDialogue();
          return;
        }
      }
    }
  }

  // Show dialogue text
  dialogueTextEl.textContent = node.text;

  // Clear previous choices
  choicesEl.innerHTML = '';

  // Create buttons for choices
  node.choices.forEach((choice, idx) => {
    const btn = document.createElement('button');
    btn.textContent = choice.text;
    btn.type = 'button';
    btn.setAttribute('aria-label', `Choice: ${choice.text}`);
    btn.addEventListener('click', () => handleChoice(choice));
    choicesEl.appendChild(btn);
  });

  // Focus first choice button for accessibility
  if (choicesEl.firstChild) {
    choicesEl.firstChild.focus();
  }
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  // Handle special actions
  if (choice.action) {
    switch (choice.action) {
      case 'acceptQuest':
        acceptQuest(choice.questId);
        break;
      case 'completeQuest':
        completeQuest(choice.questId);
        break;
      case 'addClue':
        addClue(choice.clue);
        break;
      default:
        break;
    }
  }

  // Move to next dialogue node or end conversation
  if (choice.next) {
    currentNodeKey = choice.next;
    renderDialogue();
  } else {
    // End conversation: clear dialogue and choices
    npcNameEl.textContent = currentNpc.name;
    dialogueTextEl.textContent = "Conversation ended. Select another NPC or start again.";
    choicesEl.innerHTML = '';
  }
}

/**
 * Accept a quest and add it to player state.
 * @param {string} questId 
 */
function acceptQuest(questId) {
  if (!currentNpc.quests[questId]) return;
  if (playerState.quests[questId]) return; // Already accepted

  playerState.quests[questId] = {
    title: currentNpc.quests[questId].title,
    status: 'active',
  };
  updatePlayerStatus();
}

/**
 * Complete a quest and update player state.
 * @param {string} questId 
 */
function completeQuest(questId) {
  if (!playerState.quests[questId]) return;
  if (playerState.quests[questId].status === 'completed') return;

  playerState.quests[questId].status = 'completed';

  // Mark quest as completed in NPC data (simulate game progression)
  if (currentNpc.quests[questId]) {
    currentNpc.quests[questId].completed = true;
  }

  updatePlayerStatus();
}

 * @param {string} clue

}

/**
 * Select an NPC by id and start dialogue.
 * @param {string} npcId 
 */
function selectNpc(npcId) {
  if (currentNpc && currentNpc.id === npcId) return; // Already selected

  currentNpc = npcs.find(n => n.id === npcId);
  currentNodeKey = 'start';

  // Highlight selected NPC in list
  Array.from(npcListEl.children).forEach(li => {
    li.classList.toggle('active', li.dataset.npcId === npcId);
  });

  renderDialogue();
}

/**
 * Render the current dialogue node for the selected NPC.
 */
function renderDialogue() {
  if (!currentNpc || !currentNodeKey) {
    npcNameTextEl.textContent = '';
    dialogueTextEl.textContent = 'Select an NPC to start conversation.';
    choicesEl.innerHTML = '';
    updateNpcFace('neutral');
    return;
  }

  const node = currentNpc.dialogueTree[currentNodeKey];
  if (!node) {
    dialogueTextEl.textContent = '...';
    choicesEl.innerHTML = '';
    updateNpcFace(currentNpc.mood || 'neutral');
    return;
  }

  // Update NPC name text
  npcNameTextEl.textContent = currentNpc.name;

  // Update NPC face expression based on current mood
  updateNpcFace(currentNpc.mood || 'neutral');

  // Special handling for quest completion dialogue
  if (currentNodeKey === 'start' && currentNpc.quests) {
    // Check if any quest is active and completed by player
    for (const questId in currentNpc.quests) {
      const quest = currentNpc.quests[questId];
      if (playerState.quests[questId]?.status === 'active' && quest.completed) {
        // Switch dialogue to questComplete if defined
        if (currentNpc.dialogueTree['questComplete']) {
          currentNodeKey = 'questComplete';
          renderDialogue();
          return;
        }
      }
    }
  }

  // Show dialogue text
  dialogueTextEl.textContent = node.text;

  // Clear previous choices
  choicesEl.innerHTML = '';

  // Create buttons for choices
  node.choices.forEach((choice) => {
    const btn = document.createElement('button');
    btn.textContent = choice.text;
    btn.type = 'button';
    btn.setAttribute('aria-label', `Choice: ${choice.text}`);
    btn.addEventListener('click', () => handleChoice(choice));
    choicesEl.appendChild(btn);
  });

  // Focus first choice button for accessibility
  if (choicesEl.firstChild) {
    choicesEl.firstChild.focus();
  }
}

/**
 * Handle a player's choice selection.
 * @param {object} choice 
 */
function handleChoice(choice) {
  // Handle special actions
  if (choice.action) {
    switch (choice.action) {
      case 'acceptQuest':
        acceptQuest(choice.questId);
        break;
      case 'completeQuest':
        completeQuest(choice.questId);
        break;
      case 'addClue':
        addClue(choice.clue);
        break;
      default:
        break;
    }
  }

  // Move to next dialogue node or end conversation
  if (choice.next) {
    currentNodeKey = choice.next;
    renderDialogue();
  } else {
    // End conversation: clear dialogue and choices
    npcNameTextEl.textContent = currentNpc.name;
    dialogueTextEl.textContent = "Conversation ended. Select another NPC or start again.";
    choicesEl.innerHTML = '';
  }
}

/**
 * Accept a quest and add it to player state.
 * Also updates NPC mood positively.
 * @param {string} questId 
 */
function acceptQuest(questId) {
  if (!currentNpc.quests[questId]) return;
  if (playerState.quests[questId]) return; // Already accepted

  playerState.quests[questId] = {
    title: currentNpc.quests[questId].title,
    status: 'active',
  };
  updatePlayerStatus();

  // Improve NPC mood on quest acceptance
  changeNpcMood(currentNpc, 'happy');
}

/**
 * Complete a quest and update player state.
 * Also updates NPC mood positively.
 * @param {string} questId 
 */
function completeQuest(questId) {
  if (!playerState.quests[questId]) return;
  if (playerState.quests[questId].status === 'completed') return;

  playerState.quests[questId].status = 'completed';

  // Mark quest as completed in NPC data (simulate game progression)
  if (currentNpc.quests[questId]) {
    currentNpc.quests[questId].completed = true;
  }

  updatePlayerStatus();

  // Improve NPC mood on quest completion
  changeNpcMood(currentNpc, 'happy');
}

/**
 * Add a clue to player state.
 * Also updates NPC mood slightly positively.
 * @param {string} clue 
 */
function addClue(clue) {
  if (!clue) return;
  if (playerState.clues.has(clue)) return; // Avoid duplicates
  playerState.clues.add(clue);
  updatePlayerStatus();

  // Slightly improve NPC mood on clue given
  changeNpcMood(currentNpc, 'happy');
}

  }

  // If current NPC is this NPC, update face expression
  if (currentNpc && currentNpc.id === npc.id) {
    updateNpcFace(newMood);
  }
}

/**
 * Update the player status panel UI.
 */
function updatePlayerStatus() {
  // Update quests list
  questListEl.innerHTML = '';
  const quests = Object.entries(playerState.quests);
  if (quests.length === 0) {
    const li = document.createElement('li');
    li.textContent = 'No active quests.';
    questListEl.appendChild(li);
  } else {
    quests.forEach(([id, quest]) => {
      const li = document.createElement('li');
      li.textContent = `${quest.title} [${quest.status}]`;
      questListEl.appendChild(li);
    });
  }

  // Update clues list
  clueListEl.innerHTML = '';
  if (playerState.clues.size === 0) {
    const li = document.createElement('li');
    li.textContent = 'No clues obtained.';
    clueListEl.appendChild(li);
  } else {
    playerState.clues.forEach(clue => {
      const li = document.createElement('li');
      li.textContent = clue;
      clueListEl.appendChild(li);
    });
  }
}

/**
 * Simulate quest progression for demonstration.
 * This function can be triggered to mark the beast quest as completed.
 * Also updates NPC mood accordingly.
 */
function simulateQuestProgression() {
  // If player has accepted the hunt_beast quest and it's active, mark it completed
  if (playerState.quests['hunt_beast'] && playerState.quests['hunt_beast'].status === 'active') {
    currentNpc = npcs.find(n => n.id === 'elder_mira');
    currentNodeKey = 'questComplete';
    currentNpc.quests['hunt_beast'].completed = true;
    changeNpcMood(currentNpc, 'happy');
    renderDialogue();
    updatePlayerStatus();
  }
}
