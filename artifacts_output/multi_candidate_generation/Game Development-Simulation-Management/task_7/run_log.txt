// Game configuration constants

const CREATURE_SIZE = 50; // px width and height

// Utility function to create SVG element for a creature
function createCreatureSVG(type) {
  const xmlns = "http://www.w3.org/2000/svg";
  const svg = document.createElementNS(xmlns, "svg");
  svg.setAttribute("width", CREATURE_SIZE);
  svg.setAttribute("height", CREATURE_SIZE);
  svg.setAttribute("viewBox", "0 0 30 30");
  svg.setAttribute("aria-label", type.name);
  svg.setAttribute("role", "img");
  svg.style.userSelect = "none";

  // Background circle for ladybug or base shape
  if (type.id === 'ladybug') {
    const circle = document.createElementNS(xmlns, "circle");
    circle.setAttribute("cx", "15");
    circle.setAttribute("cy", "15");
    circle.setAttribute("r", "14");
    circle.setAttribute("fill", type.color);
    circle.setAttribute("stroke", "#880e4f");
    circle.setAttribute("stroke-width", "2");
    svg.appendChild(circle);

    // Black spots
    for (let i = 0; i < 5; i++) {
      const spot = document.createElementNS(xmlns, "circle");
      const angle = (i * 72) * Math.PI / 180;
      const r = 6;
      const cx = 15 + Math.cos(angle) * 7;
      const cy = 15 + Math.sin(angle) * 7;
      spot.setAttribute("cx", cx);
      spot.setAttribute("cy", cy);
      spot.setAttribute("r", r);
      spot.setAttribute("fill", "#000");
      svg.appendChild(spot);
    }
  } else {
    // For other creatures, create a path with fill color
    const path = document.createElementNS(xmlns, "path");
    path.setAttribute("d", type.svgPath);
    path.setAttribute("fill", type.color);
    path.setAttribute("stroke", "#333");
    path.setAttribute("stroke-width", "1");
    svg.appendChild(path);
  }

  return svg;
}

// Creature class to manage individual creatures
class Creature {
  constructor(type, x, y) {
    this.type = type;
    this.x = x; // top-left x in px relative to game area
    this.y = y; // top-left y in px relative to game area
    this.speedX = (Math.random() * 0.6 + 0.2) * (Math.random() < 0.5 ? -1 : 1); // horizontal speed px/frame
    this.speedY = (Math.random() * 0.4 + 0.1) * (Math.random() < 0.5 ? -1 : 1); // vertical speed px/frame
    this.element = document.createElement('div');
    this.element.classList.add('creature');
    this.element.style.width = CREATURE_SIZE + 'px';
    this.element.style.height = CREATURE_SIZE + 'px';
    this.element.style.left = this.x + 'px';
    this.element.style.top = this.y + 'px';
    this.element.setAttribute('tabindex', '-1');
    this.element.setAttribute('aria-label', this.type.name + ' creature');
    this.element.setAttribute('role', 'img');
    this.element.style.position = 'absolute';
    this.element.style.userSelect = 'none';

    // Append SVG inside creature div
    const svg = createCreatureSVG(type);
    this.element.appendChild(svg);

    // Tooltip
    const tooltip = document.createElement('div');
    tooltip.classList.add('tooltip');
    tooltip.textContent = type.name;
    this.element.appendChild(tooltip);

    // Add to game area
    gameArea.appendChild(this.element);
  }

  // Update position based on speed and bounce off edges
  update() {
    this.x += this.speedX;
    this.y += this.speedY;

    // Bounce horizontally
    if (this.x <= 0) {
      this.x = 0;
      this.speedX = -this.speedX;
    } else if (this.x + CREATURE_SIZE >= gameArea.clientWidth) {
      this.x = gameArea.clientWidth - CREATURE_SIZE;
      this.speedX = -this.speedX;
    }
    // Bounce vertically
    if (this.y <= 0) {
      this.y = 0;
      this.speedY = -this.speedY;
    } else if (this.y + CREATURE_SIZE >= gameArea.clientHeight) {
      this.y = gameArea.clientHeight - CREATURE_SIZE;
      this.speedY = -this.speedY;
    }

    this.element.style.left = this.x + 'px';
    this.element.style.top = this.y + 'px';
  }

  // Get center coordinates of creature for collision detection
  getCenter() {
    return {
      x: this.x + CREATURE_SIZE / 2,
      y: this.y + CREATURE_SIZE / 2
    };
  }

  // Remove creature from DOM
  remove() {
    if (this.element.parentNode) {
      this.element.parentNode.removeChild(this.element);
    }
  }
}

// Spawn a new creature at a random position inside the game area
function spawnCreature() {
  if (creatures.length >= MAX_CREATURES) return;

  const type = CREATURE_TYPES[Math.floor(Math.random() * CREATURE_TYPES.length)];
  const maxX = gameArea.clientWidth - CREATURE_SIZE;
  const maxY = gameArea.clientHeight - CREATURE_SIZE;
  const x = Math.random() * maxX;
  const y = Math.random() * maxY;

  const creature = new Creature(type, x, y);
  creatures.push(creature);
}

// Update all creatures positions
function updateCreatures() {
  creatures.forEach(creature => creature.update());
}

// Capture creatures inside net radius at given coordinates
function captureCreaturesAt(x, y) {
  // Find creatures within NET_RADIUS of (x,y)
  const captured = [];
  creatures.forEach(creature => {
    const center = creature.getCenter();
    const dx = center.x - x;
    const dy = center.y - y;
    const distance = Math.sqrt(dx * dx + dy * dy);
    if (distance <= NET_RADIUS) {
      captured.push(creature);
    }
  });

  if (captured.length === 0) {
    return false; // no capture
  }

  // Remove captured creatures from game and add to collection
  captured.forEach(creature => {
    creature.remove();
    creatures = creatures.filter(c => c !== creature);
    addToCollection(creature.type.id);
  });

  return true;
}

// Add a creature type to the player's collection
function addToCollection(creatureId) {
  if (!collection[creatureId]) {
    collection[creatureId] = 0;
  }
  collection[creatureId]++;
  renderCollection();
}

// Render the collection panel
function renderCollection() {
  collectionList.innerHTML = '';
  // Sort by name alphabetically
  const sortedTypes = CREATURE_TYPES.slice().sort((a, b) => a.name.localeCompare(b.name));
  sortedTypes.forEach(type => {
    if (collection[type.id]) {
      const count = collection[type.id];
      const item = document.createElement('div');
      item.classList.add('collected-item');
      item.setAttribute('aria-label', `${type.name} x${count}`);

      // Create SVG icon for collection item (smaller)
      const svg = createCreatureSVG(type);
      svg.setAttribute('width', '40');
      svg.setAttribute('height', '40');
      item.appendChild(svg);

      // Count badge
      const countBadge = document.createElement('div');
      countBadge.classList.add('collected-count');
      countBadge.textContent = count;
      item.appendChild(countBadge);

      collectionList.appendChild(item);
    }
  });
}

// Handle net throw button toggle
function toggleNetMode() {
  netMode = !netMode;
  throwNetBtn.setAttribute('aria-pressed', netMode.toString());
  throwNetBtn.textContent = netMode ? 'Cancel Net' : 'Throw Net';
  if (!netMode) {
    netVisual.style.display = 'none';
  }
}

// Handle game area click when in net mode
function onGameAreaClick(event) {
  if (!netMode) return;

  const now = Date.now();
  if (now - lastNetThrowTime < NET_THROW_COOLDOWN) {
    // Cooldown active, ignore throw
    return;
  }
  lastNetThrowTime = now;

  // Get click coordinates relative to game area
  const rect = gameArea.getBoundingClientRect();
  const clickX = event.clientX - rect.left;
  const clickY = event.clientY - rect.top;

  // Show net visual at click position
  netVisual.style.left = clickX + 'px';
  netVisual.style.top = clickY + 'px';
  netVisual.style.display = 'block';

  // Animate net throw (scale up and fade out)
  netVisual.style.transition = 'none';
  netVisual.style.transform = 'translate(-50%, -50%) scale(0.5)';
  netVisual.style.opacity = '0.8';
  requestAnimationFrame(() => {
    netVisual.style.transition = 'transform 0.4s ease-out, opacity 0.4s ease-out';
    netVisual.style.transform = 'translate(-50%, -50%) scale(1.2)';
    netVisual.style.opacity = '0';
  });

  // After animation, hide net visual
  setTimeout(() => {
    netVisual.style.display = 'none';
    netVisual.style.opacity = '0.8';
    netVisual.style.transform = 'translate(-50%, -50%) scale(1)';
  }, 400);

  // Check for captures
  const captured = captureCreaturesAt(clickX, clickY);

  // Provide feedback
  if (captured) {
    // Flash game area green briefly
    flashGameArea('#a0f7a0');
  } else {
    // Flash game area red briefly
    flashGameArea('#f7a0a0');
  }

  // Exit net mode after throw
  toggleNetMode();
}

// Flash the game area background color briefly for feedback
function flashGameArea(color) {
  const original = gameArea.style.backgroundColor || '#d0f0ff';
  gameArea.style.backgroundColor = color;
  setTimeout(() => {
    gameArea.style.backgroundColor = original;
  }, 300);
}

// Game loop for updating creatures
function gameLoop() {
  updateCreatures();
  requestAnimationFrame(gameLoop);
}

// Spawn creatures periodically
setInterval(spawnCreature, SPAWN_INTERVAL);

// Event listeners
throwNetBtn.addEventListener('click', () => {
  toggleNetMode();
});

gameArea.addEventListener('click', onGameAreaClick);

// Keyboard accessibility: allow space/enter to toggle net mode when button focused
throwNetBtn.addEventListener('keydown', e => {
  if (e.key === ' ' || e.key === 'Enter') {
    e.preventDefault();
    toggleNetMode();
  }
});

// Initialize game
function init() {
  renderCollection();
  gameLoop();
  // Spawn initial creatures
  for (let i = 0; i < 5; i++) {
    spawnCreature();
  }
}
init();

// Creature class to manage individual creatures
class Creature {
  constructor(type, x, y) {
    this.type = type;
    this.x = x; // top-left x in px relative to game area
    this.y = y; // top-left y in px relative to game area
    this.speedX = (Math.random() * 0.6 + 0.2) * (Math.random() < 0.5 ? -1 : 1); // horizontal speed px/frame
    this.speedY = (Math.random() * 0.4 + 0.1) * (Math.random() < 0.5 ? -1 : 1); // vertical speed px/frame
    this.element = document.createElement('div');
    this.element.classList.add('creature');
    this.element.style.width = CREATURE_SIZE + 'px';
    this.element.style.height = CREATURE_SIZE + 'px';
    this.element.style.left = this.x + 'px';
    this.element.style.top = this.y + 'px';
    this.element.setAttribute('tabindex', '-1');
    this.element.setAttribute('aria-label', this.type.name + ' creature');
    this.element.setAttribute('role', 'img');
    this.element.style.position = 'absolute';
    this.element.style.userSelect = 'none';

    // Append SVG inside creature div
    const svg = createCreatureSVG(type);
    this.element.appendChild(svg);

    // Tooltip
    const tooltip = document.createElement('div');
    tooltip.classList.add('tooltip');
    tooltip.textContent = type.name;
    this.element.appendChild(tooltip);

    // Add to game area
    gameArea.appendChild(this.element);
  }

  // Update position based on speed and bounce off edges
  update() {
    this.x += this.speedX;
    this.y += this.speedY;

    // Bounce horizontally
    if (this.x <= 0) {
      this.x = 0;
      this.speedX = -this.speedX;
    } else if (this.x + CREATURE_SIZE >= gameArea.clientWidth) {
      this.x = gameArea.clientWidth - CREATURE_SIZE;
      this.speedX = -this.speedX;
    }
    // Bounce vertically
    if (this.y <= 0) {
      this.y = 0;
      this.speedY = -this.speedY;
    } else if (this.y + CREATURE_SIZE >= gameArea.clientHeight) {
      this.y = gameArea.clientHeight - CREATURE_SIZE;
      this.speedY = -this.speedY;
    }

    this.element.style.left = this.x + 'px';
    this.element.style.top = this.y + 'px';
  }

  // Get center coordinates of creature for collision detection
  getCenter() {
    return {
      x: this.x + CREATURE_SIZE / 2,
      y: this.y + CREATURE_SIZE / 2
    };
  }

  // Remove creature from DOM
  remove() {
    if (this.element.parentNode) {
      this.element.parentNode.removeChild(this.element);
    }
  }
}

// Spawn a new creature at a random position inside the game area
function spawnCreature() {
  if (creatures.length >= MAX_CREATURES) return;

  const type = CREATURE_TYPES[Math.floor(Math.random() * CREATURE_TYPES.length)];
  const maxX = gameArea.clientWidth - CREATURE_SIZE;
  const maxY = gameArea.clientHeight - CREATURE_SIZE;
  const x = Math.random() * maxX;
  const y = Math.random() * maxY;

  const creature = new Creature(type, x, y);
  creatures.push(creature);
}

// Update all creatures positions
function updateCreatures() {
  creatures.forEach(creature => creature.update());
}

// Capture creatures inside net radius at given coordinates
function captureCreaturesAt(x, y) {
  // Find creatures within NET_RADIUS of (x,y)
  const captured = [];
  creatures.forEach(creature => {
    const center = creature.getCenter();
    const dx = center.x - x;
    const dy = center.y - y;
    const distance = Math.sqrt(dx * dx + dy * dy);
    if (distance <= NET_RADIUS) {
      captured.push(creature);
    }
  });

  if (captured.length === 0) {
    return false; // no capture
  }

  // Remove captured creatures from game and add to collection
  captured.forEach(creature => {
    creature.remove();
    creatures = creatures.filter(c => c !== creature);
    addToCollection(creature.type.id);
  });

  return true;
}

// Add a creature type to the player's collection
function addToCollection(creatureId) {
  if (!collection[creatureId]) {
    collection[creatureId] = 0;
  }
  collection[creatureId]++;
  renderCollection();
}

// Render the collection panel

  collectionList.innerHTML = '';
  // Sort by name alphabetically

// Handle game area click when in net mode
function onGameAreaClick(event) {
  if (!netMode) return;

  const now = Date.now();
  if (now - lastNetThrowTime < NET_THROW_COOLDOWN) {
    // Cooldown active, ignore throw
    return;
  }
  lastNetThrowTime = now;

  // Get click coordinates relative to game area
  const rect = gameArea.getBoundingClientRect();
  const clickX = event.clientX - rect.left;
  const clickY = event.clientY - rect.top;

  // Show net visual at click position
  netVisual.style.left = clickX + 'px';
  netVisual.style.top = clickY + 'px';
  netVisual.style.display = 'block';

  // Animate net throw (scale up and fade out)
  netVisual.style.transition = 'none';
  netVisual.style.transform = 'translate(-50%, -50%) scale(0.5)';
  netVisual.style.opacity = '0.8';
  requestAnimationFrame(() => {
    netVisual.style.transition = 'transform 0.4s ease-out, opacity 0.4s ease-out';
    netVisual.style.transform = 'translate(-50%, -50%) scale(1.2)';
    netVisual.style.opacity = '0';

  // After animation, hide net visual
  setTimeout(() => {
    netVisual.style.display = 'none';
    netVisual.style.opacity = '0.8';
    netVisual.style.transform = 'translate(-50%, -50%) scale(1)';

  // Check for captures
  const captured = captureCreaturesAt(clickX, clickY);

  if (captured) {
    // Flash game area green briefly
    flashGameArea('#a0f7a0');

    // Flash game area red briefly

}

    this.speedY = (Math.random() * 0.4 + 0.1) * (Math.random() < 0.5 ? -1 : 1); // vertical speed px/frame
    this.element = document.createElement('div');
    this.element.classList.add('creature');
    this.element.style.width = CREATURE_SIZE + 'px';
    this.element.style.height = CREATURE_SIZE + 'px';
    this.element.style.left = this.x + 'px';
    this.element.style.top = this.y + 'px';
    this.element.setAttribute('tabindex', '-1');
    this.element.setAttribute('aria-label', this.type.name + ' creature');
    this.element.setAttribute('role', 'img');
    this.element.style.position = 'absolute';
    this.element.style.userSelect = 'none';

    // Append SVG inside creature div
    const svg = createCreatureSVG(type);
    this.element.appendChild(svg);

    // Tooltip
    const tooltip = document.createElement('div');
    tooltip.classList.add('tooltip');
    tooltip.textContent = type.name;
    this.element.appendChild(tooltip);

    // Add to game area
    gameArea.appendChild(this.element);
  }

  // Update position based on speed and bounce off edges
  update() {
    this.x += this.speedX;
    this.y += this.speedY;

    // Bounce horizontally
    if (this.x <= 0) {
      this.x = 0;
      this.speedX = -this.speedX;
    } else if (this.x + CREATURE_SIZE >= gameArea.clientWidth) {
      this.x = gameArea.clientWidth - CREATURE_SIZE;
      this.speedX = -this.speedX;
    }
    // Bounce vertically
    if (this.y <= 0) {
      this.y = 0;
      this.speedY = -this.speedY;
    } else if (this.y + CREATURE_SIZE >= gameArea.clientHeight) {
      this.y = gameArea.clientHeight - CREATURE_SIZE;
      this.speedY = -this.speedY;
    }

    this.element.style.left = this.x + 'px';
    this.element.style.top = this.y + 'px';
  }

  // Get center coordinates of creature for collision detection
  getCenter() {
    return {
      x: this.x + CREATURE_SIZE / 2,
      y: this.y + CREATURE_SIZE / 2
    };
  }

  // Remove creature from DOM
  remove() {
    if (this.element.parentNode) {
      this.element.parentNode.removeChild(this.element);
    }
  }
}

// Spawn a new creature at a random position inside the game area
function spawnCreature() {
  if (creatures.length >= MAX_CREATURES) return;

  const type = CREATURE_TYPES[Math.floor(Math.random() * CREATURE_TYPES.length)];
  const maxX = gameArea.clientWidth - CREATURE_SIZE;
  const maxY = gameArea.clientHeight - CREATURE_SIZE;
  const x = Math.random() * maxX;
  const y = Math.random() * maxY;

  const creature = new Creature(type, x, y);
  creatures.push(creature);
}

// Update all creatures positions
function updateCreatures() {
  creatures.forEach(creature => creature.update());
}

// Capture creatures inside net radius at given coordinates
function captureCreaturesAt(x, y) {
  // Find creatures within current tool radius of (x,y)
  const radius = TOOLS.find(t => t.id === activeTool)?.radius || NET_RADIUS;
  const captured = [];
  creatures.forEach(creature => {
    const center = creature.getCenter();
    const dx = center.x - x;
    const dy = center.y - y;
    const distance = Math.sqrt(dx * dx + dy * dy);
    if (distance <= radius) {
      captured.push(creature);
    }
  });

  if (captured.length === 0) {
    return false; // no capture
  }

  // Remove captured creatures from game and add to collection
  captured.forEach(creature => {
    creature.remove();
    creatures = creatures.filter(c => c !== creature);
    addToCollection(creature.type.id);
  });

  // Update total captures and XP
  totalCaptures += captured.length;
  gainXP(XP_PER_CAPTURE * captured.length);

  // Check achievements after capture
  checkAchievements();

  // Check collection milestones and rewards
  checkCollectionMilestones();

  return true;
}

// Add a creature type to the player's collection
function addToCollection(creatureId) {
  if (!collection[creatureId]) {
    collection[creatureId] = 0;
  }
  collection[creatureId]++;
  renderCollection();
}

// Render the collection panel
function renderCollection() {
  collectionList.innerHTML = '';
  // Sort by name alphabetically
  const sortedTypes = CREATURE_TYPES.slice().sort((a, b) => a.name.localeCompare(b.name));
  sortedTypes.forEach(type => {
    if (collection[type.id]) {
      const count = collection[type.id];
      const item = document.createElement('div');
      item.classList.add('collected-item');
      item.setAttribute('aria-label', `${type.name} x${count}`);

      // Create SVG icon for collection item (smaller)
      const svg = createCreatureSVG(type);
      svg.setAttribute('width', '40');
      svg.setAttribute('height', '40');
      item.appendChild(svg);

      // Count badge
      const countBadge = document.createElement('div');
      countBadge.classList.add('collected-count');
      countBadge.textContent = count;
      item.appendChild(countBadge);

      collectionList.appendChild(item);
    }
  });
}

// Gain XP and handle leveling up
function gainXP(amount) {
  currentXP += amount;
  while (currentXP >= xpToNextLevel) {
    currentXP -= xpToNextLevel;
    levelUp();
  }
  updateXPBar();
  updateLevelDisplay();
  updateUnlockedTools();
}

// Level up the player
function levelUp() {
  level++;
  xpToNextLevel = Math.floor(xpToNextLevel * LEVEL_XP_MULTIPLIER);
  flashGameArea('#a0f7a0'); // green flash on level up
  checkAchievements();
  updateUnlockedTools();
}

// Update XP bar UI
function updateXPBar() {
  const percent = Math.min(100, (currentXP / xpToNextLevel) * 100);
  xpBar.style.width = percent + '%';
  xpBar.setAttribute('aria-valuenow', Math.floor(percent));
}

// Update level display UI
function updateLevelDisplay() {
  levelDisplay.textContent = `Level: ${level}`;
}

// Update unlocked tools UI and enable/disable throw net button accordingly
function updateUnlockedTools() {
  // Unlock tools based on level
  TOOLS.forEach(tool => {
    if (level >= tool.unlockedAtLevel && !unlockedTools.has(tool.id)) {
      unlockedTools.add(tool.id);
      addToolToUI(tool);
      showAchievementNotification(`Unlocked tool: ${tool.name}!`);
    }
  });

  // Update tools list UI highlight for active tool
  const toolItems = unlockedToolsList.querySelectorAll('.tool-item');
  toolItems.forEach(item => {
    if (item.dataset.tool === activeTool) {
      item.style.fontWeight = '700';
      item.style.backgroundColor = '#a0f7a0';
      item.style.borderColor = '#007700';
    } else {
      item.style.fontWeight = '400';
      item.style.backgroundColor = '#e0f4ff';
      item.style.borderColor = '#0077cc';
    }
  });

  // Enable or disable throw net button based on unlocked tools
  throwNetBtn.disabled = unlockedTools.size === 0;
}

// Add a tool to the unlocked tools UI list
function addToolToUI(tool) {
  // Avoid duplicates
  if (unlockedToolsList.querySelector(`[data-tool="${tool.id}"]`)) return;

  const toolItem = document.createElement('div');
  toolItem.classList.add('tool-item');
  toolItem.textContent = tool.name;
  toolItem.dataset.tool = tool.id;
  toolItem.title = tool.description;
  toolItem.tabIndex = 0;
  toolItem.setAttribute('role', 'button');
  toolItem.setAttribute('aria-pressed', tool.id === activeTool ? 'true' : 'false');

  // Click to select tool
  toolItem.addEventListener('click', () => {
    if (activeTool !== tool.id) {
      activeTool = tool.id;
      updateUnlockedTools();
      if (netMode) {
        toggleNetMode(); // cancel net mode if active
      }
    }
  });
  // Keyboard accessibility
  toolItem.addEventListener('keydown', e => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      toolItem.click();
    }
  });

  unlockedToolsList.appendChild(toolItem);
}

// Check and unlock achievements
function checkAchievements() {
  ACHIEVEMENTS.forEach(achievement => {
    if (!unlockedAchievements.has(achievement.id) && achievement.condition({
      level,
      collection,
      totalCaptures
    })) {
      unlockedAchievements.add(achievement.id);
      addAchievementToUI(achievement);
      showAchievementNotification(`Achievement unlocked: ${achievement.name}!`);
    }
  });
}

// Check collection milestones and provide rewards
function checkCollectionMilestones() {

  Object.entries(collection).forEach(([creatureId, count]) => {
    if (count === 3 && !milestoneReached(`3-${creatureId}`)) {

      gainXP(20); // bonus XP reward
      showAchievementNotification(`Milestone reached: Collected 3 ${getCreatureName(creatureId)}! Bonus XP awarded.`);
    }
  });

  // Milestone: complete collection handled by achievement
}

// Handle game area click when in net mode
function onGameAreaClick(event) {
  if (!netMode) return;

  const now = Date.now();
  const toolConfig = TOOLS.find(t => t.id === activeTool) || TOOLS[0];
  if (now - lastNetThrowTime < toolConfig.cooldown) {
    // Cooldown active, ignore throw
    return;
  }
  lastNetThrowTime = now;

  // Get click coordinates relative to game area
  const rect = gameArea.getBoundingClientRect();
  const clickX = event.clientX - rect.left;
  const clickY = event.clientY - rect.top;

  // Show net visual at click position
  netVisual.style.left = clickX + 'px';
  netVisual.style.top = clickY + 'px';
  netVisual.style.display = 'block';

  // Animate net throw (scale up and fade out)
  netVisual.style.transition = 'none';
  netVisual.style.transform = 'translate(-50%, -50%) scale(0.5)';
  netVisual.style.opacity = '0.8';
  requestAnimationFrame(() => {
    netVisual.style.transition = 'transform 0.4s ease-out, opacity 0.4s ease-out';
    netVisual.style.transform = 'translate(-50%, -50%) scale(1.2)';
    netVisual.style.opacity = '0';
  });

  // After animation, hide net visual
  setTimeout(() => {
    netVisual.style.display = 'none';
    netVisual.style.opacity = '0.8';
    netVisual.style.transform = 'translate(-50%, -50%) scale(1)';
  }, 400);

  // Check for captures
  const captured = captureCreaturesAt(clickX, clickY);

  // Provide feedback
  if (captured) {
    // Flash game area green briefly
    flashGameArea('#a0f7a0');
  } else {
    // Flash game area red briefly
    flashGameArea('#f7a0a0');
  }

  // Exit net mode after throw
  toggleNetMode();
}

// Flash the game area background color briefly for feedback
function flashGameArea(color) {
  const original = gameArea.style.backgroundColor || '#d0f0ff';
  gameArea.style.backgroundColor = color;
  setTimeout(() => {
    gameArea.style.backgroundColor = original;
  }, 300);
}

// Game loop for updating creatures
function gameLoop() {
  updateCreatures();
  requestAnimationFrame(gameLoop);
}
