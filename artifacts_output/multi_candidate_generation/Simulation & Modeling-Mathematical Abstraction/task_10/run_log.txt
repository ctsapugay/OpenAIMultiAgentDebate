  const LON_DIV = 180; // longitude divisions (from -180 to +180)

    requestAnimationFrame(render);
  }

  function getMouseLatLon(clientX, clientY) {
    const rect = canvas.getBoundingClientRect();
    const x = ((clientX - rect.left) / rect.width) * 2 - 1;
    const y = -(((clientY - rect.top) / rect.height) * 2 - 1);

    // Inverse MVP matrix to get ray direction in world space
    const aspect = canvas.width / canvas.height;
    let proj = perspective(Math.PI/4, aspect, 0.1, 100);
    let view = createIdentity();
    view = translate(view, [0, 0, -3]);
    view = rotateX(view, rotationX);
    view = rotateY(view, rotationY);
    const mvp = multiplyMat4(proj, view);

    // Invert MVP matrix
    const invMVP = invertMat4(mvp);
    if(!invMVP) return null;

    // Near and far points in clip space
    const nearPoint = [x, y, -1, 1];
    const farPoint = [x, y, 1, 1];

    // Transform to world space
    const nearWorld = multiplyMat4Vec4(invMVP, nearPoint);
    const farWorld = multiplyMat4Vec4(invMVP, farPoint);

    // Perspective divide
    for(let i=0; i<3; i++) {
      nearWorld[i] /= nearWorld[3];
      farWorld[i] /= farWorld[3];
    }

    // Ray direction
    const dir = [
      farWorld[0] - nearWorld[0],
      farWorld[1] - nearWorld[1],
      farWorld[2] - nearWorld[2]
    ];
    const dirLen = Math.hypot(...dir);
    if(dirLen === 0) return null;
    const dirNorm = dir.map(v => v / dirLen);

    // Ray origin
    const origin = nearWorld.slice(0,3);

    // Ray-sphere intersection (sphere radius 1 at origin)
    const t = raySphereIntersect(origin, dirNorm, [0,0,0], 1);
    if(t === null) return null;

    // Intersection point
    const ix = origin[0] + dirNorm[0]*t;
    const iy = origin[1] + dirNorm[1]*t;
    const iz = origin[2] + dirNorm[2]*t;

    // Convert to lat/lon
    const lat = Math.asin(iy); // -π/2 to π/2
    const lon = Math.atan2(iz, ix); // -π to π

    // Convert to indices
    let latIndex = Math.round((lat + Math.PI/2) / Math.PI * (LAT_DIV - 1));
    let lonIndex = Math.round((lon + Math.PI) / (2 * Math.PI) * (LON_DIV - 1));

    latIndex = Math.min(Math.max(latIndex, 0), LAT_DIV - 1);
    lonIndex = (lonIndex + LON_DIV) % LON_DIV;

    return {latIndex, lonIndex};

    const a = dot(dir, dir);
    const b = 2 * dot(oc, dir);
    const c = dot(oc, oc) - radius*radius;
    const discriminant = b*b - 4*a*c;
    if(discriminant < 0) return null;
    const sqrtD = Math.sqrt(discriminant);
    const t1 = (-b - sqrtD) / (2*a);
    const t2 = (-b + sqrtD) / (2*a);
    if(t1 >= 0) return t1;
    if(t2 >= 0) return t2;
    return null;
