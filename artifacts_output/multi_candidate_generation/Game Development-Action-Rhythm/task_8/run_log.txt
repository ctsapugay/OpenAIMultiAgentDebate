        let colorIndex = frame[y][x];
        if (colorIndex === 0) continue;
        let color = COLORS[colorIndex];
        if (highlight === 'correct') color = COLORS[7];
        else if (highlight === 'wrong') color = COLORS[8];
        ctx.fillStyle = color;
        ctx.fillRect(x * pixelSize, y * pixelSize, pixelSize, pixelSize);
      }
    }
  }

  // Arrow SVG icons for prompts
  const ARROW_SVGS = {
    left: `<svg viewBox="0 0 24 24" aria-hidden="true" focusable="false"><path d="M15 19l-7-7 7-7"/></svg>`,
    down: `<svg viewBox="0 0 24 24" aria-hidden="true" focusable="false"><path d="M19 9l-7 7-7-7"/></svg>`,

    promptsContainer.innerHTML = '';
    directions.forEach(dir => {

      div.classList.add('arrow');
      div.dataset.direction = dir;

  function activatePrompts() {
    const arrows = promptsContainer.querySelectorAll('.arrow');
    arrows.forEach(arrow => arrow.classList.add('active'));

  // Generate a random dance step (1-3 directions)
  function generateStep() {
    const count = Math.floor(Math.random() * 3) + 1; // 1 to 3 arrows
    const stepDirs = [];
    while (stepDirs.length < count) {
      const dir = DIRECTIONS[Math.floor(Math.random() * DIRECTIONS.length)];
      if (!stepDirs.includes(dir)) stepDirs.push(dir);
    }
    return stepDirs;
  }

  // Generate full dance sequence
  function generateSequence() {
    danceSequence = [];
    for (let i = 0; i < SEQUENCE_LENGTH; i++) {
      danceSequence.push(generateStep());
    }
  }

  // Play background music (simple beat)
  // We'll generate a simple beep rhythm using Web Audio API
  function playMusic() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    if (musicSource) {
      musicSource.stop();
      musicSource.disconnect();
      musicSource = null;
    }
    // Create a simple beat sound buffer
    const sampleRate = audioCtx.sampleRate;
    const duration = 0.1; // 100ms beep
    const frameCount = sampleRate * duration;
    const buffer = audioCtx.createBuffer(1, frameCount, sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < frameCount; i++) {
      // Simple sine wave beep 440Hz with fade out
      data[i] = Math.sin(2 * Math.PI * 440 * i / sampleRate) * (1 - i / frameCount);
    }
    musicBuffer = buffer;

    // Create a buffer source and loop it every BEAT_INTERVAL
    musicSource = audioCtx.createBufferSource();
    musicSource.buffer = musicBuffer;
    musicSource.loop = true;
    musicSource.loopStart = 0;
    musicSource.loopEnd = duration;
    musicSource.connect(audioCtx.destination);
    musicSource.start();

    musicStartTime = audioCtx.currentTime;
  }

    if (musicSource) {
      musicSource.stop();
      musicSource.disconnect();
      musicSource = null;
    }

    if (!audioCtx) return 0;
    return (audioCtx.currentTime - musicStartTime) * 1000;

  function gameLoop(timestamp) {
    if (!gameStarted) return;

    const now = getMusicTime();

    // Check if it's time for next beat
    if (now - lastBeatTime >= BEAT_INTERVAL) {
      lastBeatTime += BEAT_INTERVAL;

      // Show next prompt

        currentStep++;
      } else {
        // Sequence finished - end game
        endGame();
        return;
      }
    }

    animateDancer(now);

    animationFrameId = requestAnimationFrame(gameLoop);
  }

  // Animate dancer frames synced to beat
  function animateDancer(now) {
    if (!currentPrompt) {
      drawFrame(frames[0]);
      return;
    }
    // Calculate frame index based on time within beat interval
    const beatProgress = ((now - lastBeatTime) % BEAT_INTERVAL) / BEAT_INTERVAL;
    // 8 frames per beat cycle
    const frameIndex = Math.floor(beatProgress * 8) % 8;
    drawFrame(frames[frameIndex]);
  }

  // Handle player input
  function handleInput(e) {
    if (!gameStarted) return;

    if (e.repeat) return; // ignore held keys

    const key = e.key;

    if (key === ' ') {
      // Space pressed - check if input matches prompt
      if (!awaitingSpace) return;

      // Check timing window
      const now = getMusicTime();
      const delta = Math.abs(now - lastBeatTime);
      if (delta > INPUT_WINDOW) {
        // Too early or late
        failStep();
        return;
      }

      // Check if all directions were input correctly
      if (arraysEqualUnordered(inputBuffer, currentPrompt)) {
        // Success
        successStep();
      } else {
        // Fail
        failStep();
      }
    } else if (KEY_TO_DIR[key]) {
      // Direction key pressed
      if (!awaitingSpace) return;

      // Add to input buffer if not already present
      const dir = KEY_TO_DIR[key];
      if (!inputBuffer.includes(dir)) {
        inputBuffer.push(dir);
        // Highlight pressed arrow
        highlightPressedArrow(dir);
      }
    }
  }

  // Highlight pressed arrow in prompts
  function highlightPressedArrow(dir) {
    const arrow = promptsContainer.querySelector(`.arrow[data-direction="${dir}"]`);
    if (!arrow) return;
    arrow.classList.add('active');
  }

  // Compare two arrays ignoring order
  function arraysEqualUnordered(a, b) {
    if (a.length !== b.length) return false;
    const sortedA = [...a].sort();
    const sortedB = [...b].sort();
    for (let i = 0; i < sortedA.length; i++) {
      if (sortedA[i] !== sortedB[i]) return false;
    }
    return true;
  }

  // Success step feedback
  function successStep() {
    awaitingSpace = false;
    combo++;
    score += 100 + combo * 10;
    updateScore();
    infoDisplay.textContent = `Great! Combo x${combo}`;
    // Highlight prompts green
    currentPrompt.forEach(dir => highlightPrompt(dir, 'correct'));
    // Flash dancer green
    flashDancer('correct');
    clearTimeout(promptTimeout);
  }

  // Fail step feedback
  function failStep() {
    awaitingSpace = false;
    combo = 0;
    infoDisplay.textContent = `Miss! Try to keep the rhythm!`;
    // Highlight prompts red
    if (currentPrompt) currentPrompt.forEach(dir => highlightPrompt(dir, 'wrong'));
    // Flash dancer red
    flashDancer('wrong');
    clearTimeout(promptTimeout);
  }

  // Flash dancer with highlight color for 400ms
  let flashTimeout = null;
  function flashDancer(type) {
    drawFrame(frames[0], type);
    if (flashTimeout) clearTimeout(flashTimeout);
    flashTimeout = setTimeout(() => {
      drawFrame(frames[0]);
    }, 400);
  }

  // Update score display
  function updateScore() {
    scoreDisplay.textContent = `Score: ${score}`;
  }

  // Start game
  function startGame() {
    if (gameStarted) return;
    gameStarted = true;
    score = 0;
    combo = 0;
    currentStep = 0;
    updateScore();
    infoDisplay.textContent = 'Get ready...';
    generateSequence();
    playMusic();
    lastBeatTime = getMusicTime();
    currentPrompt = null;
    inputBuffer = [];
    awaitingSpace = false;
    promptsContainer.innerHTML = '';
    startScreen.style.display = 'none';
    animationFrameId = requestAnimationFrame(gameLoop);
  }

  // End game
  function endGame() {
    gameStarted = false;
    stopMusic();
    promptsContainer.innerHTML = '';
    infoDisplay.textContent = `Game Over! Final Score: ${score}`;
    startScreen.style.display = 'flex';
    startBtn.textContent = 'Restart';
    drawFrame(frames[0]);
    if (animationFrameId) cancelAnimationFrame(animationFrameId);
  }

  // Event listeners
  window.addEventListener('keydown', handleInput);
  startBtn.addEventListener('click', () => {
    if (!audioCtx) {
      // Resume audio context on user interaction
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      audioCtx.resume();
    }
    startGame();
  });

  // Initial draw
  drawFrame(frames[0]);

  // Accessibility: focus start button on load
  window.addEventListener('load', () => {
    startBtn.focus();
  });
})();

    const pixelSize = 2;
    ctx.clearRect(0, 0, dancerCanvas.width, dancerCanvas.height);
    for (let y = 0; y < 16; y++) {
      for (let x = 0; x < 16; x++) {
        let colorIndex = frame[y][x];
        if (colorIndex === 0) continue;
        let color = COLORS[colorIndex];
        if (highlight === 'correct') color = COLORS[7];
        else if (highlight === 'wrong') color = COLORS[8];
        ctx.fillStyle = color;
        ctx.fillRect(x * pixelSize, y * pixelSize, pixelSize, pixelSize);
      }
    }
  }

    directions.forEach(dir => {

      div.setAttribute('role', 'listitem');

      div.innerHTML = ARROW_SVGS[dir];
      promptsContainer.appendChild(div);
    });
  }

  function highlightPrompt(direction, result) {
    const arrows = promptsContainer.querySelectorAll('.arrow');
    arrows.forEach(arrow => {
      if (arrow.dataset.direction === direction) {
        arrow.classList.remove('active');
        arrow.classList.add(result);
      }
    });
  }

  function clearPromptHighlights() {
    const arrows = promptsContainer.querySelectorAll('.arrow');
    arrows.forEach(arrow => {
      arrow.classList.remove('correct', 'wrong', 'active');
    });
  }

  function activatePrompts() {
    const arrows = promptsContainer.querySelectorAll('.arrow');
    arrows.forEach(arrow => arrow.classList.add('active'));
  }

  function generateStep() {
    const count = Math.floor(Math.random() * 3) + 1;
    const stepDirs = [];
    while (stepDirs.length < count) {
      const dir = DIRECTIONS[Math.floor(Math.random() * DIRECTIONS.length)];
      if (!stepDirs.includes(dir)) stepDirs.push(dir);
    }
    return stepDirs;
  }

  function generateSequence() {
    danceSequence = [];
    for (let i = 0; i < SEQUENCE_LENGTH; i++) {
      danceSequence.push(generateStep());
    }
  }

  // Play background music (simple beep)
  function playMusic() {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    if (musicSource) {
      musicSource.stop();
      musicSource.disconnect();
      musicSource = null;
    }
    const sampleRate = audioCtx.sampleRate;
    const duration = 0.1;
    const frameCount = sampleRate * duration;
    const buffer = audioCtx.createBuffer(1, frameCount, sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < frameCount; i++) {
      data[i] = Math.sin(2 * Math.PI * 440 * i / sampleRate) * (1 - i / frameCount);
    }
    musicBuffer = buffer;

    musicSource = audioCtx.createBufferSource();
    musicSource.buffer = musicBuffer;
    musicSource.loop = true;
    musicSource.loopStart = 0;
    musicSource.loopEnd = duration;
    musicSource.connect(audioCtx.destination);
    musicSource.start();

    musicStartTime = audioCtx.currentTime;
  }

  function stopMusic() {
    if (musicSource) {
      musicSource.stop();
      musicSource.disconnect();
      musicSource = null;
    }
  }

  function getMusicTime() {
    if (!audioCtx) return 0;
    return (audioCtx.currentTime - musicStartTime) * 1000;
  }

  function gameLoop(timestamp) {
    if (!gameStarted) return;

    const now = getMusicTime();

    if (now - lastBeatTime >= BEAT_INTERVAL) {
      lastBeatTime += BEAT_INTERVAL;

      if (currentStep < danceSequence.length) {
        currentPrompt = danceSequence[currentStep];
        createPromptElements(currentPrompt);
        activatePrompts();
        awaitingSpace = true;
        inputBuffer = [];
        clearPromptHighlights();

        if (promptTimeout) clearTimeout(promptTimeout);
        promptTimeout = setTimeout(() => {
          if (awaitingSpace) {
            failStep();
          }
        }, PROMPT_DISPLAY_TIME);

        // Record step for replay
        if (!replayMode) {
          replayData.push({
            step: currentStep,
            prompt: currentPrompt.slice(),
            timestamp: now,
            input: [],
            success: null,
          });
        }

        currentStep++;
      } else {
        endGame();
        return;
      }
    }

    animateDancer(now);

    animationFrameId = requestAnimationFrame(gameLoop);
  }

  function animateDancer(now) {
    if (!currentPrompt) {
      drawFrame(frames[0]);
      return;
    }
    const beatProgress = ((now - lastBeatTime) % BEAT_INTERVAL) / BEAT_INTERVAL;
    const frameIndex = Math.floor(beatProgress * 8) % 8;
    drawFrame(frames[frameIndex]);
  }

  function handleInput(e) {
    if (!gameStarted) return;
    if (e.repeat) return;

    const key = e.key;

    if (key === ' ') {
      if (!awaitingSpace) return;

      const now = getMusicTime();
      const delta = Math.abs(now - lastBeatTime);
      if (delta > INPUT_WINDOW) {
        failStep();
        return;
      }

      if (arraysEqualUnordered(inputBuffer, currentPrompt)) {
        successStep();
      } else {
        failStep();
      }
    } else if (KEY_TO_DIR[key]) {
      if (!awaitingSpace) return;

      const dir = KEY_TO_DIR[key];
      if (!inputBuffer.includes(dir)) {
        inputBuffer.push(dir);
        highlightPressedArrow(dir);
      }
    }
  }

  function highlightPressedArrow(dir) {
    const arrow = promptsContainer.querySelector(`.arrow[data-direction="${dir}"]`);
    if (!arrow) return;
    arrow.classList.add('active');
  }

  function arraysEqualUnordered(a, b) {
    if (a.length !== b.length) return false;
    const sortedA = [...a].sort();
    const sortedB = [...b].sort();
    for (let i = 0; i < sortedA.length; i++) {
      if (sortedA[i] !== sortedB[i]) return false;
    }
    return true;
  }

  function successStep() {
    awaitingSpace = false;
    combo++;
    score += 100 + combo * 10;
    updateScore();
    infoDisplay.textContent = `Great! Combo x${combo}`;
    currentPrompt.forEach(dir => highlightPrompt(dir, 'correct'));
    flashDancer('correct');
    clearTimeout(promptTimeout);

    // Record success for replay
    if (!replayMode && replayData.length > 0) {
      replayData[replayData.length - 1].input = inputBuffer.slice();
      replayData[replayData.length - 1].success = true;
    }
  }

  function failStep() {
    awaitingSpace = false;
    combo = 0;
    infoDisplay.textContent = `Miss! Try to keep the rhythm!`;
    if (currentPrompt) currentPrompt.forEach(dir => highlightPrompt(dir, 'wrong'));
    flashDancer('wrong');

    if (!replayMode && replayData.length > 0) {
      replayData[replayData.length - 1].input = inputBuffer.slice();
      replayData[replayData.length - 1].success = false;
    }

  function updateScore() {

      const stepData = replayData[replayIndex];
      if (now >= stepData.timestamp) {
        currentPrompt = stepData.prompt;

        clearPromptHighlights();
        // Highlight inputs and results
        if (stepData.input && stepData.input.length > 0) {
          stepData.input.forEach(dir => highlightPressedArrow(dir));
        }
        if (stepData.success === true) {
          stepData.prompt.forEach(dir => highlightPrompt(dir, 'correct'));
          combo++;
          score += 100 + combo * 10;
        } else if (stepData.success === false) {
          stepData.prompt.forEach(dir => highlightPrompt(dir, 'wrong'));
          combo = 0;
        }
        updateScore();
        replayIndex++;
        lastBeatTime = stepData.timestamp;
      }
    } else {
      // Replay finished
      replayMode = false;
      stopMusic();
      infoDisplay.textContent = `Replay finished. Final Score: ${score}`;
      startScreen.style.display = 'flex';
      startBtn.textContent = 'Restart';
      replayBtn.style.display = 'inline-block';
      drawFrame(frames[0]);
      return;
    }

    animateDancer(now);
    animationFrameId = requestAnimationFrame(replayLoop);
  }
