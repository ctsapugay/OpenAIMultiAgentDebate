          chosenType = enemyTypes[i];
          break;
        }
      }

      enemies.push(new Enemy(chosenType, level));
    }

  function circleCollision(x1, y1, r1, x2, y2, r2) {
    return distance(x1, y1, x2, y2) < r1 + r2;

  function handleCollisions() {
    // Player bullets vs enemies
    bullets.forEach(bullet => {
      if (bullet.owner !== 'player') return;
      enemies.forEach(enemy => {
        if (!enemy.isDead && circleCollision(bullet.x, bullet.y, bullet.radius, enemy.x, enemy.y, enemy.radius)) {
          enemy.health -= bullet.damage;
          bullet.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += 10 * level;
          }
        }
      });
    });

    // Enemy bullets vs player
    enemyBullets.forEach(bullet => {
      if (bullet.owner !== 'enemy') return;
      if (target && circleCollision(bullet.x, bullet.y, bullet.radius, target.x, target.y, target.radius)) {
        target.health -= bullet.damage;
        bullet.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });

    // Enemies vs player (collision)
    enemies.forEach(enemy => {
      if (!enemy.isDead && target && circleCollision(enemy.x, enemy.y, enemy.radius, target.x, target.y, target.radius)) {
        // Both take damage
        target.health -= enemy.damage * 1.5;
        enemy.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });
  }

    return Math.hypot(x2 - x1, y2 - y1);

    return a + (b - a) * t;

      this.energy += this.energyRegen * delta;

  // Spawn enemies based on level and time
  function spawnEnemies(delta) {
    const now = performance.now();
    if (now - lastEnemySpawn > enemySpawnInterval) {
      lastEnemySpawn = now;

      // Determine enemy type probabilities based on level
      const enemyTypes = ['basic', 'fast', 'tank', 'shooter'];
      let weights = [60, 25, 10, 5]; // base weights
      // Increase weights for tougher enemies as level grows
      weights = weights.map((w, i) => w + i * level * 2);

      // Normalize weights
      const totalWeight = weights.reduce((a, b) => a + b, 0);
      const rand = Math.random() * totalWeight;
      let sum = 0;
      let chosenType = 'basic';
      for (let i = 0; i < enemyTypes.length; i++) {
        sum += weights[i];
        if (rand <= sum) {
          chosenType = enemyTypes[i];
          break;
        }
      }

      enemies.push(new Enemy(chosenType, level));
    }
  }

  // Collision detection between circles
  function circleCollision(x1, y1, r1, x2, y2, r2) {
    return distance(x1, y1, x2, y2) < r1 + r2;
  }

  // Handle all collisions
  function handleCollisions() {
    // Player bullets vs enemies
    bullets.forEach(bullet => {
      if (bullet.owner !== 'player') return;
      enemies.forEach(enemy => {
        if (!enemy.isDead && circleCollision(bullet.x, bullet.y, bullet.radius, enemy.x, enemy.y, enemy.radius)) {
          enemy.health -= bullet.damage;
          bullet.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += 10 * level;
          }
        }
      });
    });

    // Enemy bullets vs player
    enemyBullets.forEach(bullet => {
      if (bullet.owner !== 'enemy') return;
      if (target && circleCollision(bullet.x, bullet.y, bullet.radius, target.x, target.y, target.radius)) {
        target.health -= bullet.damage;
        bullet.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });

    // Enemies vs player (collision)
    enemies.forEach(enemy => {
      if (!enemy.isDead && target && circleCollision(enemy.x, enemy.y, enemy.radius, target.x, target.y, target.radius)) {
        // Both take damage
        target.health -= enemy.damage * 1.5;
        enemy.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });
  }

      ctx.save();
      ctx.globalAlpha = Math.max(this.life / this.maxLife, 0);
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 10;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
    isDead() {
      return this.life <= 0;
    }
  }

  // Target class - the single controllable unit
  class Target {
    constructor() {
      this.x = GAME_WIDTH / 2;
      this.y = GAME_HEIGHT - EARTH_RADIUS - 100;
      this.radius = TARGET_RADIUS;
      this.speed = upgrades.speed;
      this.color = '#0ff';
      this.angle = -Math.PI / 2; // Facing upward initially
      this.energy = energy;
      this.health = health;
      this.damage = upgrades.damage;
      this.fireRate = upgrades.fireRate;
      this.lastShot = 0;
      this.energyRegen = upgrades.energyRegen;
      this.maxHealth = upgrades.maxHealth;
      this.maxEnergy = upgrades.maxEnergy;
    }
    update(delta) {
      // Movement based on keys
      let dx = 0, dy = 0;
      if (keys.left) dx -= 1;
      if (keys.right) dx += 1;
      if (keys.up) dy -= 1;
      if (keys.down) dy += 1;

      // Normalize diagonal movement
      if (dx !== 0 && dy !== 0) {
        dx *= Math.SQRT1_2;
        dy *= Math.SQRT1_2;
      }

      // Apply weather effect on speed (solar flare reduces speed)
      let speedModifier = 1;
      if (weather.active && weather.type === 'solarFlare') {
        speedModifier = lerp(1, 0.5, weather.intensity);
      }

      this.x += dx * this.speed * speedModifier;
      this.y += dy * this.speed * speedModifier;

      // Clamp inside canvas and above Earth
      const minY = 20;
      const maxY = EARTH_POSITION.y - EARTH_RADIUS - this.radius - 10;
      this.x = clamp(this.x, this.radius + 10, GAME_WIDTH - this.radius - 10);
      this.y = clamp(this.y, minY, maxY);

      // Update angle to face upward (fixed)
      this.angle = -Math.PI / 2;

      // Regenerate energy with possible weather effect (solar flare reduces regen)
      let energyRegenModifier = 1;
      if (weather.active && weather.type === 'solarFlare') {
        energyRegenModifier = lerp(1, 0.3, weather.intensity);
      }
      this.energy += this.energyRegen * delta * energyRegenModifier;
      if (this.energy > this.maxEnergy) this.energy = this.maxEnergy;

      // Update health and energy globals
      health = this.health;
      energy = this.energy;
    }
    canShoot() {
      const now = performance.now();
      return (now - this.lastShot) >= (shotCooldown / this.fireRate) && this.energy >= 10;
    }
    shoot() {
      if (!this.canShoot()) return false;
      this.lastShot = performance.now();
      this.energy -= 10;
      // Create a bullet going upward from target position
      bullets.push(new Bullet(this.x, this.y - this.radius - 8, -Math.PI / 2, this.damage, 'player'));
      return true;
    }
    draw(ctx) {
      // Draw target body - a glowing triangle pointing upward
      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.rotate(this.angle);
      // Glow
      ctx.shadowColor = '#0ff';
      ctx.shadowBlur = 12;
      ctx.fillStyle = '#0ff';
      ctx.beginPath();
      ctx.moveTo(0, -this.radius);
      ctx.lineTo(this.radius * 0.7, this.radius);
      ctx.lineTo(-this.radius * 0.7, this.radius);
      ctx.closePath();
      ctx.fill();
      ctx.shadowBlur = 0;

      // Inner shape
      ctx.fillStyle = '#004d4d';
      ctx.beginPath();
      ctx.moveTo(0, -this.radius * 0.7);
      ctx.lineTo(this.radius * 0.5, this.radius * 0.7);
      ctx.lineTo(-this.radius * 0.5, this.radius * 0.7);
      ctx.closePath();
      ctx.fill();

      ctx.restore();

      // Energy bar above target
      const barWidth = 50;
      const barHeight = 6;
      const energyRatio = this.energy / this.maxEnergy;
      ctx.fillStyle = '#003322';
      ctx.fillRect(this.x - barWidth / 2, this.y - this.radius - 20, barWidth, barHeight);
      ctx.fillStyle = '#0ff';
      ctx.fillRect(this.x - barWidth / 2, this.y - this.radius - 20, barWidth * energyRatio, barHeight);
      ctx.strokeStyle = '#0ff';
      ctx.lineWidth = 1;
      ctx.strokeRect(this.x - barWidth / 2, this.y - this.radius - 20, barWidth, barHeight);

      // Health bar below target
      const healthRatio = this.health / this.maxHealth;
      ctx.fillStyle = '#330000';
      ctx.fillRect(this.x - barWidth / 2, this.y + this.radius + 10, barWidth, barHeight);
      ctx.fillStyle = '#f00';
      ctx.fillRect(this.x - barWidth / 2, this.y + this.radius + 10, barWidth * healthRatio, barHeight);
      ctx.strokeStyle = '#f00';
      ctx.lineWidth = 1;
      ctx.strokeRect(this.x - barWidth / 2, this.y + this.radius + 10, barWidth, barHeight);
    }
  }

  // Bullet class for player and enemy projectiles
  class Bullet {
    constructor(x, y, angle, damage, owner) {
      this.x = x;
      this.y = y;
      this.angle = angle;
      this.speed = 7;
      this.radius = 4;
      this.damage = damage;
      this.owner = owner; // 'player' or 'enemy'
      this.color = owner === 'player' ? '#0ff' : '#f44';
      this.isDead = false;
    }
    update(delta) {
      this.x += Math.cos(this.angle) * this.speed;
      this.y += Math.sin(this.angle) * this.speed;

      // Remove if out of bounds
      if (this.x < -this.radius || this.x > GAME_WIDTH + this.radius ||
          this.y < -this.radius || this.y > GAME_HEIGHT + this.radius) {
        this.isDead = true;
      }
    }
    draw(ctx) {
      ctx.save();
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 8;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;
      ctx.restore();
    }
  }

  // Enemy class
  class Enemy {
    constructor(type, level) {
      this.type = type; // 'basic', 'fast', 'tank', 'shooter'
      this.level = level;
      this.isDead = false;
      this.spawnTime = performance.now();

      // Enemy stats based on type and level
      switch (type) {
        case 'basic':
          this.radius = 14;
          this.speed = 1 + 0.1 * level;
          this.health = 10 + 5 * level;
          this.maxHealth = this.health;
          this.color = '#f55';
          this.damage = 10;
          this.canShoot = false;
          break;
        case 'fast':
          this.radius = 12;
          this.speed = 2 + 0.15 * level;
          this.health = 6 + 3 * level;
          this.maxHealth = this.health;
          this.color = '#fa0';
          this.damage = 8;
          this.canShoot = false;
          break;
        case 'tank':
          this.radius = 22;
          this.speed = 0.6 + 0.05 * level;
          this.health = 30 + 15 * level;
          this.maxHealth = this.health;
          this.color = '#a00';
          this.damage = 20;
          this.canShoot = false;
          break;
        case 'shooter':
          this.radius = 16;
          this.speed = 1 + 0.1 * level;
          this.health = 15 + 7 * level;
          this.maxHealth = this.health;
          this.color = '#f0f';
          this.damage = 12;
          this.canShoot = true;
          this.lastShot = 0;
          this.shootInterval = 2500 - Math.min(level * 100, 1500);
          break;
        default:
          this.radius = 14;
          this.speed = 1;
          this.health = 10;
          this.maxHealth = 10;
          this.color = '#f55';
          this.damage = 10;
          this.canShoot = false;
      }

      // Spawn position: random x at top edge, y = -radius
      this.x = randomRange(this.radius + 20, GAME_WIDTH - this.radius - 20);
      this.y = -this.radius;

      // Movement path: enemies move roughly downward toward Earth center
      this.targetX = EARTH_POSITION.x + randomRange(-EARTH_RADIUS / 2, EARTH_RADIUS / 2);
      this.targetY = EARTH_POSITION.y + EARTH_RADIUS / 2;
    }
    update(delta) {
      // Move toward target position (Earth)
      const angleToEarth = angleBetween(this.x, this.y, EARTH_POSITION.x, EARTH_POSITION.y);

      // Apply weather effect on enemy speed (meteor shower slows enemies)
      let speedModifier = 1;
      if (weather.active && weather.type === 'meteorShower') {
        speedModifier = lerp(1, 0.6, weather.intensity);
      }

      this.x += Math.cos(angleToEarth) * this.speed * speedModifier;
      this.y += Math.sin(angleToEarth) * this.speed * speedModifier;

      // Shooting behavior for shooter enemies
      if (this.canShoot) {
        const now = performance.now();
        if (now - this.lastShot > this.shootInterval) {
          this.lastShot = now;
          // Shoot bullet toward target (player)
          if (target) {
            const angleToTarget = angleBetween(this.x, this.y, target.x, target.y);
            enemyBullets.push(new Bullet(this.x, this.y, angleToTarget, this.damage, 'enemy'));
          }
        }
      }

      // Check if reached Earth (collision)
      if (distance(this.x, this.y, EARTH_POSITION.x, EARTH_POSITION.y) < EARTH_RADIUS + this.radius) {
        // Damage Earth and die
        health -= this.damage;
        this.isDead = true;
        if (health < 0) health = 0;
      }
    }
    draw(ctx) {
      // Draw enemy body with glow
      ctx.save();
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 10;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;

      // Health bar above enemy
      const barWidth = this.radius * 2;
      const barHeight = 4;
      const healthRatio = this.health / this.maxHealth;
      ctx.fillStyle = '#330000';
      ctx.fillRect(this.x - this.radius, this.y - this.radius - 10, barWidth, barHeight);
      ctx.fillStyle = '#f00';
      ctx.fillRect(this.x - this.radius, this.y - this.radius - 10, barWidth * healthRatio, barHeight);
      ctx.strokeStyle = '#f00';
      ctx.lineWidth = 1;
      ctx.strokeRect(this.x - this.radius, this.y - this.radius - 10, barWidth, barHeight);

      ctx.restore();
    }
  }

  // Earth drawing function
  function drawEarth(ctx) {
    const gradient = ctx.createRadialGradient(
      EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS * 0.3,
      EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS
    );
    gradient.addColorStop(0, '#0a4');
    gradient.addColorStop(0.7, '#030');
    gradient.addColorStop(1, '#001100');
    ctx.fillStyle = gradient;
    ctx.beginPath();
    ctx.arc(EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS, 0, Math.PI * 2);
    ctx.fill();

    // Earth glow
    ctx.save();
    ctx.shadowColor = '#0f0';
    ctx.shadowBlur = 20;
    ctx.beginPath();
    ctx.arc(EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS + 6, 0, Math.PI * 2);
    ctx.strokeStyle = '#0f0a';
    ctx.lineWidth = 4;
    ctx.stroke();
    ctx.restore();

    // Simple continents shape (abstract)
    ctx.fillStyle = '#073';
    ctx.beginPath();
    ctx.moveTo(EARTH_POSITION.x - 20, EARTH_POSITION.y - 10);
    ctx.bezierCurveTo(EARTH_POSITION.x - 10, EARTH_POSITION.y - 30, EARTH_POSITION.x + 10, EARTH_POSITION.y - 30, EARTH_POSITION.x + 20, EARTH_POSITION.y - 10);
    ctx.bezierCurveTo(EARTH_POSITION.x + 30, EARTH_POSITION.y + 10, EARTH_POSITION.x + 10, EARTH_POSITION.y + 30, EARTH_POSITION.x - 20, EARTH_POSITION.y + 10);
    ctx.closePath();
    ctx.fill();
  }

  // Spawn enemies based on level and time
  function spawnEnemies(delta) {
    const now = performance.now();
    if (now - lastEnemySpawn > enemySpawnInterval) {
      lastEnemySpawn = now;

      // Determine enemy type probabilities based on level
      const enemyTypes = ['basic', 'fast', 'tank', 'shooter'];
      let weights = [60, 25, 10, 5]; // base weights
      // Increase weights for tougher enemies as level grows
      weights = weights.map((w, i) => w + i * level * 2);

      // Normalize weights
      const totalWeight = weights.reduce((a, b) => a + b, 0);
      const rand = Math.random() * totalWeight;
      let sum = 0;
      let chosenType = 'basic';
      for (let i = 0; i < enemyTypes.length; i++) {
        sum += weights[i];
        if (rand <= sum) {
          chosenType = enemyTypes[i];
          break;
        }
      }

      enemies.push(new Enemy(chosenType, level));
    }
  }

  // Collision detection between circles
  function circleCollision(x1, y1, r1, x2, y2, r2) {
    return distance(x1, y1, x2, y2) < r1 + r2;
  }

  // Handle all collisions
  function handleCollisions() {
    // Player bullets vs enemies
    bullets.forEach(bullet => {
      if (bullet.owner !== 'player') return;
      enemies.forEach(enemy => {
        if (!enemy.isDead && circleCollision(bullet.x, bullet.y, bullet.radius, enemy.x, enemy.y, enemy.radius)) {
          enemy.health -= bullet.damage;
          bullet.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += 10 * level;
          }
        }
      });
    });

    // Enemy bullets vs player
    enemyBullets.forEach(bullet => {
      if (bullet.owner !== 'enemy') return;
      if (target && circleCollision(bullet.x, bullet.y, bullet.radius, target.x, target.y, target.radius)) {
        target.health -= bullet.damage;
        bullet.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });

    // Enemies vs player (collision)
    enemies.forEach(enemy => {
      if (!enemy.isDead && target && circleCollision(enemy.x, enemy.y, enemy.radius, target.x, target.y, target.radius)) {
        // Both take damage
        target.health -= enemy.damage * 1.5;
        enemy.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });
  }

  // Remove dead entities
  function cleanupEntities() {
    bullets = bullets.filter(b => !b.isDead);
    enemyBullets = enemyBullets.filter(b => !b.isDead);
    enemies = enemies.filter(e => !e.isDead);
  }

  // Game loop
  function gameLoop(timestamp = 0) {
    if (!lastFrameTime) lastFrameTime = timestamp;
    const delta = (timestamp - lastFrameTime) / 16.6667; // approx frames (60fps base)
    lastFrameTime = timestamp;

    if (gameState === 'playing') {
      update(delta);
      draw();
      updateWeather(delta);
      drawWeather();
    } else {
      // Still draw weather effects if active
      if (weather.active) {
        updateWeather(delta);
        drawWeather();
      } else {
        clearWeatherCanvas();
      }
    }

    requestAnimationFrame(gameLoop);
  }

  // Update game state
  function update(delta) {
    // Spawn enemies
    spawnEnemies(delta);

    // Update target
    target.update(delta);

    // Update bullets
    bullets.forEach(b => b.update(delta));
    enemyBullets.forEach(b => b.update(delta));

    // Update enemies
    enemies.forEach(e => e.update(delta));

    // Handle collisions
    handleCollisions();

    // Cleanup dead entities
    cleanupEntities();

    // Update UI
    updateUI();

    // Check level progression
    if (score >= level * 100) {
      level++;
      enemySpawnInterval = Math.max(600, 2000 - level * 150);
      shotCooldown = Math.max(100, 300 - level * 15);
      levelUp();
    }

    // Check health
    if (health <= 0) {
      gameOver();
    }
  }

  // Draw game state
  function draw() {
    // Clear canvas
    ctx.clearRect(0, 0, GAME_WIDTH, GAME_HEIGHT);

    // Draw Earth
    drawEarth(ctx);

    // Draw enemies
    enemies.forEach(e => e.draw(ctx));

    // Draw bullets
    bullets.forEach(b => b.draw(ctx));
    enemyBullets.forEach(b => b.draw(ctx));

    // Draw target
    target.draw(ctx);
  }

  // Accessibility: focus trap for dialogs
  function trapFocus(element) {
    const focusableElements = element.querySelectorAll('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])');
    const firstFocusable = focusableElements[0];
    const lastFocusable = focusableElements[focusableElements.length - 1];
    element.addEventListener('keydown', e => {

      this.radius = TARGET_RADIUS;
      this.speed = upgrades.speed;
      this.color = '#0ff';
      this.angle = -Math.PI / 2; // Facing upward initially
      this.energy = energy;
      this.health = health;
      this.damage = upgrades.damage;
      this.fireRate = upgrades.fireRate;
      this.lastShot = 0;
      this.energyRegen = upgrades.energyRegen;
      this.maxHealth = upgrades.maxHealth;
      this.maxEnergy = upgrades.maxEnergy;
    }
    update(delta) {
      // Movement based on keys
      let dx = 0, dy = 0;
      if (keys.left) dx -= 1;
      if (keys.right) dx += 1;
      if (keys.up) dy -= 1;
      if (keys.down) dy += 1;

      // Normalize diagonal movement
      if (dx !== 0 && dy !== 0) {
        dx *= Math.SQRT1_2;
        dy *= Math.SQRT1_2;
      }

      // Apply weather effect on speed (solar flare reduces speed)
      let speedModifier = 1;
      if (weather.active && weather.type === 'solarFlare') {
        speedModifier = lerp(1, 0.5, weather.intensity);
      }

      this.x += dx * this.speed * speedModifier;
      this.y += dy * this.speed * speedModifier;

      // Clamp inside canvas and above Earth
      const minY = 20;
      const maxY = EARTH_POSITION.y - EARTH_RADIUS - this.radius - 10;
      this.x = clamp(this.x, this.radius + 10, GAME_WIDTH - this.radius - 10);
      this.y = clamp(this.y, minY, maxY);

      // Update angle to face upward (fixed)
      this.angle = -Math.PI / 2;

      // Regenerate energy with possible weather effect (solar flare reduces regen)
      let energyRegenModifier = 1;
      if (weather.active && weather.type === 'solarFlare') {
        energyRegenModifier = lerp(1, 0.3, weather.intensity);
      }
      this.energy += this.energyRegen * delta * energyRegenModifier;
      if (this.energy > this.maxEnergy) this.energy = this.maxEnergy;

      // Update health and energy globals
      health = this.health;
      energy = this.energy;
    }
    canShoot() {
      const now = performance.now();
      return (now - this.lastShot) >= (shotCooldown / this.fireRate) && this.energy >= 10;
    }
    shoot() {
      if (!this.canShoot()) return false;
      this.lastShot = performance.now();
      this.energy -= 10;
      // Create a bullet going upward from target position
      bullets.push(new Bullet(this.x, this.y - this.radius - 8, -Math.PI / 2, this.damage, 'player'));
      return true;
    }
    draw(ctx) {
      // Draw target body - a glowing triangle pointing upward
      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.rotate(this.angle);
      // Glow
      ctx.shadowColor = '#0ff';
      ctx.shadowBlur = 12;
      ctx.fillStyle = '#0ff';
      ctx.beginPath();
      ctx.moveTo(0, -this.radius);
      ctx.lineTo(this.radius * 0.7, this.radius);
      ctx.lineTo(-this.radius * 0.7, this.radius);
      ctx.closePath();
      ctx.fill();
      ctx.shadowBlur = 0;

      // Inner shape
      ctx.fillStyle = '#004d4d';
      ctx.beginPath();
      ctx.moveTo(0, -this.radius * 0.7);
      ctx.lineTo(this.radius * 0.5, this.radius * 0.7);
      ctx.lineTo(-this.radius * 0.5, this.radius * 0.7);
      ctx.closePath();
      ctx.fill();

      ctx.restore();

      // Energy bar above target
      const barWidth = 50;
      const barHeight = 6;
      const energyRatio = this.energy / this.maxEnergy;
      ctx.fillStyle = '#003322';
      ctx.fillRect(this.x - barWidth / 2, this.y - this.radius - 20, barWidth, barHeight);
      ctx.fillStyle = '#0ff';
      ctx.fillRect(this.x - barWidth / 2, this.y - this.radius - 20, barWidth * energyRatio, barHeight);
      ctx.strokeStyle = '#0ff';
      ctx.lineWidth = 1;
      ctx.strokeRect(this.x - barWidth / 2, this.y - this.radius - 20, barWidth, barHeight);

      // Health bar below target
      const healthRatio = this.health / this.maxHealth;
      ctx.fillStyle = '#330000';
      ctx.fillRect(this.x - barWidth / 2, this.y + this.radius + 10, barWidth, barHeight);
      ctx.fillStyle = '#f00';
      ctx.fillRect(this.x - barWidth / 2, this.y + this.radius + 10, barWidth * healthRatio, barHeight);
      ctx.strokeStyle = '#f00';
      ctx.lineWidth = 1;
      ctx.strokeRect(this.x - barWidth / 2, this.y + this.radius + 10, barWidth, barHeight);
    }
  }

  // Bullet class for player and enemy projectiles
  class Bullet {
    constructor(x, y, angle, damage, owner) {
      this.x = x;
      this.y = y;
      this.angle = angle;
      this.speed = 7;
      this.radius = 4;
      this.damage = damage;
      this.owner = owner; // 'player' or 'enemy'
      this.color = owner === 'player' ? '#0ff' : '#f44';
      this.isDead = false;
    }
    update(delta) {
      this.x += Math.cos(this.angle) * this.speed;
      this.y += Math.sin(this.angle) * this.speed;

      // Remove if out of bounds
      if (this.x < -this.radius || this.x > GAME_WIDTH + this.radius ||
          this.y < -this.radius || this.y > GAME_HEIGHT + this.radius) {
        this.isDead = true;
      }
    }
    draw(ctx) {
      ctx.save();
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 8;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;
      ctx.restore();
    }
  }

  // Enemy class
  class Enemy {
    constructor(type, level) {
      this.type = type; // 'basic', 'fast', 'tank', 'shooter'
      this.level = level;
      this.isDead = false;
      this.spawnTime = performance.now();

      // Enemy stats based on type and level
      switch (type) {
        case 'basic':
          this.radius = 14;
          this.speed = 1 + 0.1 * level;
          this.health = 10 + 5 * level;
          this.maxHealth = this.health;
          this.color = '#f55';
          this.damage = 10;
          this.canShoot = false;
          break;
        case 'fast':
          this.radius = 12;
          this.speed = 2 + 0.15 * level;
          this.health = 6 + 3 * level;
          this.maxHealth = this.health;
          this.color = '#fa0';
          this.damage = 8;
          this.canShoot = false;
          break;
        case 'tank':
          this.radius = 22;
          this.speed = 0.6 + 0.05 * level;
          this.health = 30 + 15 * level;
          this.maxHealth = this.health;
          this.color = '#a00';
          this.damage = 20;
          this.canShoot = false;
          break;
        case 'shooter':
          this.radius = 16;
          this.speed = 1 + 0.1 * level;
          this.health = 15 + 7 * level;
          this.maxHealth = this.health;
          this.color = '#f0f';
          this.damage = 12;
          this.canShoot = true;
          this.lastShot = 0;
          this.shootInterval = 2500 - Math.min(level * 100, 1500);
          break;
        default:
          this.radius = 14;
          this.speed = 1;
          this.health = 10;
          this.maxHealth = 10;
          this.color = '#f55';
          this.damage = 10;
          this.canShoot = false;
      }

      // Spawn position: random x at top edge, y = -radius
      this.x = randomRange(this.radius + 20, GAME_WIDTH - this.radius - 20);
      this.y = -this.radius;

      // Movement path: enemies move roughly downward toward Earth center
      this.targetX = EARTH_POSITION.x + randomRange(-EARTH_RADIUS / 2, EARTH_RADIUS / 2);
      this.targetY = EARTH_POSITION.y + EARTH_RADIUS / 2;
    }
    update(delta) {
      // Move toward target position (Earth)
      const angleToEarth = angleBetween(this.x, this.y, EARTH_POSITION.x, EARTH_POSITION.y);

      // Apply weather effect on enemy speed (meteor shower slows enemies)
      let speedModifier = 1;
      if (weather.active && weather.type === 'meteorShower') {
        speedModifier = lerp(1, 0.6, weather.intensity);
      }

      this.x += Math.cos(angleToEarth) * this.speed * speedModifier;
      this.y += Math.sin(angleToEarth) * this.speed * speedModifier;

      // Shooting behavior for shooter enemies
      if (this.canShoot) {
        const now = performance.now();
        if (now - this.lastShot > this.shootInterval) {
          this.lastShot = now;
          // Shoot bullet toward target (player)
          if (target) {
            const angleToTarget = angleBetween(this.x, this.y, target.x, target.y);
            enemyBullets.push(new Bullet(this.x, this.y, angleToTarget, this.damage, 'enemy'));
          }
        }
      }

      // Check if reached Earth (collision)
      if (distance(this.x, this.y, EARTH_POSITION.x, EARTH_POSITION.y) < EARTH_RADIUS + this.radius) {
        // Damage Earth and die
        health -= this.damage;
        this.isDead = true;
        if (health < 0) health = 0;
      }
    }
    draw(ctx) {
      // Draw enemy body with glow
      ctx.save();
      ctx.fillStyle = this.color;
      ctx.shadowColor = this.color;
      ctx.shadowBlur = 10;
      ctx.beginPath();
      ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;

      // Health bar above enemy
      const barWidth = this.radius * 2;
      const barHeight = 4;
      const healthRatio = this.health / this.maxHealth;
      ctx.fillStyle = '#330000';
      ctx.fillRect(this.x - this.radius, this.y - this.radius - 10, barWidth, barHeight);
      ctx.fillStyle = '#f00';
      ctx.fillRect(this.x - this.radius, this.y - this.radius - 10, barWidth * healthRatio, barHeight);
      ctx.strokeStyle = '#f00';
      ctx.lineWidth = 1;
      ctx.strokeRect(this.x - this.radius, this.y - this.radius - 10, barWidth, barHeight);

      ctx.restore();
    }
  }

  // Earth drawing function
  function drawEarth(ctx) {
    const gradient = ctx.createRadialGradient(
      EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS * 0.3,
      EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS
    );
    gradient.addColorStop(0, '#0a4');
    gradient.addColorStop(0.7, '#030');
    gradient.addColorStop(1, '#001100');
    ctx.fillStyle = gradient;
    ctx.beginPath();
    ctx.arc(EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS, 0, Math.PI * 2);
    ctx.fill();

    // Earth glow
    ctx.save();
    ctx.shadowColor = '#0f0';
    ctx.shadowBlur = 20;
    ctx.beginPath();
    ctx.arc(EARTH_POSITION.x, EARTH_POSITION.y, EARTH_RADIUS + 6, 0, Math.PI * 2);
    ctx.strokeStyle = '#0f0a';
    ctx.lineWidth = 4;
    ctx.stroke();
    ctx.restore();

    // Simple continents shape (abstract)
    ctx.fillStyle = '#073';
    ctx.beginPath();
    ctx.moveTo(EARTH_POSITION.x - 20, EARTH_POSITION.y - 10);
    ctx.bezierCurveTo(EARTH_POSITION.x - 10, EARTH_POSITION.y - 30, EARTH_POSITION.x + 10, EARTH_POSITION.y - 30, EARTH_POSITION.x + 20, EARTH_POSITION.y - 10);
    ctx.bezierCurveTo(EARTH_POSITION.x + 30, EARTH_POSITION.y + 10, EARTH_POSITION.x + 10, EARTH_POSITION.y + 30, EARTH_POSITION.x - 20, EARTH_POSITION.y + 10);
    ctx.closePath();
    ctx.fill();
  }

  // Spawn enemies based on level and time
  function spawnEnemies(delta) {
    const now = performance.now();
    if (now - lastEnemySpawn > enemySpawnInterval) {
      lastEnemySpawn = now;

      // Determine enemy type probabilities based on level
      const enemyTypes = ['basic', 'fast', 'tank', 'shooter'];
      let weights = [60, 25, 10, 5]; // base weights
      // Increase weights for tougher enemies as level grows
      weights = weights.map((w, i) => w + i * level * 2);

      // Normalize weights
      const totalWeight = weights.reduce((a, b) => a + b, 0);
      const rand = Math.random() * totalWeight;
      let sum = 0;
      let chosenType = 'basic';
      for (let i = 0; i < enemyTypes.length; i++) {
        sum += weights[i];
        if (rand <= sum) {
          chosenType = enemyTypes[i];
          break;
        }
      }

      enemies.push(new Enemy(chosenType, level));
    }
  }

  // Collision detection between circles
  function circleCollision(x1, y1, r1, x2, y2, r2) {
    return distance(x1, y1, x2, y2) < r1 + r2;
  }

  // Handle all collisions
  function handleCollisions() {
    // Player bullets vs enemies
    bullets.forEach(bullet => {
      if (bullet.owner !== 'player') return;
      enemies.forEach(enemy => {
        if (!enemy.isDead && circleCollision(bullet.x, bullet.y, bullet.radius, enemy.x, enemy.y, enemy.radius)) {
          enemy.health -= bullet.damage;
          bullet.isDead = true;
          if (enemy.health <= 0) {
            enemy.isDead = true;
            score += 10 * level;
          }
        }
      });
    });

    // Enemy bullets vs player
    enemyBullets.forEach(bullet => {
      if (bullet.owner !== 'enemy') return;
      if (target && circleCollision(bullet.x, bullet.y, bullet.radius, target.x, target.y, target.radius)) {
        target.health -= bullet.damage;
        bullet.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });

    // Enemies vs player (collision)
    enemies.forEach(enemy => {
      if (!enemy.isDead && target && circleCollision(enemy.x, enemy.y, enemy.radius, target.x, target.y, target.radius)) {
        // Both take damage
        target.health -= enemy.damage * 1.5;
        enemy.isDead = true;
        if (target.health <= 0) {
          target.health = 0;
          gameOver();
        }
      }
    });
  }

  // Remove dead entities
  function cleanupEntities() {
    bullets = bullets.filter(b => !b.isDead);
    enemyBullets = enemyBullets.filter(b => !b.isDead);
    enemies = enemies.filter(e => !e.isDead);
  }

  // Game loop
  function gameLoop(timestamp = 0) {
    if (!lastFrameTime) lastFrameTime = timestamp;
    const delta = (timestamp - lastFrameTime) / 16.6667; // approx frames (60fps base)
    lastFrameTime = timestamp;

    if (gameState === 'playing') {
      update(delta);
      draw();
      updateWeather(delta);
      drawWeather();
    } else {
      // Still draw weather effects if active
      if (weather.active) {
        updateWeather(delta);
        drawWeather();
      } else {
        clearWeatherCanvas();
      }
    }

    requestAnimationFrame(gameLoop);
  }

  // Update game state
  function update(delta) {
    // Spawn enemies
    spawnEnemies(delta);

    // Update target
    target.update(delta);

    // Update bullets
    bullets.forEach(b => b.update(delta));
    enemyBullets.forEach(b => b.update(delta));

    // Update enemies
    enemies.forEach(e => e.update(delta));

    // Handle collisions
    handleCollisions();

    // Cleanup dead entities
    cleanupEntities();

    // Update UI
    updateUI();

    // Check level progression
    if (score >= level * 100) {
      level++;
      enemySpawnInterval = Math.max(600, 2000 - level * 150);
      shotCooldown = Math.max(100, 300 - level * 15);
      levelUp();
    }

    // Check health
    if (health <= 0) {
      gameOver();
    }
  }

  // Draw game state
  function draw() {
    // Clear canvas
    ctx.clearRect(0, 0, GAME_WIDTH, GAME_HEIGHT);

    // Draw Earth
    drawEarth(ctx);

    // Draw enemies
    enemies.forEach(e => e.draw(ctx));

    // Draw bullets
    bullets.forEach(b => b.draw(ctx));
    enemyBullets.forEach(b => b.draw(ctx));

    // Draw target
    target.draw(ctx);
  }

  // Accessibility: focus trap for dialogs
  function trapFocus(element) {
    const focusableElements = element.querySelectorAll('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])');
    const firstFocusable = focusableElements[0];
    const lastFocusable = focusableElements[focusableElements.length - 1];
    element.addEventListener('keydown', e => {
      if (e.key === 'Tab') {
        if (e.shiftKey) {
          if (document.activeElement === firstFocusable) {
            e.preventDefault();
            lastFocusable.focus();
          }
        } else {
          if (document.activeElement === lastFocusable) {
            e.preventDefault();
            firstFocusable.focus();
          }
        }
      }
    });
  }
  trapFocus(startScreen);
  trapFocus(upgradeScreen);

  // Handle window resize to keep canvas responsive and update game dimensions

    const rect = container.getBoundingClientRect();
    // Maintain aspect ratio 3:2 (900x600)
    let width = rect.width;
    let height = rect.height;
    const aspectRatio = 3 / 2;
    if (width / height > aspectRatio) {
      width = height * aspectRatio;
    } else {
      height = width / aspectRatio;
    }
    canvas.style.width = width + 'px';
    canvas.style.height = height + 'px';
    weatherCanvas.style.width = width + 'px';
    weatherCanvas.style.height = height + 'px';

    // Update internal canvas size for weather overlay to match displayed size
    weatherCanvas.width = width;
    weatherCanvas.height = height;

    // Update game dimensions and earth position accordingly
    GAME_WIDTH = width;
    GAME_HEIGHT = height;
    EARTH_POSITION = { x: GAME_WIDTH / 2, y: GAME_HEIGHT - EARTH_RADIUS - 20 };

    // Clamp target position if exists
    if (target) {
      target.x = clamp(target.x, target.radius + 10, GAME_WIDTH - target.radius - 10);
      target.y = clamp(target.y, 20, EARTH_POSITION.y - EARTH_RADIUS - target.radius - 10);
    }
  }

  window.addEventListener('resize', () => {
    resizeCanvas();
  });

  // Initialize canvas size on load
  resizeCanvas();

  // Weather system functions

  // Reset weather to none
  function resetWeather() {
    weather.type = 'none';
    weather.intensity = 0;
    weather.timer = 0;
    weather.duration = 0;
    weather.active = false;
    weather.particles = [];
  }

  // Trigger a random weather event on level up
  function triggerRandomWeather() {
    const rand = Math.random();
    if (rand < 0.4) {
      startSolarFlare();
    } else if (rand < 0.8) {
      startMeteorShower();
    } else {
      resetWeather();
    }
  }

  // Solar flare event: reduces player speed and energy regen for a duration
  function startSolarFlare() {
    weather.type = 'solarFlare';
    weather.intensity = 0;
    weather.timer = 0;
    weather.duration = 15000; // 15 seconds
    weather.active = true;
    weather.particles = [];
  }

  // Meteor shower event: slows enemies and shows meteor particles
  function startMeteorShower() {
    weather.type = 'meteorShower';
    weather.intensity = 0;
    weather.timer = 0;
    weather.duration = 15000; // 15 seconds
    weather.active = true;
    weather.particles = [];
  }

  // Update weather effects each frame
  function updateWeather(delta) {
    if (!weather.active) return;

    weather.timer += delta * 16.6667; // convert delta frames to ms approx

    // Increase intensity in first 3 seconds, then decrease
    if (weather.timer < 3000) {
      weather.intensity = lerp(0, 1, weather.timer / 3000);
    } else if (weather.timer > weather.duration - 3000) {
      weather.intensity = lerp(1, 0, (weather.timer - (weather.duration - 3000)) / 3000);
    } else {
      weather.intensity = 1;
    }

    // Update particles for meteor shower
    if (weather.type === 'meteorShower') {
      // Spawn new meteors based on intensity
      const spawnRate = lerp(0, 0.3, weather.intensity);
      if (Math.random() < spawnRate) {
        const x = randomRange(0, weatherCanvas.width);
        const y = -10;
        const vx = randomRange(-0.5, 0.5);
        const vy = randomRange(4, 7);
        const size = randomRange(1, 3);
        const color = 'rgba(255, 255, 255, 0.8)';
        const life = weatherCanvas.height / vy + 20;
        weather.particles.push(new Particle(x, y, vx, vy, size, color, life));
      }
      // Update particles
      weather.particles.forEach(p => p.update(delta));
      // Remove dead particles
      weather.particles = weather.particles.filter(p => !p.isDead());
    }

    // End weather event after duration
    if (weather.timer >= weather.duration) {
      resetWeather();
    }
  }

  // Draw weather effects overlay
  function drawWeather() {
    weatherCtx.clearRect(0, 0, weatherCanvas.width, weatherCanvas.height);

    if (!weather.active) return;

    if (weather.type === 'solarFlare') {
      // Draw solar flare glow overlay with pulsating effect
      const alpha = 0.15 * weather.intensity * (0.7 + 0.3 * Math.sin(Date.now() / 300));
      const gradient = weatherCtx.createRadialGradient(
        weatherCanvas.width / 2, weatherCanvas.height / 2, weatherCanvas.height / 4,
        weatherCanvas.width / 2, weatherCanvas.height / 2, weatherCanvas.height / 1.2
      );
      gradient.addColorStop(0, `rgba(255, 100, 0, ${alpha})`);
      gradient.addColorStop(1, 'rgba(0,0,0,0)');
      weatherCtx.fillStyle = gradient;
      weatherCtx.fillRect(0, 0, weatherCanvas.width, weatherCanvas.height);
    } else if (weather.type === 'meteorShower') {
      // Draw meteor particles
      weather.particles.forEach(p => p.draw(weatherCtx));
    }
  }

  // Clear weather canvas
  function clearWeatherCanvas() {
    weatherCtx.clearRect(0, 0, weatherCanvas.width, weatherCanvas.height);
  }

  // Initialize
  showStartScreen();
  shootingHandler();
  requestAnimationFrame(gameLoop);

  // Expose startGame and continueGame for buttons
  startButton.addEventListener('click', () => {
    startGame();
  });
  upgradeContinueBtn.addEventListener('click', () => {
    continueGame();
  });

  // Event listeners
  window.addEventListener('keydown', keyDownHandler);
  window.addEventListener('keyup', keyUpHandler);
})();
